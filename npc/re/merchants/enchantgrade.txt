grademk,34,184,4	script	Sratos#sratos	4_JP_GARM_H,{

	mes "[Sratos]";
	mes "Hello, dear customer who walked in, let's hope for a miracle today.";
	mes "How can I help you?";
	next;

	.@menu$ = "Enhance the equipment's grade.:Exchange Etel items:What is Grade Enhancement?";

	// Show special menu if player has Grade Ticket
	if (countitem(30055))
		.@menu$ += ":Enchant with Ticket";

	switch(select(.@menu$)) {

	//=========================================================
	// NORMAL GRADE ENCHANT
	//=========================================================
	case 1:

		mes "[Sratos]";
		mes "Our customer. You want to enhance the grade of equipment.";
		mes "It is not easy to move the magical power of jewels.";
		next;

		mes "[Sratos]";
		mes "Magical power can explode on the spot.";
		mes "Then the weapon that will inherit the magical power also explodes!";
		next;

		switch(select("I'll still do it!:I'll think about it.")) {

			case 1:

				mes "[Sratos]";
				mes "I wish good luck to our courageous customers!";
				close2;

				enchantgradeui();
				end;

			case 2:

				mes "[Sratos]";
				mes "Whenever you have the courage to challenge, please come back.";
				close;
		}
		end;

	//=========================================================
	// EXCHANGE ETEL
	//=========================================================
	case 2:

		mes "[Sratos]";
		mes "These are jewels used to enhance grades.";
		mes "Etel Dust, the jewels that will be the base, and if you give me a small fee I'll exchange it for Etel jewels.";
		close2;

		callshop "EnchantGradeExchange";
		end;

	//=========================================================
	// INFORMATION
	//=========================================================
	case 3:

		mes "[Sratos]";
		mes "Occasionally, unstable jewels with a lot of pure magical power are found.";
		mes "We call these etheric gems, right?";
		next;

		mes "[Sratos]";
		mes "You can transfer the magical power of the etheric gem to other weapons, but of course there is some risk.";
		mes "It's delicate, so if you aren't careful, pop! It will explode.";
		next;

		mes "[Sratos]";
		mes "Anyway, if you use this etheric gem, you can enhance the weapon you are using.";
		mes "Performance gets better. We call this grade enhancement.";
		next;

		mes "[Sratos]";
		mes "You too, wouldn't you be happy if the weapons you love grow one step further?";
		mes "We are the people who assist it. It's a bit risky, though. ahahaha.";
		close;

	//=========================================================
	// ENCHANT WITH TICKET
	//=========================================================
case 4:

	if (!countitem(30055)) {
		mes "[Sratos]";
		mes "You don't have a Grade Ticket.";
		close;
	}

	mes "[Sratos]";
	mes "Select equipment slot.";
	next;

	//=====================================================
	// SLOT DATA
	//=====================================================

	setarray .@eqslot,
		EQI_HEAD_LOW,
		EQI_HAND_R,
		EQI_GARMENT,
		EQI_ACC_R,
		EQI_ARMOR,
		EQI_HAND_L,
		EQI_SHOES,
		EQI_ACC_L,
		EQI_HEAD_TOP,
		EQI_HEAD_MID;

	setarray .@gradepos,
		1,
		2,
		4,
		8,
		16,
		32,
		64,
		128,
		256,
		512;

	setarray .@slotname$,
		"Lower Headgear",
		"Right Hand",
		"Garment",
		"Right Accessory",
		"Body Armor",
		"Left Hand",
		"Shoes",
		"Left Accessory",
		"Top Headgear",
		"Mid Headgear";

	.@menu$ = "";
	.@count = 0;

	deletearray .@selectedpos;
	deletearray .@selectedname$;

	//=====================================================
	// BUILD MENU
	//=====================================================

	for (.@i = 0; .@i < getarraysize(.@eqslot); .@i++) {

		.@id = getequipid(.@eqslot[.@i]);

		if (!.@id)
			continue;

		.@weaponlv = getiteminfo(.@id, ITEMINFO_WEAPONLEVEL);
		.@armorlv  = getiteminfo(.@id, ITEMINFO_ARMORLEVEL);

		// Gradeable Weapon
		if (.@weaponlv == 5) {

			.@menu$ +=
				"["+.@slotname$[.@i]+"] "+
				getitemname(.@id)+":";

			.@selectedpos[.@count] = .@gradepos[.@i];
			.@selectedname$[.@count] = getitemname(.@id);

			.@count++;
			continue;
		}

		// Gradeable Armor
		if (.@armorlv == 2) {

			.@menu$ +=
				"["+.@slotname$[.@i]+"] "+
				getitemname(.@id)+":";

			.@selectedpos[.@count] = .@gradepos[.@i];
			.@selectedname$[.@count] = getitemname(.@id);

			.@count++;
		}
	}

	if (.@count == 0) {

		mes "[Sratos]";
		mes "No gradeable equipment equipped.";
		close;
	}

	.@sel = select(.@menu$) - 1;

	.@gradeposition = .@selectedpos[.@sel];

	mes "[Sratos]";
	mes "Selected:";
	mes "^00FF00"+.@selectedname$[.@sel]+"^000000";
	next;

	if (select("Use Ticket:Cancel") == 2)
		close;

	delitem 30055,1;

	dispbottom "1 Grade Ticket consumed.";

	close2;

	//=====================================================
	// OPEN GRADE UI
	//=====================================================

		// Open grade UI directly
		atcommand "@grade " + .@gradeposition + " +4";
		end;
	}

	end;

//=========================================================
// SELECT GRADEABLE EQUIPMENT
//=========================================================
S_SelectEquipment:

	.@menu$ = "";
	.@count = 0;

	deletearray .@eqindex;
	deletearray .@eqname$;

	for (.@i = 0; .@i < 20; .@i++) {

		if (!getequipisequiped(.@i))
			continue;

		.@id = getequipid(.@i);

		if (!.@id)
			continue;

		.@weaponlv = getiteminfo(.@id, ITEMINFO_WEAPONLEVEL);
		.@armorlv  = getiteminfo(.@id, ITEMINFO_ARMORLEVEL);

		// Gradeable weapon
		if (.@weaponlv == 5) {

			.@menu$ += "[Weapon] "+getitemname(.@id)+":";

			.@eqindex[.@count] = .@i;
			.@eqname$[.@count] = getitemname(.@id);
			.@count++;

			continue;
		}

		// Gradeable armor + shadow gear
		if (.@armorlv == 2) {

			.@menu$ += "[Armor] "+getitemname(.@id)+":";

			.@eqindex[.@count] = .@i;
			.@eqname$[.@count] = getitemname(.@id);
			.@count++;

			continue;
		}
	}

	if (.@count == 0) {

		mes "[Sratos]";
		mes "No gradeable equipment equipped.";
		.@eqslot = -1;
		return;
	}

	.@sel = select(.@menu$) - 1;

	.@eqslot = .@eqindex[.@sel];

	return;

//=========================================================
// INIT
//=========================================================
OnInit:

	setunittitle(getnpcid(0), "<Grade Enhancer>");

	// Equipment slots
	setarray .slots,
		EQI_HEAD_LOW,
		EQI_HAND_R,
		EQI_GARMENT,
		EQI_ACC_L,
		EQI_ARMOR,
		EQI_HAND_L,
		EQI_SHOES,
		EQI_ACC_R,
		EQI_HEAD_TOP,
		EQI_HEAD_MID;

	setarray .slotname$,
		"Lower Headgear",
		"Right Hand",
		"Garment",
		"Left Accessory",
		"Body Armor",
		"Left Hand",
		"Shoes",
		"Right Accessory",
		"Top Headgear",
		"Mid Headgear";

	end;
}
