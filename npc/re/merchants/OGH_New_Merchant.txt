// Function to check if the player has enough items
function	script	checkItem	{
    .@item_id = getarg(0);
    .@amount = getarg(1);
    if (countitem(.@item_id) < .@amount) {
        mes "You need " + .@amount + " of item ID " + .@item_id + ".";
        return 0;
    }
    return 1;
}

// Function to reset enchantments
function	script	F_ResetEnchant	{
    switch(select("500,000 Zeny (70% success):10 Cash Points (100%):Cancel")) {
        case 1:
            if (Zeny < 500000) {
                mes "Not enough Zeny.";
                close;
            }
            Zeny -= 500000;
            if (rand(100) < 70) {
                callfunc "resetHeadgear";
            } else {
                mes "Reset failed. Item destroyed.";
                delequip EQI_HEAD_TOP;
            }
            close;
        case 2:
            if (#CASHPOINTS < 10) {
                mes "You need 10 Cash Points.";
                close;
            }
            #CASHPOINTS -= 10;
            callfunc "resetHeadgear";
            close;
        default: close;
    }
}

// Helper function to reset headgear
function	script	resetHeadgear	{
    .@eid = getequipid(EQI_HEAD_TOP);
    if (.@eid) {
	.@ref = getequiprefinerycnt(EQI_HEAD_TOP);
        .@c0 = getequipcardid(EQI_HEAD_TOP, 0);
        .@c1 = getequipcardid(EQI_HEAD_TOP, 1);
        .@c2 = getequipcardid(EQI_HEAD_TOP, 2);
        .@c3 = getequipcardid(EQI_HEAD_TOP, 3);

        if (.@c0 || .@c1 || .@c2 || .@c3) {
            mes "Enchant reset successful.";
            delequip EQI_HEAD_TOP;
    //      delitem2 .@eid, 1, 1, 0, 0, .@c0, .@c1, .@c2, .@c3; // Remove enchanted item
            getitem2 .@eid, 1, 1, .@ref, 0, 0, 0, 0, 0; // Give clean version
        } else {
            mes "This headgear has no enchantments to reset.";
        }
    } else {
        mes "You must equip the Temporal Circlet to reset its enchantments.";
    }
    return;
}

function	script	F_TempEnchantUpgrade	{

	setarray .uny[1],29706,29707,29708,29709,29710,29711,29712,29713,29714,29715;
	setarray .met[1],29061,29062,29063,29064,29065,29066,29067,29068,29069,29070;
	setarray .mag[1],29071,29072,29073,29074,29075,29076,29077,29078,29079,29080;
	setarray .acu[1],29081,29082,29083,29084,29085,29086,29087,29088,29089,29090;
	setarray .mar[1],29091,29092,29093,29094,29095,29096,29097,29098,29099,29100;
	setarray .ada[1],29101,29102,29103,29104,29105,29106,29107,29108,29109,29110;
	setarray .aff[1],29111,29112,29113,29114,29115,29116,29117,29118,29119,29120;

	setarray .cost[1],4,6,8,10,12,16,20,28,40;

	// Validate equipment
	.@eid = getequipid(EQI_HEAD_TOP);
	if (!.@eid) {
		mes "Please wear your Temporal Circlet.";
		end;
	}
	.@ref = getequiprefinerycnt(EQI_HEAD_TOP);
	.@c0 = getequipcardid(EQI_HEAD_TOP, 0);
	.@c1 = getequipcardid(EQI_HEAD_TOP, 1);
	.@c2 = getequipcardid(EQI_HEAD_TOP, 2);
	.@c3 = getequipcardid(EQI_HEAD_TOP, 3); // special enchant slot

	if (!.@c3) {
		mes "No enchant in the 3rd slot (card slot 4).";
		end;
	}

	.@cur_lv = 0;
	.@type$ = "";

	for (.@i = 1; .@i <= 10; .@i++) {
		if (.uny[.@i] == .@c3) { .@cur_lv = .@i; .@type$ = "uny"; break; }
		if (.met[.@i] == .@c3) { .@cur_lv = .@i; .@type$ = "met"; break; }
		if (.mag[.@i] == .@c3) { .@cur_lv = .@i; .@type$ = "mag"; break; }
		if (.acu[.@i] == .@c3) { .@cur_lv = .@i; .@type$ = "acu"; break; }
		if (.mar[.@i] == .@c3) { .@cur_lv = .@i; .@type$ = "mar"; break; }
		if (.ada[.@i] == .@c3) { .@cur_lv = .@i; .@type$ = "ada"; break; }
		if (.aff[.@i] == .@c3) { .@cur_lv = .@i; .@type$ = "aff"; break; }
	}

	if (.@cur_lv == 0 || .@cur_lv >= 10) {
		mes "This enchant cannot be leveled further.";
		end;
	}

	.@next_lv = .@cur_lv + 1;
	.@need = .cost[.@cur_lv];

	if (countitem(25866) < .@need) {
		mes "You need " + .@need + " Temporal Spells.";
		end;
	}

	mes "Enchant: " + getitemname(.@c3) + " Lv" + .@cur_lv;
	mes "Leveling to Lv" + .@next_lv + " requires " + .@need + " Temporal Spells.";
	next;
	if (select("Proceed:Cancel") == 2) close;

	delitem 25866, .@need;

	if (rand(100) < 60) {
		specialeffect EF_REFINEOK;
		mes "Success! Enchant upgraded to Lv" + .@next_lv;
		if (.@type$ == "uny") .@new = .uny[.@next_lv];
		else if (.@type$ == "met") .@new = .met[.@next_lv];
		else if (.@type$ == "mag") .@new = .mag[.@next_lv];
		else if (.@type$ == "acu") .@new = .acu[.@next_lv];
		else if (.@type$ == "mar") .@new = .mar[.@next_lv];
		else if (.@type$ == "ada") .@new = .ada[.@next_lv];
		else if (.@type$ == "aff") .@new = .aff[.@next_lv];
	} else {
		if (.@cur_lv > 1) {
			.@new = getd("." + .@type$ + "[" + (.@cur_lv - 1) + "]");
			mes "Upgrade failed. Enchant downgraded to Lv" + (.@cur_lv - 1);
			specialeffect EF_REFINEFAIL;
		} else {
			.@new = .@c3;
			mes "Upgrade failed. Enchant remained at Lv1.";
			specialeffect EF_REFINEFAIL;
		}
	}

	// Replace the gear with updated slot 4 only
	delitem2 .@eid, 1, 1, .@ref, 0, .@c0, .@c1, .@c2, .@c3; // remove old
	getitem2 .@eid, 1, 1, .@ref, 0, .@c0, .@c1, .@c2, .@new; // give updated
	equip .@eid;

	return;
}
