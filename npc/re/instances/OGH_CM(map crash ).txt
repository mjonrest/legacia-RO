glast_01,143,288,5	script	Glastheim Challenge Mode Administrator	10307,{

    .@npc$ = "[Oscar]";
    .md_name$ = "Glastheim Challenge Mode";

    mes .@npc$;
    if (checkquest(200033, PLAYTIME) == 2) {
        erasequest 200033;
    }

    mes "Welcome to Old Glast Heim: Challenge Mode!";
    next;

    switch(select("create:enter")) {
        case 1:
            // Check if the player is in a party
            if (getcharid(1) == 0) {
                mes "You must be in a party to start this instance.";
                close;
            }

            // Check if the player is the party leader
            if (!is_party_leader()) {
                mes "Only the party leader can create the instance.";
                close;
            }

            // Create the instance
            set .@instance_id, instance_create(.md_name$);
            if (.@instance_id < 0) {
                mes "Failed to create the instance.";
                close;
            }

            // Get the instance map name
            set .map$, instance_mapname("1@gl_he", .@instance_id);
            mes "Instance created successfully.";
            close;
            break;

        case 2:
            // Check cooldown
            switch(checkquest(200033, PLAYTIME)) {
                case 0:
                case 1:
                    mes "You are on cooldown, fella.";
                    close;
            }

            // Try to enter instance
            switch(instance_enter(.md_name$)) {
                case IE_OTHER:
                    mes "An unknown error has occurred.";
                    close;
                case IE_NOINSTANCE:
                    mes "No instance available.";
                    close;
                case IE_OK:
                    setquest 200033;
                    mes "You and your party will now be warped to the Challenge Mode!";
                    mes "Good luck!";
                    close;
            }
            break;
    }
}
