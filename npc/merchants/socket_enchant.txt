//============================================================
//= Func_Socket
//= getarg(0) : 1 = use Spiritual Auger, 0 = normal
//= getarg(1) : item id (unslotted)
//= getarg(2) : item id (slotted result)
//= getarg(3) : success roll low bound  (success if roll > low)
//= getarg(4) : success roll high bound (success if roll < high)
//= getarg(5) : zeny fee in thousands
//= getarg(6),getarg(7) : material 1 id, amount
//= getarg(8),getarg(9) : material 2 id, amount (optional)
//============================================================
function	script	Func_Socket	{
	.@auger    = getarg(0);
	.@item_id  = getarg(1);
	.@new_id   = getarg(2);
	.@low      = getarg(3);
	.@high     = getarg(4);
	.@zeny_req = getarg(5) * 1000;
	setarray .@item_req[0], getarg(6), getarg(8,0);
	setarray .@req_amount[0], getarg(7), getarg(9,0);
	.@auger_id = 6396; // Spiritual Auger

	disable_items;

	// Find the slot where the item is actually equipped
	// (works for two-handed weapons, multi-slot headgears and both accessory slots)
	.@item_loc = -1;
	setarray .@pos[0], EQI_HAND_R, EQI_HAND_L, EQI_ARMOR, EQI_GARMENT, EQI_SHOES,
	                   EQI_HEAD_TOP, EQI_HEAD_MID, EQI_HEAD_LOW, EQI_ACC_L, EQI_ACC_R;
	for (.@i = 0; .@i < 10; .@i++) {
		if (getequipid(.@pos[.@i]) == .@item_id) {
			.@item_loc = .@pos[.@i];
			break;
		}
	}
	if (.@item_loc < 0) {
		mes "[Seiyablem]";
		mes "You need to equip your ^0000FF" + getitemname(.@item_id) + "^000000 to continue.";
		close;
	}

	// Material text
	.@mat$ = .@req_amount[0] + " " + getitemname(.@item_req[0]);
	if (.@item_req[1] && .@req_amount[1])
		.@mat$ = .@mat$ + ", " + .@req_amount[1] + " " + getitemname(.@item_req[1]);
	if (.@auger)
		.@mat$ = "1 " + getitemname(.@auger_id) + ", " + .@mat$;

	mes "[Seiyablem]";
	mes "You want to add a Slot to a " + getitemname(.@item_id) + "?";
	mes "Alright, please bring me ^FF0000" + .@mat$ + "^000000 and my ^FF0000" + callfunc("F_InsertComma", .@zeny_req) + " zeny^000000 service fee.";
	next;
	mes "[Seiyablem]";
	if (getiteminfo(.@item_id, ITEMINFO_LOCATIONS) & EQP_HAND_R) {
		mes "I can try to add a slot now if you have the required items and zeny.";
		mes "However, you should know that there's a chance that I might fail.";
		mes "Therefore, I need to give you a fair warning...";
		next;
		mes "[Seiyablem]";
		if (.@auger)
			mes "Thanks to the Spiritual Auger, your ^0000FFWeapon will NOT be destroyed^000000 if I fail, but the materials and zeny will be used.";
		else
			mes "If this attempt to add a Slot to your Weapon fails, then the ^FF0000Weapon^000000, ^FF0000and any Cards compounded to it will be destroyed^000000.";
	} else {
		mes "If you have all the required materials, my zeny service fee and the Armor, then we can go ahead with the Slot Addition attempt.";
		mes "But before that, I must warn you of the risk.";
		next;
		mes "[Seiyablem]";
		if (.@auger)
			mes "Thanks to the Spiritual Auger, your ^0000FFArmor will NOT be destroyed^000000 if I fail, but the materials and zeny will be used.";
		else
			mes "If this attempt to add a Slot to your Armor fails, then the ^FF0000Armor^000000, ^FF0000its upgrades^000000 ^FF0000and any Cards compounded to it will be destroyed^000000.";
	}
	mes "Do you still want to try to add a Slot?";
	next;
	if (select("Attempt Slot Addition:Cancel") == 2) {
		mes "[Seiyablem]";
		mes "Need some time to think about it, huh?";
		mes "Alright, I can understand.";
		mes "Just remember that life's no fun if you're always playing it safe~";
		close;
	}

	// Requirement check
	// Script && / || do not short-circuit, so only call countitem on real item IDs.
	.@missing = (Zeny < .@zeny_req || countitem(.@item_req[0]) < .@req_amount[0] || getequipid(.@item_loc) != .@item_id);
	if (.@item_req[1] && .@req_amount[1]) {
		if (countitem(.@item_req[1]) < .@req_amount[1])
			.@missing = 1;
	}
	if (.@auger) {
		if (countitem(.@auger_id) < 1)
			.@missing = 1;
	}
	if (.@missing) {
		mes "[Seiyablem]";
		mes "I'd like to go ahead with this Slot Addition attempt, but you're missing a few things.";
		mes "You sure that you have the equipment, required materials and the zeny?";
		close;
	}

	// Save refine + random options before touching the item
	.@ref = getequiprefinerycnt(.@item_loc);
	for (.@j = 0; .@j < 5; .@j++) {
		.@OptID[.@j]    = getequiprandomoption(.@item_loc, .@j, ROA_ID);
		.@OptVal[.@j]   = getequiprandomoption(.@item_loc, .@j, ROA_VALUE);
		.@OptParam[.@j] = getequiprandomoption(.@item_loc, .@j, ROA_PARAM);
	}

	mes "[Seiyablem]";
	mes "Alright then, let the work begin!";
	mes "You'd better pray for a successful result.";
	next;

	// Consume materials and fee (always)
	delitem .@item_req[0], .@req_amount[0];
	if (.@item_req[1] && .@req_amount[1])
		delitem .@item_req[1], .@req_amount[1];
	if (.@auger)
		delitem .@auger_id, 1;
	Zeny -= .@zeny_req;

	.@roll = rand(1,100);
	if (.@roll > .@low && .@roll < .@high) {
		specialeffect (.@high == 51 ? EF_LORD : EF_SANCTUARY);
		delequip .@item_loc;
		getitem3 .@new_id, 1, 1, .@ref, 0, 0, 0, 0, 0, .@OptID, .@OptVal, .@OptParam;
		mes "[Seiyablem]";
		mes "Great, it seems to be successful.";
		mes "It looks pretty well done. Congratulations!";
		next;
		mes "[Seiyablem]";
		mes "See you again, buddy!";
		close;
	}

	specialeffect EF_SUI_EXPLOSION;
	if (!.@auger)
		delequip .@item_loc; // destroyed only without Spiritual Auger
	mes "[Seiyablem]";
	mes "Wah! ...I am so sorry, it failed.";
	if (.@auger)
		mes "Luckily the Spiritual Auger protected your equipment.";
	mes "However, I am completely innocent.";
	mes "This is your luck, and it is destined by god, okay?";
	mes "Don't be so disappointed, and try next time.";
	next;
	mes "[Seiyablem]";
	mes "I wish you good luck next time!";
	close;
}
