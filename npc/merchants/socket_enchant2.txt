//============================================================
//= Func_Socket2
//= getarg(0) : 1 = use Spiritual Auger, 0 = normal
//= getarg(1) : item id (unslotted)
//= getarg(2) : item id (slotted result)
//= getarg(3) : success roll low bound  (success if roll > low)
//= getarg(4) : success roll high bound (success if roll < high)
//= getarg(5) : zeny fee in thousands
//= getarg(6),getarg(7) : material 1 id, amount
//= getarg(8),getarg(9) : material 2 id, amount (optional)
//============================================================
function	script	Func_Socket2	{
	.@auger    = getarg(0);
	.@item_id  = getarg(1);
	.@new_id   = getarg(2);
	.@low      = getarg(3);
	.@high     = getarg(4);
	.@zeny_req = getarg(5) * 1000;
	setarray .@item_req[0], getarg(6), getarg(8,0);
	setarray .@req_amount[0], getarg(7), getarg(9,0);
	.@auger_id = 6396; // Spiritual Auger
	.@type$ = (getiteminfo(.@item_id, ITEMINFO_LOCATIONS) & EQP_HAND_R) ? "weapon" : "armor";

	disable_items;

	// Find the slot where the item is actually equipped
	.@item_loc = -1;
	setarray .@pos[0], EQI_HAND_R, EQI_HAND_L, EQI_ARMOR, EQI_GARMENT, EQI_SHOES,
	                   EQI_HEAD_TOP, EQI_HEAD_MID, EQI_HEAD_LOW, EQI_ACC_L, EQI_ACC_R;
	for (.@i = 0; .@i < 10; .@i++) {
		if (getequipid(.@pos[.@i]) == .@item_id) {
			.@item_loc = .@pos[.@i];
			break;
		}
	}
	if (.@item_loc < 0) {
		mes "[Leablem]";
		mes "You need to equip your ^0000FF" + getitemname(.@item_id) + "^000000 to continue.";
		close;
	}

	// Material text
	.@mat$ = .@req_amount[0] + "ea " + getitemname(.@item_req[0]);
	if (.@item_req[1] && .@req_amount[1])
		.@mat$ = .@mat$ + " and " + .@req_amount[1] + "ea " + getitemname(.@item_req[1]);
	if (.@auger)
		.@mat$ = "1 " + getitemname(.@auger_id) + ", " + .@mat$;

	mes "[Leablem]";
	mes getitemname(.@item_id) + "? Okay, then you need to bring me";
	mes "^FF0000" + callfunc("F_InsertComma", .@zeny_req) + " zeny as service charge,";
	mes "^FF0000" + .@mat$ + ".";
	mes "^FF0000Of course, you need to wear the " + getitemname(.@item_id) + ".^000000";
	next;
	mes "[Leablem]";
	mes "Did you already bring all of them?";
	if (.@auger) {
		mes "Thanks to the Spiritual Auger, your " + .@type$ + " will ^0000FFNOT be destroyed^000000 if I fail,";
		mes "but the materials and zeny will still be used.";
	} else {
		mes "For your information, if you fail to create a slot,";
		mes "you will lose all the item requirement as well as the target " + .@type$ + ".";
	}
	mes "If I succeed, the upgrade level is kept, but any compounded ^FF0000Cards will be lost^000000.";
	next;
	if (select("Ask for slot creation.:Try next time.") == 2) {
		mes "[Leablem]";
		mes "See you next time.";
		close;
	}

	// Requirement check
	if (Zeny < .@zeny_req
	 || countitem(.@item_req[0]) < .@req_amount[0]
	 || (.@item_req[1] && .@req_amount[1] && countitem(.@item_req[1]) < .@req_amount[1])
	 || (.@auger && countitem(.@auger_id) < 1)
	 || getequipid(.@item_loc) != .@item_id) {
		mes "[Leablem]";
		mes "Are you stupid or what? You didn't bring all of the required items!";
		mes "Go bring them quick!";
		close;
	}

	// Save refine + random options before touching the item
	.@ref = getequiprefinerycnt(.@item_loc);
	for (.@j = 0; .@j < 5; .@j++) {
		.@OptID[.@j]    = getequiprandomoption(.@item_loc, .@j, ROA_ID);
		.@OptVal[.@j]   = getequiprandomoption(.@item_loc, .@j, ROA_VALUE);
		.@OptParam[.@j] = getequiprandomoption(.@item_loc, .@j, ROA_PARAM);
	}

	mes "[Leablem]";
	mes "Alright then, let the work begin!";
	mes "You'd better pray for a successful result.";
	next;

	// Consume materials and fee (always)
	delitem .@item_req[0], .@req_amount[0];
	if (.@item_req[1] && .@req_amount[1])
		delitem .@item_req[1], .@req_amount[1];
	if (.@auger)
		delitem .@auger_id, 1;
	Zeny -= .@zeny_req;

	.@roll = rand(1,100);
	if (.@roll > .@low && .@roll < .@high) {
		specialeffect EF_LORD;
		delequip .@item_loc;
		getitem3 .@new_id, 1, 1, .@ref, 0, 0, 0, 0, 0, .@OptID, .@OptVal, .@OptParam;
		mes "[Leablem]";
		mes "Great, it seems to be successful.";
		mes "It looks pretty well done. Congratulations!";
		next;
		mes "[Leablem]";
		mes "See you again, buddy!";
		close;
	}

	specialeffect EF_SUI_EXPLOSION;
	if (!.@auger)
		delequip .@item_loc; // destroyed only without Spiritual Auger
	mes "[Leablem]";
	mes "Wah! ...I am so sorry, it failed.";
	if (.@auger)
		mes "Luckily the Spiritual Auger protected your " + .@type$ + ".";
	mes "However, I am completely innocent.";
	mes "This is your luck, and it is destined by god, okay?";
	mes "Don't be so disappointed, and try next time.";
	next;
	mes "[Leablem]";
	mes "See you again, buddy!";
	close;
}
