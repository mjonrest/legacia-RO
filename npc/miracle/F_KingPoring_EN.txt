OnSellItem:

	.@size = getarraysize(@sold_nameid);
	.@total = 0;

	// Sum ALL sold quantities
	for (.@i = 0; .@i < .@size; .@i++)
		.@total += @sold_quantity[.@i];

	if (.@total != .card_count) {
		mes "[King Poring]";
		mes "You must place exactly ^FF0000" + .card_count + " cards^000000.";
		close;
	}

	//--------------------------------------------------
	// Validate cards
	//--------------------------------------------------
	for (.@i = 0; .@i < .@size; .@i++) {

		if (getiteminfo(@sold_nameid[.@i], 2) != IT_CARD) {
			mes "[King Poring]";
			mes "Only cards are allowed.";
			close;
		}
	}

	//--------------------------------------------------
	// Confirmation
	//--------------------------------------------------
	if (select("Recycle cards:Cancel") == 2) close;

	Zeny -= .zeny_cost;

	for (.@i = 0; .@i < .@size; .@i++)
		delitem @sold_nameid[.@i], @sold_quantity[.@i];

	//--------------------------------------------------
	// Roll reward
	//--------------------------------------------------
	.@rand = rand(1,100);

	if (.@rand <= .rate_mvp) {
		// 1% MVP
		.@mvp_rand = .MVPCards[rand(.mvp_size)];
		getitem .@mvp_rand, 1;
		mes "[King Poring]";
		mes "You received a ^FF0000MVP Card^000000!";
		announce "[King Poring]: "+strcharinfo(0)+" just obtained "+itemlink(.@mvp_rand)+".", bc_all, 0xFF2A2A;
	}
	else if (.@rand <= .rate_mini) {
		// 3% Mini (2–4)
		.@mini_rand = .MiniCards[rand(.mini_size)];
		getitem .@mini_rand, 1;
		mes "[King Poring]";
		mes "You received a ^0000FFMini Boss Card^000000!";
		announce "[King Poring]: "+strcharinfo(0)+" just obtained "+itemlink(.@mini_rand)+".", bc_all, 0x4FC3F7;
	}
	else {
		// 96% Normal
		.@normal_rand = .NormalCards[rand(.normal_size)];
		getitem .@normal_rand, 1;
		mes "[King Poring]";
		mes "You received a ^00FF00Normal Card^000000!";
		announce "[King Poring]: "+strcharinfo(0)+" just obtained "+itemlink(.@normal_rand)+".", bc_all, 0x00FF00;
	}
	close;
