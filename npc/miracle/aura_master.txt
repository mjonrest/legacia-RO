//==============================================================
// LegaciaRO - Aura Master (Classic Auras)
//==============================================================
// Sells the Classic Auras: the official level 99 aura with the
// pictures of one of 163 custom aura packs (set_aura_pack). The
// game draws it with its own motion; legacia.dll swaps in the
// pack's pictures (data/texture/effect/legacia_aura_pack/NNN/,
// made by tools/legacia_aura/genpacks.py, list in packs.txt).
// It shows at base level 99 (max_lv) only. Players without the
// plugin see the normal blue level 99 aura on pack wearers.
// A Classic Aura is unlocked once (account-wide); switching
// between owned ones is free. @aura opens the menu. Browsing works
// like the hair Stylist: Next / Previous show each aura on the
// character right away, then Pick (or Unlock) it.
//
// The old Gepard-style 3-part auras (ground circle, rings,
// sparkles) were dropped on 2026-11-17 (owner's choice, no refund):
// every login takes off worn parts (set_aura 0,0,0) and unlocked
// parts are cleared from #AURA_OWN.
//
// Variables
//  #AURA_OWN[n]   : account bitmask of unlocked auras (31 per entry);
//                   Classic Aura n is bit slot 304 + n - 1 (slots
//                   0-303 were the old parts)
//  #AURA_OLDGONE  : 1 once the account's old parts were cleared
//  AURA_PREVIEW   : 4000 + old pack while browsing (undone at login)
//==============================================================

prontera,142,223,4	script	Aura Master	4_M_SAGE_C,{
	mes "[Aura Master]";
	mes "I can give you a Classic Aura: the level " + .PackLv + " aura in one of " + .Max + " looks.";
	mes "Unlock one once and every character on your account can wear it.";
	next;
	callsub S_Menu;
	end;

OnCmd:
	callsub S_Menu;
	end;

OnInit:
	// ---- settings ----
	.PackZeny = 50000000;	// zeny per Classic Aura (0 = can't pay with zeny)
	.PackCash = 51;		// Cash Points (#CASHPOINTS) per Classic Aura (0 = off)
	.PackLv = 99;		// base level the Classic Aura shows at (the server's max_lv)
	// ------------------
	.Max = 163;
	.Base = 304;	// bit slot of Classic Aura #1 in #AURA_OWN
	// Classic Aura names, same order as tools/legacia_aura/packs.txt
	setarray .PackName$[1], "Abstract Flower", "Aiyu Current", "Aiyu Old", "Amulet", "Angelic Aura", "AniM0rph Aura", "Crest Moon Aura", "Solestice Aura";
	setarray .PackName$[9], "Appel", "Arcane Aura", "Arcane1", "Arcane2", "Arcane3", "Arcane4", "ArticGalaxyAura-by-Melz", "Aura of King";
	setarray .PackName$[17], "AuraFlame-by-Melz", "Autumnleaf by WP", "Beautiful dolphin", "Big Bad Voodoo", "Blood Aura", "Bloody sun", "Blubble", "Blubble v.2";
	setarray .PackName$[25], "Blue Heaven Aura", "Blue Moon", "Blue fume", "Blush Aura", "Butterfly Aura", "Butterfly v.2", "Celestia Tattoo Aura", "Celtic";
	setarray .PackName$[33], "Celtic green", "Celtic protection", "Celtic flower2", "Celtic heart", "Chococat", "Circle of Power", "Color Swirls", "Crest Aura";
	setarray .PackName$[41], "Cross Aura", "Daisy Aura", "Dark Summoning", "Demon Symbol", "Devotion", "Dolphin", "Dragon", "Dragon head";
	setarray .PackName$[49], "Dragonflies", "Dream of Anime Aura", "Envy Aura", "Fairy", "Fire Sun Aura", "Flamy sun", "Flamy sun v2", "Flashy";
	setarray .PackName$[57], "Flashy2", "Flashy3", "Flower Aura", "Force", "Full Metal Alchemist", "Funky colors", "Ghost1 Aura", "Ghost2 Aura";
	setarray .PackName$[65], "Ghost3 Aura", "Ghost4 Aura", "Gluttony Aura", "Grape", "Greed Aura", "Green plant", "Green swirls", "Green wing";
	setarray .PackName$[73], "Greeneffect12 aura", "Grunge green", "Guiding star", "Heart", "HelloKitty", "Hellsing Aura", "Holy 1", "Holy Aura";
	setarray .PackName$[81], "Human Emblem", "Lady bug", "Lauriers", "Lead Circlet Aura", "Leal Life Aura", "Lifesource Aura", "Limitless Flowers Aura", "Lionheart Aura";
	setarray .PackName$[89], "Little rounds", "Lord of the Rings", "Lotus", "Love", "Lust Aura", "Magic circle", "Manic", "Medieval Heart";
	setarray .PackName$[97], "Medieval Heart v.2", "Merlin", "Metallic butterfly", "Midnight Fenrir Aura", "Mist Of Azure", "Musical", "Nature Aura", "Necromancy Aura";
	setarray .PackName$[105], "Phoenix", "Pretty lotus", "Pride Aura", "Pumpkin by WP", "Purple Flower v1", "Purple Space", "Purple moon", "PurpleStarsAura-by-melz";
	setarray .PackName$[113], "Red Smoke", "Red Stars", "Red Wave", "Red cross", "Red shield", "Red splash", "Rose 2", "Rose Aura";
	setarray .PackName$[121], "Rose necklace", "Scribble", "Server Current", "Shadow Aura", "Skull", "Sleepy moon", "Sloth Aura", "Snow";
	setarray .PackName$[129], "SnowAura", "Solar", "Solar v.2", "Solar v.3", "Solar v.4", "Something", "Spike Aura", "Splash circle";
	setarray .PackName$[137], "Square", "Star", "Strawberry", "Sun and Moon", "Triangle", "Tribal Aura", "Tricky Sun", "TriggerSymbol Aura";
	setarray .PackName$[145], "Turquoise", "Undead Green Sigil", "Undead Sigil", "Undead Sigil Red", "Unholy Aura", "Unicorn", "Unknown", "Unwanted Happening Aura";
	setarray .PackName$[153], "Wavy circle", "Wedding by WP", "White Feather Aura", "White Labyrinth", "White Swirl", "Wrath Aura", "Yellow Smoke", "Zeet's Batman Aura";
	setarray .PackName$[161], "Zeet's Blue Aura", "Zeet's Pink Aura", "Zeet's Red Aura";
	.@p$ = "";
	if (.PackZeny)
		.@p$ = callfunc("F_InsertComma", .PackZeny) + " zeny";
	if (.PackCash)
		.@p$ += (.@p$ == "" ? "" : " or ") + .PackCash + " Cash Points";
	.Price$ = .@p$ == "" ? "free" : .@p$;
	bindatcmd "aura", strnpcinfo(3) + "::OnCmd", 0, 99;
	waitingroom "[ Classic Auras ]", 0;
	end;

OnPCLoginEvent:
	// The old 3-part auras are gone: take off worn parts, forget unlocked ones (no refund).
	if (get_aura_part(1) || get_aura_part(2) || get_aura_part(3))
		set_aura 0, 0, 0;
	if (!#AURA_OLDGONE) {
		for (.@i = 0; .@i < 9; .@i++)
			#AURA_OWN[.@i] = 0;	// slots 0-278
		#AURA_OWN[9] = #AURA_OWN[9] & ~((1 << 25) - 1);	// slots 279-303
		#AURA_OLDGONE = 1;
	}
	// Logged out while browsing: put the aura worn before back.
	if (AURA_PREVIEW) {
		if (AURA_PREVIEW / 1000 == 4)
			set_aura_pack AURA_PREVIEW % 1000;
		AURA_PREVIEW = 0;
	}
	end;

// ---------------- main menu ----------------
S_Menu:
	while (1) {
		clear;
		mes "[Aura Master]";
		mes "Your Classic Aura: " + (get_aura_pack() ? "^0000FF" + callsub(S_Label, get_aura_pack()) + "^000000" : "none");
		mes "Price: " + .Price$;
		mes "It shows at base level " + .PackLv + ".";
		.@s = select(
			"Try Classic Auras (" + callsub(S_CountOwned) + "/" + .Max + " owned)",
			"My Classic Auras",
			(get_aura_pack() ? "Take off my aura" : ""),
			"Close");
		switch (.@s) {
		case 1:
			callsub S_Browse;
			break;
		case 2:
			callsub S_Mine;
			break;
		case 3:
			set_aura_pack 0;
			break;
		default:
			close;
		}
	}

// ---------------- stylist-style browser ----------------
// The aura changes on the character as soon as the player moves
// with Next / Previous, like the hair Stylist. "Pick this one"
// wears it (unlocks it first if needed); Cancel puts the old one back.
S_Browse:
	.@max = .Max;
	.@old = get_aura_pack();
	.@cur = (.@old > 0 && .@old <= .@max) ? .@old : 1;
	AURA_PREVIEW = 4000 + .@old;	// undone at login if the player logs out mid-browse
	while (1) {
		set_aura_pack .@cur;
		.@next = (.@cur % .@max) + 1;
		.@prev = ((.@cur + .@max - 2) % .@max) + 1;
		.@own = callsub(S_Owned, .@cur);
		clear;
		mes "[Aura Master]";
		mes "^0000FFClassic Aura " + callsub(S_Label, .@cur) + "^000000 (of " + .@max + ")";
		if (BaseLevel < .PackLv)
			mes "^FF0000It shows once you reach base level " + .PackLv + ".^000000";
		if (.@cur == .@old)
			mes "^0000FFYou are wearing this one.^000000";
		else if (.@own)
			mes "^008800Owned^000000, free to wear.";
		else
			mes "Price: " + .Price$;
		mes "Look at your character, then keep browsing or pick it.";
		.@s = prompt(
			"Next  >> #" + .@next,
			"<< Previous  #" + .@prev,
			"Jump to number",
			(.@own ? "Pick this one" : "Unlock and wear this one"),
			"Cancel");
		switch (.@s) {
		case 1:
			.@cur = .@next;
			break;
		case 2:
			.@cur = .@prev;
			break;
		case 3:
			clear;
			mes "[Aura Master]";
			mes "Which Classic Aura number? (1-" + .@max + ")";
			if (input(.@j, 1, .@max) == 0)
				.@cur = .@j;
			break;
		case 4:
			if (!.@own) {
				if (!callsub(S_Pay, .@cur))
					break;	// no money or said No: keep browsing
				callsub S_Give, .@cur;
				AURA_PREVIEW = 0;
				clear;
				mes "[Aura Master]";
				mes "Unlocked! You are now wearing Classic Aura " + callsub(S_Label, .@cur) + ".";
				next;
				return;
			}
			AURA_PREVIEW = 0;
			return;
		default:	// Cancel or closed the window
			set_aura_pack .@old;
			AURA_PREVIEW = 0;
			if (.@s == 255)
				end;
			return;
		}
	}

// ---------------- owned list ----------------
S_Mine:
	.@c = 0;
	for (.@i = 1; .@i <= .Max; .@i++)
		if (callsub(S_Owned, .@i))
			.@ids[.@c++] = .@i;
	clear;
	mes "[Aura Master]";
	if (!.@c) {
		mes "You have not unlocked any Classic Aura yet.";
		next;
		return;
	}
	mes "Pick one to wear. Switching is free.";
	.@menu$ = "";
	for (.@i = 0; .@i < .@c; .@i++)
		.@menu$ += callsub(S_Label, .@ids[.@i]) + (.@ids[.@i] == get_aura_pack() ? " ^0000FF[wearing]^000000" : "") + ":";
	.@menu$ += "Back";
	.@s = select(.@menu$) - 1;
	if (.@s < .@c)
		set_aura_pack .@ids[.@s];
	return;

// ---------------- helpers ----------------
S_Pay:
	if (!.PackZeny && !.PackCash)
		return 1;	// free
	clear;
	mes "[Aura Master]";
	mes "Unlock Classic Aura " + callsub(S_Label, getarg(0)) + "?";
	mes "Price: " + .Price$;
	.@how = select(
		(.PackZeny ? "Pay " + callfunc("F_InsertComma", .PackZeny) + " zeny" : ""),
		(.PackCash ? "Pay " + .PackCash + " Cash Points (you have " + #CASHPOINTS + ")" : ""),
		"No");
	if (.@how == 1) {
		if (Zeny < .PackZeny) {
			mes "You don't have enough zeny.";
			next;
			return 0;
		}
		Zeny -= .PackZeny;
		return 1;
	}
	if (.@how == 2) {
		if (#CASHPOINTS < .PackCash) {
			mes "You don't have enough Cash Points.";
			next;
			return 0;
		}
		#CASHPOINTS -= .PackCash;
		dispbottom "Used " + .PackCash + " Cash Points. Left: " + #CASHPOINTS + ".";
		return 1;
	}
	return 0;

// "#n Name"
S_Label:
	return "#" + getarg(0) + " " + .PackName$[getarg(0)];

// bit slot of Classic Aura n, -1 if out of range
S_Slot:
	if (getarg(0) < 1 || getarg(0) > .Max)
		return -1;
	return .Base + getarg(0) - 1;

S_Owned:
	.@s = callsub(S_Slot, getarg(0));
	if (.@s < 0)
		return 0;
	return (#AURA_OWN[.@s / 31] & (1 << (.@s % 31))) ? 1 : 0;

S_Give:
	.@s = callsub(S_Slot, getarg(0));
	if (.@s >= 0)
		#AURA_OWN[.@s / 31] = #AURA_OWN[.@s / 31] | (1 << (.@s % 31));
	return;

S_CountOwned:
	.@c = 0;
	for (.@i = 1; .@i <= .Max; .@i++)
		if (callsub(S_Owned, .@i))
			.@c++;
	return .@c;
}
