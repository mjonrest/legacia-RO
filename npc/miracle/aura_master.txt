//==============================================================
// LegaciaRO - Aura Master (our own aura system)
//==============================================================
// Replaces the Gepard aura renderer. An aura has 3 parts, same
// as Gepard: Ground circle (part 1), Rings (part 2) and Sparkles
// (part 3). Players unlock a part once (account-wide) and can
// switch between owned parts for free. @aura opens the menu.
// Browsing works like the hair Stylist: Next / Previous show each
// part on the character right away, then Pick (or Unlock) it.
//
// How it works
//  - The aura is still stored by the Gepard commands
//    (set_aura_part / AURA_DATA), so old auras, @auras and the
//    old Princess Mary NPC (npc/miracle/aura.txt) keep working.
//  - The client plugin legacia.dll draws it in 3D (ground circle,
//    rings, sparkles); the map-server sends the auras around each
//    player through the voice server (src/map/voice.cpp, AURAS).
//    Textures: plugins/auras/ next to legacia_plugin.dll.
//  - Old way, off by default: pc_aura_apply() in src/map/pc.cpp shows
//    it as hat effects (aura_hateffect in conf/battle/feature.conf).
//
// Variables
//  #AURA_OWN[n] : account bitmask of unlocked parts (31 per entry)
//  AURA_GRANDF  : 1 once the character's old aura was marked owned
//==============================================================

prontera,142,223,4	script	Aura Master	4_M_SAGE_C,{
	mes "[Aura Master]";
	mes "I can surround you with a lasting aura.";
	mes "An aura has three parts: a ground circle, rings and sparkles. Mix them as you like.";
	mes "Unlock a part once and every character on your account can wear it.";
	next;
	callsub S_Menu;
	end;

OnCmd:
	callsub S_Menu;
	end;

OnInit:
	// ---- settings ----
	.Zeny = 50000000;	// zeny per aura part (0 = can't pay with zeny)
	.Cash = 51;		// Cash Points (#CASHPOINTS) per aura part, the other way to pay (0 = off)
	// ------------------
	setarray .PartMax[1], 152, 117, 35;
	setarray .PartName$[1], "Ground Circle", "Ring", "Sparkle";
	setarray .PartBase[1], 0, 152, 269;	// bit slot of #1 of each part
	.Price$ = "";
	if (.Zeny)
		.Price$ = callfunc("F_InsertComma", .Zeny) + " zeny";
	if (.Cash)
		.Price$ += (.Price$ == "" ? "" : " or ") + .Cash + " Cash Points";
	if (.Price$ == "")
		.Price$ = "free";
	bindatcmd "aura", strnpcinfo(3) + "::OnCmd", 0, 99;
	waitingroom "[ Auras ]", 0;
	end;

OnPCLoginEvent:
	// Auras bought before this NPC (Princess Mary / GM) become owned once.
	if (!AURA_GRANDF) {
		for (.@p = 1; .@p <= 3; .@p++)
			if (get_aura_part(.@p) > 0)
				callsub S_Give, .@p, get_aura_part(.@p);
		AURA_GRANDF = 1;
		end;
	}
	// Logged out during a preview: put the part worn before back.
	// (Only previews are undone, so auras given with the GM command @auras stay.)
	if (AURA_PREVIEW) {
		set_aura_part AURA_PREVIEW / 1000, AURA_PREVIEW % 1000;
		AURA_PREVIEW = 0;
	}
	end;

// ---------------- main menu ----------------
S_Menu:
	while (1) {
		clear;
		mes "[Aura Master]";
		mes "Your aura:";
		for (.@p = 1; .@p <= 3; .@p++)
			mes " " + .PartName$[.@p] + ": " + (get_aura_part(.@p) ? "^0000FF#" + get_aura_part(.@p) + "^000000" : "none");
		mes "Price per part: " + .Price$;
		.@s = select(
			"Try ground circles (" + callsub(S_CountOwned, 1) + "/" + .PartMax[1] + " owned)",
			"Try rings (" + callsub(S_CountOwned, 2) + "/" + .PartMax[2] + " owned)",
			"Try sparkles (" + callsub(S_CountOwned, 3) + "/" + .PartMax[3] + " owned)",
			"My auras",
			"Remove a part",
			"Close");
		switch (.@s) {
		case 1:
		case 2:
		case 3:
			callsub S_Browse, .@s;
			break;
		case 4:
			callsub S_Mine;
			break;
		case 5:
			clear;
			mes "[Aura Master]";
			mes "Which part should I remove?";
			.@r = select(
				(get_aura_part(1) ? "Ground circle #" + get_aura_part(1) : ""),
				(get_aura_part(2) ? "Ring #" + get_aura_part(2) : ""),
				(get_aura_part(3) ? "Sparkle #" + get_aura_part(3) : ""),
				"All parts",
				"Back");
			if (.@r <= 3)
				set_aura_part .@r, 0;
			else if (.@r == 4)
				set_aura 0, 0, 0;
			break;
		default:
			close;
		}
	}

// ---------------- stylist-style browser ----------------
// The aura changes on the character as soon as the player moves
// with Next / Previous, like the hair Stylist. "Pick this one"
// wears it (unlocks it first if needed); Cancel puts the old one back.
S_Browse:
	.@part = getarg(0);
	.@max = .PartMax[.@part];	// plain copy: min()/max() given an array element walk the array
	.@old = get_aura_part(.@part);
	.@cur = (.@old > 0 && .@old <= .@max) ? .@old : 1;
	AURA_PREVIEW = .@part * 1000 + .@old;	// undone at login if the player logs out mid-browse
	while (1) {
		set_aura_part .@part, .@cur;
		.@next = (.@cur % .@max) + 1;
		.@prev = ((.@cur + .@max - 2) % .@max) + 1;
		.@own = callsub(S_Owned, .@part, .@cur);
		clear;
		mes "[Aura Master]";
		mes "^0000FF" + .PartName$[.@part] + " #" + .@cur + "^000000 (of " + .@max + ")";
		if (.@cur == .@old)
			mes "^0000FFYou are wearing this one.^000000";
		else if (.@own)
			mes "^008800Owned^000000, free to wear.";
		else
			mes "Price: " + .Price$;
		mes "Look at your character, then keep browsing or pick it.";
		.@s = prompt(
			"Next  >> #" + .@next,
			"<< Previous  #" + .@prev,
			"Jump to number",
			(.@own ? "Pick this one" : "Unlock and wear this one"),
			"Cancel");
		switch (.@s) {
		case 1:
			.@cur = .@next;
			break;
		case 2:
			.@cur = .@prev;
			break;
		case 3:
			clear;
			mes "[Aura Master]";
			mes "Which " + .PartName$[.@part] + " number? (1-" + .@max + ")";
			if (input(.@j, 1, .@max) == 0)
				.@cur = .@j;
			break;
		case 4:
			if (!.@own) {
				if (!callsub(S_Pay, .@part, .@cur))
					break;	// no money or said No: keep browsing
				callsub S_Give, .@part, .@cur;
				AURA_PREVIEW = 0;
				clear;
				mes "[Aura Master]";
				mes "Unlocked! You are now wearing " + .PartName$[.@part] + " #" + .@cur + ".";
				next;
				return;
			}
			AURA_PREVIEW = 0;
			return;
		default:	// Cancel or closed the window
			set_aura_part .@part, .@old;
			AURA_PREVIEW = 0;
			if (.@s == 255)
				end;
			return;
		}
	}

// ---------------- owned list ----------------
S_Mine:
	clear;
	mes "[Aura Master]";
	mes "Which part do you want to change?";
	.@part = select("Ground circles:Rings:Sparkles:Back");
	if (.@part == 4)
		return;
	.@c = 0;
	for (.@i = 1; .@i <= .PartMax[.@part]; .@i++)
		if (callsub(S_Owned, .@part, .@i))
			.@ids[.@c++] = .@i;
	clear;
	mes "[Aura Master]";
	if (!.@c) {
		mes "You have not unlocked any " + .PartName$[.@part] + " yet.";
		next;
		return;
	}
	mes "Pick one to wear. Switching is free.";
	.@menu$ = "";
	for (.@i = 0; .@i < .@c; .@i++)
		.@menu$ += .PartName$[.@part] + " #" + .@ids[.@i] + (.@ids[.@i] == get_aura_part(.@part) ? " ^0000FF[wearing]^000000" : "") + ":";
	.@menu$ += "Back";
	.@s = select(.@menu$) - 1;
	if (.@s < .@c)
		set_aura_part .@part, .@ids[.@s];
	return;

// ---------------- helpers ----------------
S_Pay:
	if (!.Zeny && !.Cash)
		return 1;	// free
	clear;
	mes "[Aura Master]";
	mes "Unlock " + .PartName$[getarg(0)] + " #" + getarg(1) + "?";
	mes "Price: " + .Price$;
	.@how = select(
		(.Zeny ? "Pay " + callfunc("F_InsertComma", .Zeny) + " zeny" : ""),
		(.Cash ? "Pay " + .Cash + " Cash Points (you have " + #CASHPOINTS + ")" : ""),
		"No");
	if (.@how == 1) {
		if (Zeny < .Zeny) {
			mes "You don't have enough zeny.";
			next;
			return 0;
		}
		Zeny -= .Zeny;
		return 1;
	}
	if (.@how == 2) {
		if (#CASHPOINTS < .Cash) {
			mes "You don't have enough Cash Points.";
			next;
			return 0;
		}
		#CASHPOINTS -= .Cash;
		dispbottom "Used " + .Cash + " Cash Points. Left: " + #CASHPOINTS + ".";
		return 1;
	}
	return 0;

// slot of (part, number), -1 if out of range
S_Slot:
	.@part = getarg(0);
	.@num = getarg(1);
	if (.@part < 1 || .@part > 3 || .@num < 1 || .@num > .PartMax[.@part])
		return -1;
	return .PartBase[.@part] + .@num - 1;

S_Owned:
	.@s = callsub(S_Slot, getarg(0), getarg(1));
	if (.@s < 0)
		return 0;
	return (#AURA_OWN[.@s / 31] & (1 << (.@s % 31))) ? 1 : 0;

S_Give:
	.@s = callsub(S_Slot, getarg(0), getarg(1));
	if (.@s >= 0)
		#AURA_OWN[.@s / 31] = #AURA_OWN[.@s / 31] | (1 << (.@s % 31));
	return;

S_CountOwned:
	.@part = getarg(0);
	.@c = 0;
	for (.@i = 1; .@i <= .PartMax[.@part]; .@i++)
		if (callsub(S_Owned, .@part, .@i))
			.@c++;
	return .@c;
}
