S_RegisterDaily:

        "SELECT last_ip FROM login "

    // ------------------------------------

        "SELECT last_unique_id FROM login "
        +"WHERE account_id="+.@acc,
        .@uid$

    // Check today's registration
    // ------------------------------------

        "SELECT COUNT(*) FROM dailyquest_register "

            +"last_ip='"+escape_sql(.@ip$)+"' "

S_GenerateNextQuest:

	if (#dailyquest_donecount >= .QuestCount)
		return;

	.@i = #dailyquest_step;

	#dailyquest_done[.@i] = 0;
	#dailyquest_progress[.@i] = 0;

	.@type = rand(0,2);
	#dailyquest_type[.@i] = .@type;

	.@diff = rand(0, getarraysize(.HuntNeed)-1);
	#dailyquest_diff[.@i] = .@diff;

	// Hunting
	if (.@type == 0) {
		.@idx = rand(getarraysize(.HuntMob)-1);
		#dailyquest_id[.@i] = .HuntMob[.@idx];
		#dailyquest_need[.@i] = .HuntNeed[.@diff];
	}

	// Collecting
	else if (.@type == 1) {

	if (.@diff == 0) {
		.@idx = rand(getarraysize(.CollectItem_Easy)-1);
		#dailyquest_id[.@i] = .CollectItem_Easy[.@idx];
	}
	else if (.@diff == 1) {
		.@idx = rand(getarraysize(.CollectItem_Medium)-1);
		#dailyquest_id[.@i] = .CollectItem_Medium[.@idx];
	}
	else {
		.@idx = rand(getarraysize(.CollectItem_Hard)-1);
		#dailyquest_id[.@i] = .CollectItem_Hard[.@idx];
	}

	#dailyquest_need[.@i] = .CollectNeed[.@diff];
	}
	
	// Instance
	else {
		#dailyquest_id[.@i] = 0;
		#dailyquest_need[.@i] = .InstanceDailyNeed;
	}

	return;


// ==========================================
// CLAIM REWARD AND MOVE NEXT
// ==========================================
S_ClaimReward:

	.@i = #dailyquest_step;

	if (#dailyquest_progress[.@i] < #dailyquest_need[.@i])
		return;

	.@r = rand(getarraysize(.RewardItem)-1);
	.@amount = rand(.RewardMin[.@r], .RewardMax[.@r]);

	getitem .RewardItem[.@r], .@amount;

	// log to SQL
	callfunc "F_LogDailyQuestReward",
	.@i,                       // quest step index
	.RewardItem[.@r],          // item id
	.@amount,                  // amount
	#dailyquest_type[.@i],     // quest type
	#dailyquest_diff[.@i],     // difficulty
	#dailyquest_id[.@i];       // target mob/item

	mes "[Daily Quest Manager]";
	mes "You received "+getitemname(.RewardItem[.@r])+" x"+.@amount;

	#dailyquest_done[.@i] = 1;
	#dailyquest_donecount++;

	#dailyquest_step++;

	// ==========================================
	// ANNOUNCEMENT PER QUEST COMPLETION
	// ==========================================
	.@gender$ = (Sex == 1) ? "his" : "her";
	.@order$ = .Ordinal$[#dailyquest_donecount - 1];

	announce "[Daily Quest]: "+strcharinfo(0) + " just completed " + .@gender$ + " " + .@order$ + " Daily Quest!", bc_all;

	if (#dailyquest_donecount < .QuestCount) {
		callsub S_GenerateNextQuest;
		mes "New daily quest has been assigned!";
	}
	else {
		mes "^00AA00All daily quests completed!^000000";
	}

	return;

// ==========================================
// GM MANUAL RESET (SELF)
// ==========================================
S_GM_ResetDaily:

	.@today$ = gettimestr("%Y-%m-%d",21);

	#dailyquest_date$ = .@today$;

	// reset progression
	#dailyquest_step = 0;
	#dailyquest_donecount = 0;

	// wipe all quest data arrays
	deletearray #dailyquest_type[0], 5;
	deletearray #dailyquest_id[0], 5;
	deletearray #dailyquest_need[0], 5;
	deletearray #dailyquest_progress[0], 5;
	deletearray #dailyquest_done[0], 5;
	deletearray #dailyquest_diff[0], 5;

	// generate first quest of the day
	callsub S_GenerateNextQuest;

	mes "[Daily Quest Manager]";
	mes "^FF0000[GM]^000000 Your daily quest has been reset.";
	return;
}

function	script	F_LogDailyQuestReward	{
	.@acc = getcharid(3);
	.@cid = getcharid(0);
	.@name$ = strcharinfo(0);

	.@step = getarg(0);
	.@item = getarg(1);
	.@amount = getarg(2);
	.@type = getarg(3);
	.@diff = getarg(4);
	.@target = getarg(5);

	query_sql(
		"INSERT INTO `dailyquest_reward_log` " +
		"(account_id, char_id, char_name, quest_step, reward_item, reward_amount, quest_type, quest_diff, quest_target) VALUES ("+
		.@acc +","+ .@cid +",'"+ escape_sql(.@name$) +"',"+
		.@step +","+ .@item +","+ .@amount +","+ .@type +","+ .@diff +","+ .@target +")"
	);

	return;
}
