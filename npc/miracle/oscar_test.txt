//===== By: ================================================
//= ChatGPT
//===== Version: ============================================
//= 1.0
//===========================================================
prontera,126,148,4	script	Accessory Enchanter	100,{
	mes "[Accessory Enchanter]";
	if (!getequipisequiped(EQI_ACC_R)) {
		mes "Please equip an accessory in your ^FF0000left slot^000000.";
		close;
	}

	set .@eq_id, getequipid(EQI_ACC_R);
	set .@refine, getequiprefinerycnt(EQI_ACC_R);
	set .@card1, getequipcardid(EQI_ACC_R, 0); // real card slot
	set .@card2, getequipcardid(EQI_ACC_R, 1); // enchant slot 1
	set .@card3, getequipcardid(EQI_ACC_R, 2); // enchant slot 2
	set .@card4, getequipcardid(EQI_ACC_R, 3); // enchant slot 3

	// Count used enchant slots (index 1 to 3)
	set .@used_slots, 0;
	if (.@card2) set .@used_slots, .@used_slots + 1;
	if (.@card3) set .@used_slots, .@used_slots + 1;
	if (.@card4) set .@used_slots, .@used_slots + 1;

	if (.@used_slots >= 3) {
		mes "Your accessory already has 3 enchantments (slot 2~4).";
		close;
	}

	// Determine which index to write next (1 ? 2 ? 3)
	set .@target_index, .@used_slots + 1;

	// Orb list
	setarray .orbs[0],
		30000,30001,30002,30003,30004,30005,30006,30007,30008,30009,
		30010,30011,30012,30013,30014,30015,30016,30017,30018,30019,
		30020,30021,30022,30023,30024,30025,30026,30027,30028;

	// Build menu
	set .@menu$, "";
	set .@count, 0;
	for (.@i = 0; .@i < getarraysize(.orbs); .@i++) {
		set .@have, countitem(.orbs[.@i]);
		if (.@have > 0) {
			set .@menu$, .@menu$ + getitemname(.orbs[.@i]) + " (" + .@have + "):";
			set .@orb_index[.@count], .@i;
			set .@count++;
		}
	}

	if (.@menu$ == "") {
		mes "You don't have any enchantment orbs.";
		close;
	}

	mes "Choose an enchantment orb to apply to Slot " + (.@target_index + 1) + ":";
	set .@select, select(.@menu$) - 1;
	set .@orb_id, .orbs[.@orb_index[.@select]];

	if (Zeny < 10000000) {
		mes "You need 10,000,000 Zeny.";
		close;
	}
	if (!countitem(.@orb_id)) {
		mes "You no longer have that orb.";
		close;
	}

	mes "Enchant with ^0000FF" + getitemname(.@orb_id) + "^000000 in slot " + (.@target_index + 1) + "?";
	if (select("Yes:No") == 2) close;

	// Remove original item and orb
	delitem .@eq_id, 1;
	delitem .@orb_id, 1;
	Zeny -= 10000000;

	// Copy current slot data, apply new orb
	setarray .@cards[0], .@card1, .@card2, .@card3, .@card4;
	set .@cards[.@target_index], .@orb_id;

	// Re-give enchanted accessory
	getitem2 .@eq_id, 1, 1, .@refine, 0, .@cards[0], .@cards[1], .@cards[2], .@cards[3];

	mes "Enchant applied to Slot " + (.@target_index + 1) + "!";
	close;
}
