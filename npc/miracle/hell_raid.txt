	setunitdata .@gid,UMOB_RES,.BossRES[.@round];

	setunitdata .@gid,UMOB_MRES,.BossRES[.@round];

	setunitdata .@gid,UMOB_DAMAGETAKEN,100;

	//--------------------------------------------------
	// Additional Boss Settings

	setunitdata .@gid,
		UMOB_LEVEL,
		100 + (.@round * 10);

	return;


//==================================================
// Boss Dead
//==================================================

OnBossDead:

	mapannounce .EventMap$,
		"[Hell Raid] Round "+.round+" cleared!",
		bc_map;

	//--------------------------------------------------
	// Reward Mail
	//--------------------------------------------------

.@count = getmapunits(BL_PC,.EventMap$,.@aid[0]);

for (.@i = 0; .@i < .@count; .@i++) {

	if (!attachrid(.@aid[.@i]))
		continue;

	.@charid = getcharid(0);

	setarray .@item[0], .RewardItem;
	setarray .@amount[0], .RewardAmount[.round];

	mail .@charid,
		"Hell Raid",
		"Round "+.round+" Reward",
		"Congratulations on clearing Round "+.round+" of Hell Raid.",
		0,
		.@item,
		.@amount;

	detachrid;
}

	//--------------------------------------------------
	// Final Round
	//--------------------------------------------------

	if (.round >= .MaxRound) {

		announce "[Hell Raid] Final Round Cleared!",bc_blue|bc_all;

		goto OnStopEvent;
	}

	//--------------------------------------------------
	// Enable Continue NPC
	//--------------------------------------------------

	enablenpc "Hell Raid Guide";

	mapannounce .EventMap$,
		"[Hell Raid] Speak to the Guide to continue.",
		bc_map;

	addtimer 300000,
		strnpcinfo(0)+"::OnContinueTimeout";

	end;


//==================================================
// Continue Timeout
//==================================================

OnContinueTimeout:

	if (.nextround)
		end;

	mapannounce .EventMap$,
		"[Hell Raid] No one continued the raid.",
		bc_map;

	goto OnStopEvent;

//==================================================
// Check Remaining Players
//==================================================

OnPlayerCheck:

	while (.start) {

		if (getmapusers(.EventMap$) <= 0)
			goto OnStopEvent;

		if ((gettimetick(2) - .StartTime) >= 3600) {

			announce "[Hell Raid] Maximum duration reached. Event terminated.",bc_blue|bc_all;

			goto OnStopEvent;
		}

		sleep 10000;
	}

	end;
//==================================================
// Stop Event
//==================================================

OnStopEvent:

	killmonsterall .EventMap$;

	disablenpc "Hell Raid Guide";

	.start = 0;
	.round = 0;
	.mob = 0;
	.nextround = 0;
	.BossGID = 0;
	.StartTime = 0;

	mapannounce .EventMap$,
		"[Hell Raid] Event ended.",
		bc_map;

	end;

OnContinueRound:

	if (.RoundLock)
		end;

	.RoundLock = 1;

	if (.nextround) {
		.RoundLock = 0;
		end;
	}

	.nextround = 1;
	.round++;

	if (.round > .MaxRound) {
		.RoundLock = 0;
		goto OnStopEvent;
	}

	mapannounce .EventMap$,
		"[Hell Raid] Donation target reached. Starting Round "+.round+".",
		bc_map;

	donpcevent strnpcinfo(0)+"::OnStartRound";

	.RoundLock = 0;
	end;
}

ordeal_3-1,150,145,4	script	Hell Raid Guide	4_F_KAFRA1,{

	mes "[Hell Raid Guide]";
	mes "Current Round : "+getvariableofnpc(.round,"Hell Raid Guardian");

	.@start = getvariableofnpc(.StartTime,"Hell Raid Guardian");

	if (.@start > 0) {

		.@remain = 3600 - (gettimetick(2) - .@start);

		if (.@remain < 0)
			.@remain = 0;

		.@hour = .@remain / 3600;
		.@min = (.@remain % 3600) / 60;
		.@sec = .@remain % 60;

		mes "Time Remaining : "+.@hour+"h "+.@min+"m "+.@sec+"s";
	}

	next;

	if (!getvariableofnpc(.nextround,"Hell Raid Guardian")) {

		.@need = getvariableofnpc(.RequiredPoint,"Hell Raid Guardian");
		.@cur = getvariableofnpc(.CurrentPoint,"Hell Raid Guardian");

		mes "[Hell Raid Guide]";
		mes "Next Round Donation";
		mes "Progress : "+.@cur+" / "+.@need+" Points";

		.@remain = .@need - .@cur;

		if (.@remain < 1) {
			mes "Donation goal already reached.";
			close;
		}

		mes "Remaining : "+.@remain+" Points";
		next;

		switch(select("Donate Points:Leave")) {

		case 1:

			//--------------------------------------------------
			// One donation per account
			//--------------------------------------------------

			.@aid = getcharid(3);

			for (.@i = 0; .@i < getarraysize(getvariableofnpc(.DonatedAID,"Hell Raid Guardian")); .@i++) {

				if (getvariableofnpc(.DonatedAID[.@i],"Hell Raid Guardian") == .@aid) {

					mes "[Hell Raid Guide]";
					mes "You have already donated this round.";
					close;
				}
			}

			mes "[Hell Raid Guide]";
			mes "Enter donation amount.";
			mes "Maximum : "+.@remain;
			next;

			input .@donate;

			if (.@donate <= 0) {
				mes "[Hell Raid Guide]";
				mes "Invalid amount.";
				close;
			}

			//--------------------------------------------------
			// Auto cap donation
			//--------------------------------------------------

			if (.@donate > .@remain)
				.@donate = .@remain;

			if (#EVENTVARIABLE < .@donate) {

				mes "[Hell Raid Guide]";
				mes "You do not have enough Event Points.";
				mes "Current Points : "+#EVENTVARIABLE;
				close;
			}

			#EVENTVARIABLE -= .@donate;

			//--------------------------------------------------
			// Save donor
			//--------------------------------------------------

			.@size = getarraysize(getvariableofnpc(.DonatedAID,"Hell Raid Guardian"));

			setd(
				"Hell Raid Guardian::.DonatedAID["+.@size+"]",
				.@aid
			);

			//--------------------------------------------------
			// Add donation
			//--------------------------------------------------

			set getvariableofnpc(.CurrentPoint,"Hell Raid Guardian"),
				getvariableofnpc(.CurrentPoint,"Hell Raid Guardian") + .@donate;

			dispbottom "[Hell Raid] You donated "+.@donate+" Event Points.";

			.@newtotal = getvariableofnpc(.CurrentPoint,"Hell Raid Guardian");

			mapannounce "ordeal_3-1",
				"[Hell Raid] Donation Progress: "+.@newtotal+"/"+.@need+" Points",
				bc_map;

			//--------------------------------------------------
			// Goal reached
			//--------------------------------------------------

			if (.@newtotal >= .@need) {

				mapannounce "ordeal_3-1",
					"[Hell Raid] Donation goal reached! Starting next round.",
					bc_map;

				donpcevent "Hell Raid Guardian::OnContinueRound";
			}

			close;

		case 2:
			warp "SavePoint",0,0;
			close;
		}
	}

	mes "[Hell Raid Guide]";
	mes "The next round is already being prepared.";
	next;

	switch(select("Leave")) {

	case 1:
		warp "SavePoint",0,0;
		close;
	}

	close;
}
