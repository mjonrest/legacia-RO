//--------------------------------------------------
// PvP Achievement Check
//--------------------------------------------------
function	script	F_CheckPvPAchievement	{

	.@count = #achieve_pvp_count;

	setarray .@m[0],10,25,100;
	setarray .@r[0],500,1000,1500;

	for (.@i = 0; .@i < getarraysize(.@m); .@i++) {

		// skip if already claimed this milestone
		if (#achieve_pvp_claimed >= .@m[.@i])
			continue;

		if (.@count >= .@m[.@i]) {

			.@amount = .@r[.@i];

			getitem 30018, .@amount;
			#achievement_coin_total += .@amount;

			#achieve_pvp_claimed = .@m[.@i];

			dispbottom "PvP Achievement Unlocked!", 0xFFD700;
			dispbottom "Activity " + .@m[.@i] + " Reward: " + .@amount + " pcs", 0xFFFF00;

			announce strcharinfo(0) + " reached PvP Activity " + .@m[.@i] + "!", bc_all;

			break; // only one reward per check
		}
	}
	return;
}

//--------------------------------------------------
// MVP Achievement Check
//--------------------------------------------------
function	script	F_CheckMVPAchievement	{

	.@count = #achieve_mvp_count;

	setarray .@m[0],50,100,200,500;
	setarray .@r[0],300,450,750,1000;

	for (.@i = 0; .@i < getarraysize(.@m); .@i++) {

		// skip if already claimed
		if (#achieve_mvp_claimed >= .@m[.@i])
			continue;

		if (.@count >= .@m[.@i]) {

			.@amount = .@r[.@i];

			getitem 30018, .@amount;
			#achievement_coin_total += .@amount;

			#achieve_mvp_claimed = .@m[.@i];

			dispbottom "MVP Achievement Unlocked!", 0xFFD700;
			dispbottom "MVP Killed " + .@m[.@i] + " Reward: " + .@amount + " pcs", 0xFFFF00;

			announce strcharinfo(0) + " has slain " + .@m[.@i] + " MVPs!", bc_all;

			break;
		}
	}
	return;
}

/**

DROP TABLE IF EXISTS `achievement_account_log`;

CREATE TABLE `achievement_account_log` (
  `account_id` INT NOT NULL PRIMARY KEY,
  `char_id` INT NOT NULL,
  `char_name` VARCHAR(30),

  `instance_count` INT DEFAULT 0,
  `instance_claimed` INT DEFAULT 0,

  `pvp_count` INT DEFAULT 0,
  `mvp_count` INT DEFAULT 0,

  `maxlevel_claimed` INT DEFAULT 0,
  `coin_total` INT DEFAULT 0,

  `updated_at` DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

**/

function	script	F_SaveAchievementLog	{

	.@acc = getcharid(3);
	.@cid = getcharid(0);
	.@name$ = strcharinfo(0);

	//========================================
	// READ ACCOUNT VARIABLES FROM SQL
	//========================================
	query_sql("SELECT `value` FROM acc_reg_num WHERE account_id="+.@acc+" AND `key`='#instance_achieve_count'", .@inst);
	query_sql("SELECT `value` FROM acc_reg_num WHERE account_id="+.@acc+" AND `key`='#instance_achieve_claimed'", .@inst_claim);

	query_sql("SELECT `value` FROM acc_reg_num WHERE account_id="+.@acc+" AND `key`='#achieve_pvp_count'", .@pvp);
	query_sql("SELECT `value` FROM acc_reg_num WHERE account_id="+.@acc+" AND `key`='#achieve_pvp_claimed'", .@pvp_claim);

	query_sql("SELECT `value` FROM acc_reg_num WHERE account_id="+.@acc+" AND `key`='#achieve_mvp_count'", .@mvp);
	query_sql("SELECT `value` FROM acc_reg_num WHERE account_id="+.@acc+" AND `key`='#achieve_mvp_claimed'", .@mvp_claim);

	query_sql("SELECT `value` FROM acc_reg_num WHERE account_id="+.@acc+" AND `key`='#achieve_maxlevel_claimed'", .@max);
	query_sql("SELECT `value` FROM acc_reg_num WHERE account_id="+.@acc+" AND `key`='#achievement_coin_total'", .@coin);

	//========================================
	// FALLBACK (if SQL returned NULL)
	//========================================
	if (.@inst < 0)        .@inst       = #instance_achieve_count;
	if (.@inst_claim < 0)  .@inst_claim = #instance_achieve_claimed;

	if (.@pvp < 0)         .@pvp        = #achieve_pvp_count;
	if (.@pvp_claim < 0)   .@pvp_claim  = #achieve_pvp_claimed;

	if (.@mvp < 0)         .@mvp        = #achieve_mvp_count;
	if (.@mvp_claim < 0)   .@mvp_claim  = #achieve_mvp_claimed;

	if (.@max < 0)         .@max        = #achieve_maxlevel_claimed;
	if (.@coin < 0)        .@coin       = #achievement_coin_total;

	//========================================
	// UPSERT INTO LOG TABLE
	//========================================
	query_sql(
		"INSERT INTO achievement_account_log "+
		"(account_id,char_id,char_name,instance_count,instance_claimed,pvp_count,pvp_claimed,mvp_count,mvp_claimed,maxlevel_claimed,coin_total) VALUES ("+
		.@acc+","+.@cid+",'"+escape_sql(.@name$)+"',"+
		.@inst+","+.@inst_claim+","+.@pvp+","+.@pvp_claim+","+.@mvp+","+.@mvp_claim+","+.@max+","+.@coin+") "+
		"ON DUPLICATE KEY UPDATE "+
		"char_id="+.@cid+","+
		"char_name='"+escape_sql(.@name$)+"',"+
		"instance_count="+.@inst+","+
		"instance_claimed="+.@inst_claim+","+
		"pvp_count="+.@pvp+","+
		"pvp_claimed="+.@pvp_claim+","+
		"mvp_count="+.@mvp+","+
		"mvp_claimed="+.@mvp_claim+","+
		"maxlevel_claimed="+.@max+","+
		"coin_total="+.@coin
	);

	dispbottom "[Achievement] Progress snapshot saved.", 0x00FF00;
	return;
}

-	script	loginlogout_achievement	-1,{
OnPCLoginEvent:
	callfunc "F_SaveAchievementLog";
	end;
OnPCLogoutEvent:
	callfunc "F_SaveAchievementLog";
	end;
}
