function	script	F_Comma	{
	.@n$ = ""+getarg(0); // convert to string
	.@len = getstrlen(.@n$);
	.@res$ = "";

	// handle negative numbers
	.@neg = 0;
	if (charat(.@n$,0) == "-") {
		.@neg = 1;
		.@n$ = substr(.@n$,1,.@len-1);
		.@len = getstrlen(.@n$);
	}

	for (.@i = .@len - 1; .@i >= 0; .@i--) {
		.@res$ = charat(.@n$,.@i) + .@res$;
		.@count++;
		if (.@count == 3 && .@i > 0) {
			.@res$ = "," + .@res$;
			.@count = 0;
		}
	}

	if (.@neg) .@res$ = "-" + .@res$;

	return .@res$;
}
