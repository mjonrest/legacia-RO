//===== Legacia RO Script ====================================
//= Hunting Mission Board
//===== Description: =========================================
//= Daily kill missions for Base / Job EXP.
//=
//= - The board keeps 3 missions per character, picked from
//=   normal field/dungeon monsters near the player's Base Level
//=   (list built from the loaded pre-re spawn files).
//= - Turning in a mission gives a % of the EXP needed for the
//=   next Base / Job level, so it is useful from Lv 1 to 99.
//= - Daily cap (.daily_max) is shared by the account AND the
//=   Gepard device (get_unique_id), so alts can't farm it.
//= - Finishing every mission of the day pays a bonus: Miracle
//=   Tonic (12259) + extra EXP.
//= - One free reroll per account per day, extra rerolls cost zeny.
//= - Party kills count for party members near the killer, but only
//=   one member per Gepard device gets the credit for a kill, and
//=   only members within .party_lv_gap levels of the killer.
//= - Kills made while autohunting count at .ah_rate % (default 50,
//=   same as the autohunt EXP penalty; 0 = they don't count).
//= - @mission shows your missions from anywhere.
//=
//= Variables: HM_MOB/HM_NEED/HM_HAVE/HM_AHP (char, per slot),
//= #HM_DAY/#HM_DONE/#HM_RR (account), $HM_DEV_<uid> = day*100+done.
//= Settings are in OnInit. GM (level >= .gm_level) gets test tools.
//============================================================

prontera,168,186,4	script	Hunting Board	4_BOARD3,{
	.@n$ = "[Hunting Board]";
	callsub S_Fill;

	if (getgmlevel() >= .gm_level) {
		mes .@n$;
		mes "GM tools are available.";
		next;
		switch (select("Normal menu:Reset today's count (account + device):Finish my missions:Reroll my missions")) {
		case 2:
			#HM_DAY = 0; #HM_DONE = 0; #HM_RR = 0;
			.@uid = get_unique_id();
			if (.@uid)
				setd callsub(S_DevVar, .@uid), 0;
			mes .@n$;
			mes "Today's count reset.";
			close;
		case 3:
			for (.@i = 0; .@i < .slots; .@i++)
				HM_HAVE[.@i] = HM_NEED[.@i];
			mes .@n$;
			mes "All missions filled.";
			close;
		case 4:
			for (.@i = 0; .@i < .slots; .@i++)
				callsub S_Roll, .@i;
			mes .@n$;
			mes "Missions rerolled.";
			close;
		}
	}

	while (1) {
		.@done = callsub(S_Done);
		mes .@n$;
		mes "Hello, " + strcharinfo(0) + "!";
		mes "Missions done today: ^0055FF" + .@done + "/" + .daily_max + "^000000";
		mes " ";
		callsub S_List;
		next;
		.@rr$ = (#HM_RR == callsub(S_Today)) ? "Reroll missions (" + callfunc("F_InsertComma", .reroll_zeny) + " z)" : "Reroll missions (free today)";
		switch (select("Turn in finished missions:" + .@rr$ + ":How does it work?:Leave")) {
		case 1:
			callsub S_TurnIn;
			break;
		case 2:
			callsub S_Reroll;
			break;
		case 3:
			mes .@n$;
			mes "I always hold ^0055FF" + .slots + " missions^000000 for you,";
			mes "picked for your Base Level.";
			mes "Kill the monsters, then come back";
			mes "to turn them in for ^0055FFBase/Job EXP^000000";
			mes "(" + .exp_pct + "% of your next level each).";
			mes " ";
			mes "You can turn in ^0055FF" + .daily_max + " missions a day^000000";
			mes "per account and per computer.";
			mes "Finish all " + .daily_max + " for a bonus:";
			mes "^0055FF" + .bonus_amt + "x " + getitemname(.bonus_item) + "^000000 + " + .bonus_exp_pct + "% EXP.";
			next;
			mes .@n$;
			mes "^0055FFParty:^000000 kills by party members";
			mes "near you count too (one character";
			mes "per computer, level gap " + .party_lv_gap + " max).";
			if (.ah_rate < 100) {
				mes " ";
				if (.ah_rate > 0)
					mes "^0055FFAutohunt:^000000 kills count at " + .ah_rate + "%.";
				else
					mes "^0055FFAutohunt:^000000 kills don't count.";
			}
			mes " ";
			mes "You get one ^0055FFfree reroll^000000 a day.";
			mes "Type ^0055FF@mission^000000 to see your progress.";
			next;
			break;
		default:
			close;
		}
	}
	end;

// Today's date as YYYYMMDD.
S_Today:
	return atoi(gettimestr("%Y%m%d", 9));

// Name of the device counter variable for a Gepard id.
S_DevVar:
	.@dv_uid = getarg(0);
	return "$HM_DEV_" + (.@dv_uid < 0 ? "n" + (-.@dv_uid) : "" + .@dv_uid);

// Missions turned in today by this account or this device (higher one).
S_Done:
	.@sd_today = callsub(S_Today);
	if (#HM_DAY != .@sd_today) {
		#HM_DAY = .@sd_today;
		#HM_DONE = 0;
	}
	.@sd_done = #HM_DONE;
	.@sd_uid = get_unique_id();
	if (.@sd_uid) {
		.@sd_v = getd(callsub(S_DevVar, .@sd_uid));
		if (.@sd_v / 100 == .@sd_today) {
			if (.@sd_v % 100 > .@sd_done)
				.@sd_done = .@sd_v % 100;
		}
	}
	return .@sd_done;

// Gives every empty slot a mission.
S_Fill:
	for (.@sf_i = 0; .@sf_i < .slots; .@sf_i++) {
		if (!HM_MOB[.@sf_i])
			callsub S_Roll, .@sf_i;
	}
	return;

// Picks a new monster for slot getarg(0).
S_Roll:
	.@r_slot = getarg(0);
	.@r_lo = BaseLevel - .lv_below;
	.@r_hi = BaseLevel + .lv_above;
	for (.@r_try = 0; .@r_try < 6; .@r_try++) {
		deletearray .@r_c;
		.@r_cn = 0;
		for (.@r_i = 0; .@r_i < .mob_count; .@r_i++) {
			.@r_lv = .mob_lv[.@r_i];
			if (.@r_lv < .@r_lo || .@r_lv > .@r_hi)
				continue;
			// no monster twice on the board
			.@r_dup = 0;
			for (.@r_j = 0; .@r_j < .slots; .@r_j++) {
				if (.@r_j != .@r_slot) {
					if (HM_MOB[.@r_j] == .mob_id[.@r_i])
						.@r_dup = 1;
				}
			}
			if (.@r_dup)
				continue;
			.@r_c[.@r_cn] = .@r_i;
			.@r_cn++;
		}
		if (.@r_cn >= .slots)
			break;
		.@r_lo -= 5;
		.@r_hi += 5;
	}
	if (.@r_cn < 1) {
		HM_MOB[.@r_slot] = 0;
		HM_NEED[.@r_slot] = 0;
		HM_HAVE[.@r_slot] = 0;
		HM_AHP[.@r_slot] = 0;
		return;
	}
	.@r_pick = 0;
	if (.@r_cn > 1)
		.@r_pick = rand(.@r_cn);
	.@r_steps = (.kill_max - .kill_min) / 10 + 1;
	.@r_need = .kill_min;
	if (.@r_steps > 1)
		.@r_need += rand(.@r_steps) * 10;
	HM_MOB[.@r_slot] = .mob_id[.@r_c[.@r_pick]];
	HM_NEED[.@r_slot] = .@r_need;
	HM_HAVE[.@r_slot] = 0;
	HM_AHP[.@r_slot] = 0;
	return;

// Index in .mob_id of a monster id (-1 if not found).
S_MobIdx:
	.@mi_id = getarg(0);
	for (.@mi_i = 0; .@mi_i < .mob_count; .@mi_i++) {
		if (.mob_id[.@mi_i] == .@mi_id)
			return .@mi_i;
	}
	return -1;

// Prints the missions (works with mes, used by the board and @mission).
S_List:
	for (.@sl_i = 0; .@sl_i < .slots; .@sl_i++) {
		if (!HM_MOB[.@sl_i]) {
			mes (.@sl_i + 1) + ". ^777777(no monster for your level)^000000";
			continue;
		}
		.@sl_k = callsub(S_MobIdx, HM_MOB[.@sl_i]);
		.@sl_where$ = (.@sl_k >= 0) ? .mob_map$[.@sl_k] : "?";
		.@sl_col$ = (HM_HAVE[.@sl_i] >= HM_NEED[.@sl_i]) ? "^00AA00" : "^000000";
		mes (.@sl_i + 1) + ". " + .@sl_col$ + "Kill " + HM_NEED[.@sl_i] + " " + getmonsterinfo(HM_MOB[.@sl_i], MOB_NAME) + " (" + HM_HAVE[.@sl_i] + "/" + HM_NEED[.@sl_i] + ")^000000";
		mes "    ^777777Lv " + getmonsterinfo(HM_MOB[.@sl_i], MOB_LV) + ", found in " + .@sl_where$ + "^000000";
	}
	return;

S_TurnIn:
	.@n$ = "[Hunting Board]";
	.@t_uid = get_unique_id();
	if (!.@t_uid) {
		mes .@n$;
		mes "I can't see your Gepard ID.";
		mes "Please relog and try again.";
		next;
		return;
	}
	.@t_today = callsub(S_Today);
	.@t_got = 0;
	.@t_bexp = 0;
	.@t_jexp = 0;
	.@t_bonus = 0;
	for (.@t_i = 0; .@t_i < .slots; .@t_i++) {
		if (!HM_MOB[.@t_i])
			continue;
		if (HM_HAVE[.@t_i] < HM_NEED[.@t_i])
			continue;
		.@t_done = callsub(S_Done);
		if (.@t_done >= .daily_max)
			break;
		.@t_done++;
		#HM_DONE = .@t_done;
		setd callsub(S_DevVar, .@t_uid), .@t_today * 100 + .@t_done;
		.@t_b = NextBaseExp * .exp_pct / 100;
		.@t_j = NextJobExp * .exp_pct / 100;
		.@t_bexp += .@t_b;
		.@t_jexp += .@t_j;
		if (.@t_b > 0 || .@t_j > 0)
			getexp2 .@t_b, .@t_j;
		.@t_got++;
		HM_MOB[.@t_i] = 0;
		if (.@t_done == .daily_max)
			.@t_bonus = 1;
	}
	mes .@n$;
	if (!.@t_got) {
		if (callsub(S_Done) >= .daily_max) {
			mes "You already did ^FF0000" + .daily_max + " missions^000000 today";
			mes "on this account or computer.";
			mes "Come back tomorrow!";
		} else {
			mes "None of your missions are";
			mes "finished yet. Keep hunting!";
		}
		next;
		return;
	}
	if (.@t_bonus) {
		.@t_b = NextBaseExp * .bonus_exp_pct / 100;
		.@t_j = NextJobExp * .bonus_exp_pct / 100;
		if (.@t_b > 0 || .@t_j > 0)
			getexp2 .@t_b, .@t_j;
		if (.bonus_item)
			getitem .bonus_item, .bonus_amt;
	}
	specialeffect2 EF_ANGEL2;
	mes "^00AA00" + .@t_got + " mission" + (.@t_got > 1 ? "s" : "") + " turned in!^000000";
	mes "+" + callfunc("F_InsertComma", .@t_bexp) + " Base EXP";
	mes "+" + callfunc("F_InsertComma", .@t_jexp) + " Job EXP";
	if (.@t_bonus) {
		mes " ";
		mes "^0055FFAll " + .daily_max + " missions done today!^000000";
		mes "Bonus: +" + .bonus_exp_pct + "% Base/Job EXP";
		if (.bonus_item)
			mes "and " + .bonus_amt + "x " + getitemname(.bonus_item) + ".";
	}
	callsub S_Fill;
	next;
	return;

S_Reroll:
	.@n$ = "[Hunting Board]";
	.@rr_today = callsub(S_Today);
	.@rr_free = (#HM_RR != .@rr_today);
	mes .@n$;
	mes "Reroll all ^0055FFunfinished^000000 missions?";
	mes "Their progress is lost.";
	if (.@rr_free)
		mes "This one is ^00AA00free^000000 (1 per day).";
	else
		mes "It costs ^FF0000" + callfunc("F_InsertComma", .reroll_zeny) + " zeny^000000.";
	next;
	if (select("Yes, reroll:No") == 2)
		return;
	if (!.@rr_free) {
		if (.reroll_zeny <= 0) {
			mes .@n$;
			mes "You already used today's reroll.";
			next;
			return;
		}
		if (Zeny < .reroll_zeny) {
			mes .@n$;
			mes "You don't have enough zeny.";
			next;
			return;
		}
		Zeny -= .reroll_zeny;
	} else
		#HM_RR = .@rr_today;
	for (.@rr_i = 0; .@rr_i < .slots; .@rr_i++) {
		if (!HM_MOB[.@rr_i]) {
			callsub S_Roll, .@rr_i;
			continue;
		}
		if (HM_HAVE[.@rr_i] < HM_NEED[.@rr_i])
			callsub S_Roll, .@rr_i;
	}
	return;

// Adds a kill of monster name getarg(0) to the attached player.
// getarg(1) = 1 when the kill was made while autohunting.
S_Credit:
	.@cr_name$ = getarg(0);
	.@cr_ah = getarg(1);
	for (.@cr_i = 0; .@cr_i < .slots; .@cr_i++) {
		if (!HM_MOB[.@cr_i])
			continue;
		if (HM_HAVE[.@cr_i] >= HM_NEED[.@cr_i])
			continue;
		if (getmonsterinfo(HM_MOB[.@cr_i], MOB_NAME) != .@cr_name$)
			continue;
		if (.@cr_ah) {
			if (.ah_rate <= 0)
				return;
			HM_AHP[.@cr_i] += .ah_rate;
			if (HM_AHP[.@cr_i] < 100)
				return;
			HM_AHP[.@cr_i] -= 100;
		}
		HM_HAVE[.@cr_i]++;
		if (HM_HAVE[.@cr_i] >= HM_NEED[.@cr_i])
			dispbottom "[Hunting Board] Mission done: " + .@cr_name$ + " " + HM_NEED[.@cr_i] + "/" + HM_NEED[.@cr_i] + ". Turn it in at the board in Prontera.", 0x00CC00;
		else if (HM_HAVE[.@cr_i] % 10 == 0)
			dispbottom "[Hunting Board] " + .@cr_name$ + ": " + HM_HAVE[.@cr_i] + "/" + HM_NEED[.@cr_i], 0x00CCFF;
		return;
	}
	return;

OnNPCKillEvent:
	.@name$ = getmonsterinfo(killedrid, MOB_NAME);
	.@ah = isautohunting();
	.@kcid = getcharid(0);
	.@kaid = getcharid(3);
	.@kuid = get_unique_id();
	.@klv = BaseLevel;
	.@pid = getcharid(1);
	getmapxy(.@km$, .@kx, .@ky);
	callsub S_Credit, .@name$, .@ah;
	if (!.@pid)
		end;
	if (.party_range <= 0)
		end;

	// Party credit: one member per Gepard device (the killer's device included).
	.@un = 0;
	if (.@kuid) {
		.@uids[0] = .@kuid;
		.@un = 1;
	}
	getpartymember .@pid, 1, .@pcid;
	getpartymember .@pid, 2, .@paid;
	.@pc = $@partymembercount;
	for (.@m = 0; .@m < .@pc; .@m++) {
		if (.@pcid[.@m] == .@kcid)
			continue;
		if (!isloggedin(.@paid[.@m], .@pcid[.@m]))
			continue;
		if (!attachrid(.@paid[.@m]))
			continue;
		if (getcharid(0) != .@pcid[.@m])
			continue;
		if (Hp < 1)
			continue;
		if (.party_lv_gap > 0) {
			if (BaseLevel - .@klv > .party_lv_gap || .@klv - BaseLevel > .party_lv_gap)
				continue;
		}
		getmapxy(.@mm$, .@mx, .@my);
		if (.@mm$ != .@km$)
			continue;
		if (distance(.@kx, .@ky, .@mx, .@my) > .party_range)
			continue;
		.@uid = get_unique_id();
		if (!.@uid)
			continue;
		if (inarray(.@uids, .@uid) != -1)
			continue;
		.@uids[.@un] = .@uid;
		.@un++;
		.@mah = .@ah;
		if (isautohunting())
			.@mah = 1;
		callsub S_Credit, .@name$, .@mah;
	}
	end;

OnCmd:
	mes "[Hunting Board]";
	mes "Missions done today: ^0055FF" + callsub(S_Done) + "/" + .daily_max + "^000000";
	mes " ";
	callsub S_Fill;
	callsub S_List;
	mes " ";
	mes "Turn in at the ^0055FFHunting Board^000000";
	mes "in Prontera (168, 186).";
	close;

	.bonus_exp_pct = 10;     // extra % EXP when all .daily_max missions are done
	.bonus_item = 12259;     // Miracle Tonic, 0 = none
	.bonus_amt = 1;
	.kill_min = 40;          // kills needed per mission, rounded to 10
	.kill_max = 80;
	.lv_below = 5;           // monster level window around Base Level

	.reroll_zeny = 100000;   // cost after the free daily reroll, 0 = no paid rerolls
	.ah_rate = 50;           // % a kill counts while autohunting (0 = not at all, 100 = full)
	.party_range = 14;       // party kills count for members within this many cells (0 = off)
	.party_lv_gap = 15;      // and within this many Base Levels of the killer (0 = no limit)

	bindatcmd "mission", strnpcinfo(3) + "::OnCmd";
