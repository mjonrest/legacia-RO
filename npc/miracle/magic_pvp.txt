new_1-1,45,113,4	script	Novice Magic Tournament Gate	734,{
mes "[Battle Rumble]";
switch(select(
	"Enter.",
	"Information." +
	((getgmlevel() >= 99) ? ":GM Menu." : "") +
	":Cancel."
)) {
	case 1:
		// Entry check here
		if (!.enabled) {
			mes "The gate is currently ^FF0000closed^000000.";
			close;
		}
		goto L_CheckEntry;
	case 2:
		mes "You will be given rental gear for 30 minutes.";
		mes "You must be completely clean: no items, no equipment.";
		mes "Make sure you're ready!";
		close;
	case 3:
		if (getgmlevel() >= 99) {
			mes "[GM Menu]";
			if (.enabled)
				if (select("Disable Gate", "Back") == 1) .enabled = false;
				else close;
			else
				if (select("Enable Gate", "Back") == 1) .enabled = true;
				else close;

			mes (.enabled) ? "Gate is ^00FF00Enabled^000000." : "Gate is ^FF0000Disabled^000000.";
			close;
		}
		// fallthrough to Cancel
	default:
		close;
}

L_CheckEntry:

	// Job must be Novice (Job ID 0), and Base Level must be 99
	if (BaseLevel != 99 || Class != Job_Novice) {
		mes "Only Novices at Base Level 99 may join the Battle Rumble.";
		close;
	}
	// Inventory must be empty
	getinventorylist;
	if (@inventorylist_count) {
		mes "You must empty your inventory first.";
		close;
	}

	// Clear all buffs
	sc_end SC_ALL;
	sc_end SC_STRFood;
	sc_end SC_AGIFood;
	sc_end SC_VITFood;
	sc_end SC_INTFood;
	sc_end SC_DEXFood;
	sc_end SC_LUKFood;

	// Pick random items from each list
	.@item1 = .item_list1[rand(getarraysize(.item_list1))];
	.@item2 = .item_list2[rand(getarraysize(.item_list2))];
	.@item3 = .item_list3[rand(getarraysize(.item_list3))];

	// Store all rented items in a list
	setarray .@rent_list[0],
		.@item1,
		.@item2,
		.@item3,
		24693,24694,24695,24696,24697,1639;

	// Give all items with 30-minute rental duration
	for (.@i = 0; .@i < getarraysize(.@rent_list); .@i++) {
		rentitem .@rent_list[.@i], 1800;
	}

	// Wait for inventory to update
	sleep2 100;

	// Equip all rented items
	getinventorylist;
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		for (.@j = 0; .@j < getarraysize(.@rent_list); .@j++) {
			if (@inventorylist_id[.@i] == .@rent_list[.@j]) {
				equip @inventorylist_id[.@i];
				break;
			}
		}
	}

	// Random warp into the battle map
	percentheal 100,100;
	warp "pvp_y_1-2", 0, 0;
	end;

OnInit:
	.enabled = false; // Gate is initially closed

	// Item list 1

pvp_y_1-2,0,0,0	script	BR_Announcer	-1,{
OnPCKillEvent:
	if (strcharinfo(3) == "pvp_y_1-2"){

	.@map$ = "pvp_y_1-2";

	// Get all players on the map
	.@total = getmapunits(BL_PC,.@map$, .@map_units$[0]);

	// Filter out GMs with level > 5
	.@count = 0;
	for (.@i = 0; .@i < .@total; .@i++) {
		.@rid = .@map_units$[.@i];
		if (getgmlevel(.@rid) <= 5) {
			.@real_players[.@count] = .@rid;
			.@count++;
		}
	}

	if (.@count <= 1) {
		if (.@count == 1) {
			.@last_id = .@real_players[0];
			.@name$ = rid2name(.@last_id);
			announce "[Battle Rumble] Only one player remains: " + .@name$ + "!", 0;
		} else {
			announce "[Battle Rumble] No eligible players remain in the arena.", 0;
		}
		end;
	}

	announce "[Battle Rumble] " + .@count + " players remain:", 0;
	for (.@i = 0; .@i < .@count; .@i++) {
		announce " - " + rid2name(.@real_players[.@i]), 0;
	}
	end;
	}
	end;
}
