/**
CREATE TABLE IF NOT EXISTS `battlepass_season` (
    `season_id` INT UNSIGNED NOT NULL AUTO_INCREMENT,
    `start_at` DATETIME NOT NULL,
    `end_at` DATETIME NOT NULL,
    `status` ENUM('ACTIVE','ENDED') NOT NULL DEFAULT 'ACTIVE',
    `created_by` INT UNSIGNED NOT NULL DEFAULT 0,
    `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    `ended_at` DATETIME DEFAULT NULL,
    PRIMARY KEY (`season_id`),
    KEY `idx_bp_season_status` (`status`),
    KEY `idx_bp_season_end` (`end_at`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `battlepass_progress` (
    `char_id` INT UNSIGNED NOT NULL,
    `account_id` INT UNSIGNED NOT NULL,
    `season` INT UNSIGNED NOT NULL,
    `level` INT UNSIGNED NOT NULL DEFAULT 0,
    `points` INT UNSIGNED NOT NULL DEFAULT 0,
    `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (`char_id`, `season`),
    KEY `idx_bp_progress_account` (`account_id`),
    KEY `idx_bp_progress_season` (`season`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE IF NOT EXISTS `battlepass_log` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    `account_id` INT UNSIGNED NOT NULL,
    `char_id` INT UNSIGNED NOT NULL,
    `season` INT UNSIGNED NOT NULL,
    `action` VARCHAR(50) NOT NULL,
    `item_id` INT UNSIGNED NOT NULL DEFAULT 0,
    `amount` INT NOT NULL DEFAULT 0,
    `points` INT NOT NULL DEFAULT 0,
    `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    KEY `idx_bp_log_season` (`season`),
    KEY `idx_bp_log_char` (`char_id`),
    KEY `idx_bp_log_account` (`account_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
**/

prontera,164,166,3	script	BattlePass Manager	116,{

	callsub S_SeasonCheck;

	mes "[Battle Pass]";
	mes "-------------------";

	if (.SeasonActive) {
		mes "^0000FFSeason^000000: "+.SeasonID;
		mes "^0000FFStart^000000: "+.SeasonStart$;
		mes "^0000FFEnd^000000: "+.SeasonEnd$;
	}
	else {
		mes "^FF0000No Active Season^000000";
	}

	mes "^0000FFLevel^000000: ^FF0000"+#BP_Level;
	mes "^0000FFPoints^000000: "+#BP_Points;
	mes "-------------------";

	// No active season.
	if (!.SeasonActive) {

		mes "^FF0000The Battle Pass is currently inactive.^000000";
		next;

		if (getgmlevel() >= .GMLevel) {

			if (select(
				">> GM Season Management",
				">> Exit"
			) == 1) {

				callsub S_GMMenu;
				close;
			}
		}

		close;
	}

	// Progress information.
	if (#BP_Level < 50) {

		.@need = .NeedPoint[#BP_Level+1];

		if (.@need > 0) {

			.@remain = max(0, .@need - #BP_Points);
			.@percent = (#BP_Points * 100) / .@need;

			if (.@percent > 100)
				.@percent = 100;
		}
		else {

			.@remain = 0;
			.@percent = 0;
		}

		mes "^0000FFNext Level^000000: ^00FF00"+.@need+" pts";
		mes "^0000FFProgress^000000: ^00FF00"+.@percent+"%%";
	}
	else {

		mes "^00FF00MAX LEVEL REACHED^000000";
	}

	next;

	.@menu$ =
		">> ^FF0000Sell Items^000000:"+
		">> ^00FF00Claim Rewards^000000:"+
		">> Season Information:"+
		">> ^0000FFReward List^000000";

	if (getgmlevel() >= .GMLevel)
		.@menu$ += ":>> ^FF0000GM Season Management^000000";

	.@menu$ += ":>> Exit";

	.@choice = select(.@menu$);

	switch (.@choice) {

		case 1:
			callsub S_SellMenu;
			close;

		case 2:
			callsub S_ClaimMenu;
			close;

		case 3:
			callsub S_Info;
			close;

		case 4:
			callsub S_List;
			close;

		case 5:
			if (getgmlevel() >= .GMLevel) {
				callsub S_GMMenu;
				close;
			}
			close;

		default:
			close;
	}

	end;


	//------------------------------------------------
	// SEASON CHECK
	//------------------------------------------------

S_SeasonCheck:

	// Expire old seasons automatically.
	query_sql(
		"UPDATE battlepass_season "+
		"SET status='ENDED', ended_at=NOW() "+
		"WHERE status='ACTIVE' "+
		"AND end_at <= NOW()"
	);

	// Clear current NPC season data.
	.SeasonID = 0;
	.SeasonActive = 0;
	.SeasonStart$ = "";
	.SeasonEnd$ = "";

	// Find current active season.
	query_sql(
		"SELECT season_id, "+
		"DATE_FORMAT(start_at,'%Y-%m-%d %H:%i:%s'), "+
		"DATE_FORMAT(end_at,'%Y-%m-%d %H:%i:%s') "+
		"FROM battlepass_season "+
		"WHERE status='ACTIVE' "+
		"AND start_at <= NOW() "+
		"AND end_at > NOW() "+
		"ORDER BY season_id DESC "+
		"LIMIT 1",
		.SeasonID,
		.SeasonStart$,
		.SeasonEnd$
	);

	if (.SeasonID <= 0)
		return;

	.SeasonActive = 1;

	// New season detected for this character.
	if (#BP_Season != .SeasonID) {

		#BP_Season = .SeasonID;
		#BP_Level = 0;
		#BP_Points = 0;

		deletearray #BP_Claimed[0],50;

		// Save new season progress.
		query_sql(
			"REPLACE INTO battlepass_progress "+
			"(char_id,account_id,season,level,points,updated_at) VALUES ("+
			getcharid(0)+","+
			getcharid(3)+","+
			.SeasonID+","+
			#BP_Level+","+
			#BP_Points+","+
			"NOW())"
		);
	}

	return;


	//------------------------------------------------
	// GM SEASON MENU
	//------------------------------------------------

S_GMMenu:

	if (getgmlevel() < .GMLevel) {
		mes "[Battle Pass]";
		mes "^FF0000Access denied.^000000";
		return;
	}

	while (1) {

		mes "[Battle Pass - GM]";
		mes "-------------------";

		if (.SeasonActive) {

			mes "Current Season: ^0000FF"+.SeasonID+"^000000";
			mes "Start: "+.SeasonStart$;
			mes "End: "+.SeasonEnd$;
		}
		else {

			mes "^FF0000No active season.^000000";
		}

		next;

		switch(select(
			"Start New Season",
			"View Current Season",
			"End Current Season Now",
			"Back"
		)) {

			case 1:
				callsub S_GMStartSeason;
				break;

			case 2:
				callsub S_GMViewSeason;
				break;

			case 3:
				callsub S_GMEndSeason;
				break;

			case 4:
				return;
		}
	}

	return;


	//------------------------------------------------
	// GM - START NEW SEASON
	//------------------------------------------------

S_GMStartSeason:

	if (getgmlevel() < .GMLevel)
		return;

	if (.SeasonActive) {

		mes "[Battle Pass]";
		mes "^FF0000An active season already exists.^000000";
		mes "";
		mes "Season: "+.SeasonID;
		mes "End: "+.SeasonEnd$;

		next;
		return;
	}

	mes "[Battle Pass - New Season]";
	mes "How many days should the season last?";
	mes "";
	mes "Example:";
	mes "^00FF0015^000000 = 15 days";
	mes "^00FF0030^000000 = 30 days";
	next;

	input .@days;

	if (.@days < 1) {

		mes "Invalid duration.";
		mes "Minimum is 1 day.";
		next;
		return;
	}

	if (.@days > 365) {

		mes "Invalid duration.";
		mes "Maximum is 365 days.";
		next;
		return;
	}

	mes "[Confirmation]";
	mes "";
	mes "Duration: ^00FF00"+.@days+" days^000000";
	mes "";
	mes "The season will start now.";
	mes "The end date will be calculated automatically.";
	next;

	if (select(
		"^00FF00Start Season^000000",
		"Cancel"
	) == 2)
		return;

	// Safety check.
	query_sql(
		"UPDATE battlepass_season "+
		"SET status='ENDED', ended_at=NOW() "+
		"WHERE status='ACTIVE' "+
		"AND end_at <= NOW()"
	);

	.@active = 0;

	query_sql(
		"SELECT COUNT(*) "+
		"FROM battlepass_season "+
		"WHERE status='ACTIVE' "+
		"AND start_at <= NOW() "+
		"AND end_at > NOW()",
		.@active
	);

	if (.@active > 0) {

		mes "[Battle Pass]";
		mes "^FF0000Cannot start a new season.^000000";
		mes "An active season already exists.";
		next;
		return;
	}

	// Create new season.
	query_sql(
		"INSERT INTO battlepass_season "+
		"(start_at,end_at,status,created_by,created_at) "+
		"VALUES "+
		"(NOW(),"+
		"DATE_ADD(NOW(),INTERVAL "+.@days+" DAY),"+
		"'ACTIVE',"+
		getcharid(3)+","+
		"NOW())"
	);

	// Get new Season ID.
	query_sql(
		"SELECT LAST_INSERT_ID()",
		.SeasonID
	);

	// Load dates.
	query_sql(
		"SELECT "+
		"DATE_FORMAT(start_at,'%Y-%m-%d %H:%i:%s'), "+
		"DATE_FORMAT(end_at,'%Y-%m-%d %H:%i:%s') "+
		"FROM battlepass_season "+
		"WHERE season_id="+.SeasonID,
		.SeasonStart$,
		.SeasonEnd$
	);

	.SeasonActive = 1;

	mes "[Battle Pass]";
	mes "^00FF00Season started successfully!^000000";
	mes "";
	mes "Season ID: ^0000FF"+.SeasonID+"^000000";
	mes "";
	mes "Start:";
	mes "^00FF00"+.SeasonStart$+"^000000";
	mes "";
	mes "End:";
	mes "^FF0000"+.SeasonEnd$+"^000000";
	mes "";
	mes "Duration: "+.@days+" days";

	next;
	return;


	//------------------------------------------------
	// GM - VIEW SEASON
	//------------------------------------------------

S_GMViewSeason:

	if (getgmlevel() < .GMLevel)
		return;

	mes "[Battle Pass - Season Information]";
	mes "---------------------------";

	if (!.SeasonActive) {

		mes "^FF0000No active season.^000000";
		next;
		return;
	}

	.@remaining = 0;

	query_sql(
		"SELECT GREATEST(0,TIMESTAMPDIFF(SECOND,NOW(),end_at)) "+
		"FROM battlepass_season "+
		"WHERE season_id="+.SeasonID,
		.@remaining
	);

	.@days = .@remaining / 86400;
	.@hours = (.@remaining % 86400) / 3600;
	.@minutes = (.@remaining % 3600) / 60;

	mes "Season ID: ^0000FF"+.SeasonID+"^000000";
	mes "";
	mes "Start:";
	mes .SeasonStart$;
	mes "";
	mes "End:";
	mes .SeasonEnd$;
	mes "";
	mes "Remaining:";
	mes "^00FF00"+.@days+" days "+
		.@hours+" hours "+
		.@minutes+" minutes^000000";

	next;
	return;


	//------------------------------------------------
	// GM - END SEASON NOW
	//------------------------------------------------

S_GMEndSeason:

	if (getgmlevel() < .GMLevel)
		return;

	if (!.SeasonActive) {

		mes "[Battle Pass]";
		mes "There is no active season.";
		next;
		return;
	}

	mes "[Battle Pass - End Season]";
	mes "";
	mes "^FF0000WARNING!^000000";
	mes "";
	mes "You are about to end:";
	mes "^0000FFSeason "+.SeasonID+"^000000";
	mes "";
	mes "Scheduled end:";
	mes .SeasonEnd$;
	mes "";
	mes "Are you sure?";

	next;

	if (select(
		"^FF0000END SEASON NOW^000000",
		"Cancel"
	) == 2)
		return;

	query_sql(
		"UPDATE battlepass_season "+
		"SET status='ENDED', "+
		"ended_at=NOW(), "+
		"end_at=NOW() "+
		"WHERE season_id="+.SeasonID+" "+
		"AND status='ACTIVE'"
	);

	.SeasonActive = 0;
	.SeasonEnd$ = "";

	mes "[Battle Pass]";
	mes "^00FF00Season "+.SeasonID+" has been ended.^000000";

	next;
	return;

	//------------------------------------------------
	// SELL MENU
	//------------------------------------------------

S_SellMenu:

	switch(select(
		"Sell Normal Items",
		"Sell Refined Equipment",
		"Sell MVP Materials",
		"Back"
	)) {

		case 1:
			callsub S_SellItems;
			break;

		case 2:
			callsub S_SellRefine;
			break;

		case 3:
			callsub S_SellMVP;
			break;

		case 4:
			return;
	}

	return;


	//------------------------------------------------
	// CLAIM MENU
	//------------------------------------------------

S_ClaimMenu:

	mes "[Battle Pass]";
	mes "Available rewards:";
	next;

	.@menu$ = "";
	deletearray .@lvl[0],50;

	for (.@i = 1; .@i <= #BP_Level; .@i++) {

		if (!#BP_Claimed[.@i]) {

			.@menu$ +=
				"Lv "+.@i+" -> "+
				getitemname(.RewardItem[.@i])+":";

			.@lvl[getarraysize(.@lvl)] = .@i;
		}
	}

	if (.@menu$ == "") {

		mes "No rewards available.";
		return;
	}

	.@s = select(.@menu$)-1;

	.@level = .@lvl[.@s];

	getitem .RewardItem[.@level],.RewardAmount[.@level];

	#BP_Claimed[.@level] = 1;

	callsub S_Log,
		"CLAIM_REWARD",
		.RewardItem[.@level],
		.RewardAmount[.@level],
		0;

	mes "Reward claimed.";

	return;


	//------------------------------------------------
	// SELL NORMAL ITEMS
	//------------------------------------------------

S_SellItems:

	mes "[Battle Pass]";
	mes "Scanning inventory...";
	next;

	for (.@i = 0; .@i < getarraysize(.SellItemID); .@i++) {

		.@count = countitem(.SellItemID[.@i]);

		if (.@count) {

			mes getitemname(.SellItemID[.@i])+
				" x"+.@count+
				" -> "+
				(.SellItemPoint[.@i] * .@count)+
				" BP";

			if (select("Sell","Skip") == 1) {

				delitem .SellItemID[.@i],.@count;

				.@points =
					.SellItemPoint[.@i] * .@count;

				callsub S_AddPoints,.@points;

				callsub S_Log,
					"SELL_ITEM",
					.SellItemID[.@i],
					.@count,
					.@points;
			}
		}
	}

	return;


	//------------------------------------------------
	// SELL REFINED EQUIPMENT
	//------------------------------------------------

S_SellRefine:

	mes "[Battle Pass]";
	mes "Scanning refined equipment...";
	next;

	getinventorylist;

	.@menu$ = "";
	deletearray .@list[0],100;

	for (.@i = 0; .@i < @inventorylist_count; .@i++) {

		.@item = @inventorylist_id[.@i];
		.@refine = @inventorylist_refine[.@i];

		if (.@refine < 7)
			continue;

		.@type = getiteminfo(.@item,ITEMINFO_TYPE);

		if (.@type != IT_WEAPON &&
			.@type != IT_ARMOR)
			continue;

		if (.@type == IT_WEAPON) {

			if (getiteminfo(
				.@item,
				ITEMINFO_WEAPONLEVEL
			) != 4)
				continue;
		}

		.@menu$ +=
			getitemname(.@item)+
			" +"+.@refine+":";

		.@list[getarraysize(.@list)] = .@i;
	}

	if (.@menu$ == "") {

		mes "No refined equipment available.";
		return;
	}

	.@s = select(.@menu$)-1;

	.@idx = .@list[.@s];

	.@item = @inventorylist_id[.@idx];
	.@refine = @inventorylist_refine[.@idx];

	.@points =
		(.@refine - 6) * .RefinePoint;

	mes "Sell "+getitemname(.@item)+
		" +"+.@refine+" ->";

	mes "You will receive "+
		.@points+" BP Points.";

	next;

	if (select("Confirm","Cancel") == 1) {

		delitem .@item,1;

		callsub S_AddPoints,.@points;

		callsub S_Log,
			"SELL_REFINE",
			.@item,
			1,
			.@points;

		mes "Item sold.";
	}

	return;


	//------------------------------------------------
	// SELL MVP MATERIALS
	//------------------------------------------------

S_SellMVP:

	mes "[Battle Pass]";
	next;

	for (.@i = 0; .@i < getarraysize(.MVPMatID); .@i++) {

		.@count = countitem(.MVPMatID[.@i]);

		if (.@count) {

			mes getitemname(.MVPMatID[.@i])+
				" x"+.@count+
				" -> "+
				(.MVPMatPoint[.@i] * .@count)+
				" BP";

			if (select("Sell","Skip") == 1) {

				delitem .MVPMatID[.@i],.@count;

				.@points =
					.MVPMatPoint[.@i] * .@count;

				callsub S_AddPoints,.@points;

				callsub S_Log,
					"SELL_MVP",
					.MVPMatID[.@i],
					.@count,
					.@points;
			}
		}
	}

	return;

	//------------------------------------------------
	// ADD POINTS + LEVEL CHECK
	//------------------------------------------------

S_AddPoints:

	.@add = getarg(0);

	// Do not allow points when season is inactive.
	if (!.SeasonActive) {

		dispbottom
			"[Battle Pass] The season has ended.";

		return;
	}

	while (.@add > 0 && #BP_Level < 50) {

		.@need =
			.NeedPoint[#BP_Level+1];

		.@remain =
			.@need - #BP_Points;

		if (.@add >= .@remain) {

			// LEVEL UP
			.@add -= .@remain;

			#BP_Points = 0;

			#BP_Level++;

			dispbottom
				"[Battle Pass] Level Up! Now Lv "+
				#BP_Level;

		}
		else {

			#BP_Points += .@add;

			.@add = 0;
		}
	}

	// Save progress.
	query_sql(
		"REPLACE INTO battlepass_progress "+
		"(char_id,account_id,season,level,points,updated_at) VALUES ("+
		getcharid(0)+","+
		getcharid(3)+","+
		.SeasonID+","+
		#BP_Level+","+
		#BP_Points+","+
		"NOW())"
	);

	return;


	//------------------------------------------------
	// INFO MENU
	//------------------------------------------------

S_Info:

	while (1) {

		mes "[Battle Pass Information]";

		switch(select(
			"Season Overview",
			"Sellable Normal Items",
			"MVP Materials",
			"Refine Rewards Info",
			"Back"
		)) {

			case 1:
				callsub S_Info_Season;
				break;

			case 2:
				callsub S_Info_Normal;
				break;

			case 3:
				callsub S_Info_MVP;
				break;

			case 4:
				callsub S_Info_Refine;
				break;

			case 5:
				return;
		}
	}

	return;


	//------------------------------------------------
	// SEASON INFO
	//------------------------------------------------

S_Info_Season:
next;
	mes "[Battle Pass Season]";
	mes "Season ID: "+.SeasonID;
	mes "Start: "+.SeasonStart$;
	mes "End: "+.SeasonEnd$;
	mes "Max Level: 50";
	next;

	mes "[Your Progress]";
	mes "Level: "+#BP_Level;
	mes "Points: "+#BP_Points;

	return;


	//------------------------------------------------
	// NORMAL ITEM INFO
	//------------------------------------------------

S_Info_Normal:
	next;

	.@page = 0;
	.@total = getarraysize(.SellItemID);
	.@perpage = 20;

	if (.@total <= 0) {
		mes "[Sellable Items]";
		mes "No items configured.";
		return;
	}

	while (1) {

		mes "[Sellable Items]";
		mes "Page "+(.@page+1)+" / "+
			((.@total+.@perpage-1)/.@perpage);

		.@start = .@page * .@perpage;
		.@end = .@start + .@perpage;

		if (.@end > .@total)
			.@end = .@total;

		for (.@i = .@start; .@i < .@end; .@i++) {

			mes "- "+getitemname(.SellItemID[.@i])+
				" -> "+.SellItemPoint[.@i]+" BP";
		}

		next;

		.@menu$ = "";

		if (.@page > 0)
			.@menu$ += "Previous Page:";

		if (.@end < .@total)
			.@menu$ += "Next Page:";

		.@menu$ += "Back";

		.@s = select(.@menu$);

		.@idx = 0;

		// Previous Page
		if (.@page > 0) {

			if (.@s == 1) {
				.@page--;
				continue;
			}

			.@idx++;
		}

		// Next Page
		if (.@end < .@total) {

			if (.@s == .@idx+1) {
				.@page++;
				continue;
			}

			.@idx++;
		}

		// Back
		return;
	}

S_Info_MVP:
next;
	.@page = 0;
	.@total = getarraysize(.MVPMatID);
	.@perpage = 20;

	while (1) {

		mes "[MVP Materials]";
		mes "Page "+(.@page+1)+" / "+((.@total+.@perpage-1)/.@perpage);
		.@start = .@page * .@perpage;
		.@end = .@start + .@perpage;

		if (.@end > .@total)
			.@end = .@total;

		for (.@i = .@start; .@i < .@end; .@i++) {

			mes "- "+getitemname(.MVPMatID[.@i])+
			" -> "+.MVPMatPoint[.@i]+" BP";
		}

		next;

		.@menu$ = "";

		if (.@page > 0)
			.@menu$ += "Previous Page:";

		if (.@end < .@total)
			.@menu$ += "Next Page:";

		.@menu$ += "Back";

		.@s = select(.@menu$);

		// handle selection
		.@idx = 0;

		if (.@page > 0) {
			if (.@s == 1) {
				.@page--;
				continue;
			}
			.@idx++;
		}

		if (.@end < .@total) {
			if (.@s == .@idx+1) {
				.@page++;
				continue;
			}
			.@idx++;
		}

		return;
	}


	//------------------------------------------------
	// REFINE INFO
	//------------------------------------------------

S_Info_Refine:

	mes "[Refine Reward Info]";
	mes "Allowed:";
	mes "- Weapon Level 4";
	mes "- Armor";
	mes "- Refine +7 or higher";
	next;

	mes "Formula:";
	mes "Refine level x "+
		.RefinePoint+
		" BP";

	return;


	//------------------------------------------------
	// REWARD LIST
	//------------------------------------------------

S_List:

	for (.@i = 1; .@i <= 50; .@i++) {

		mes "Lv "+.@i+
			" -> "+
			getitemname(.RewardItem[.@i])+
			" x"+
			.RewardAmount[.@i];

		if (.@i % 15 == 0)
			next;
	}

	return;

	//------------------------------------------------
	// SQL LOG
	//------------------------------------------------

S_Log:

	query_sql(
		"INSERT INTO battlepass_log "+
		"(account_id,char_id,season,action,item_id,amount,points) "+
		"VALUES ("+
		getcharid(3)+","+
		getcharid(0)+","+
		.SeasonID+",'"+
