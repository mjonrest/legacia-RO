//--------------------------------------------------
// DAILY QUEST INIT FUNCTION
//--------------------------------------------------
function	script	F_DailyQuest_Init	{

	.@today$ = gettimestr("%Y-%m-%d", 21);

	// Check if today record exists
	query_sql(
		"SELECT COUNT(*) FROM daily_quest "+
		"WHERE account_id = "+getcharid(3)+" AND play_date = '"+.@today$+"'",
		.@exists
	);

	//--------------------------------------------------
	// NEW DAY ? CREATE NEW ROW
	//--------------------------------------------------
	if (.@exists == 0) {

		// Random buy target
		setarray .@items[0],
			1110,1113,1122,1119,1123,1126,1129,1116,
			2103,2226,2328,2307,2309,2312,2314,2627;

		set #D_BuyTarget, .@items[rand(getarraysize(.@items))];

		// Reset progress
		set #D_Point, 0;
		set #D_MobKill, 0;
		set #D_MVP, 0;
		set #D_Refine, 0;

		set #D_Q1, 0;
		set #D_Q2, 0;
		set #D_Q3, 0;
		set #D_Q4, 0;
		set #D_Q5, 0;

		.@uid$ = callfunc("F_DailyQuest_GetUID");

		query_sql(
			"INSERT INTO daily_quest "+
			"(account_id, unique_id, play_date, points, buy_target, daily_shop_item) "+
			"VALUES ("+getcharid(3)+", '"+escape_sql(.@uid$)+"', '"+.@today$+"', 0, "+#D_BuyTarget+", 0)"
		);
		dispbottom "[Daily Quest] New daily quests are available!";
		return;
	}

	//--------------------------------------------------
	// SAME DAY ? LOAD DATA
	//--------------------------------------------------
	query_sql(
		"SELECT points, buy_target FROM daily_quest "+
		"WHERE account_id = "+getcharid(3)+" AND play_date = '"+.@today$+"'",
		#D_Point, #D_BuyTarget
	);

	return;
}

//--------------------------------------------------
// GET UNIQUE ID FROM LOGIN DB
//--------------------------------------------------
function	script	F_DailyQuest_GetUID	{
	.@uid$ = "";
	query_sql(
		"SELECT last_unique_id FROM login "+
		"WHERE account_id = "+getcharid(3),
		.@uid$
	);
	// Treat empty / NULL / "0" as "no UID available"
	if (.@uid$ == "" || .@uid$ == "0")
		return "";
	return .@uid$;
}

//--------------------------------------------------
// ABUSE CHECK — return 1 if blocked
//--------------------------------------------------
function	script	F_DailyQuest_IsUIDBlocked	{
	.@today$ = gettimestr("%Y-%m-%d", 21);
	.@uid$   = callfunc("F_DailyQuest_GetUID");

	// No UID on file ? fall back to normal behaviour
	if (.@uid$ == "")
		return 0;

	query_sql(
		"SELECT COUNT(*) FROM daily_quest "+
		"WHERE play_date = '"+.@today$+"' "+
		"AND unique_id = '"+escape_sql(.@uid$)+"' "+
		"AND account_id <> "+getcharid(3),
		.@abuse
	);
	return (.@abuse > 0) ? 1 : 0;
}
