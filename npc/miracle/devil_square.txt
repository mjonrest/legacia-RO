prontera,138,176,5	script	Devil Square Guardian#1	406,{
	set .@gm_level, 99; // GM level required to start/stop the event

	mes "[Devil Square Guardian]";
	mes "This is the entrance to ^FF0000Devil Square^000000.";
	mes "If you fail or die, the door will close and you must wait until the event starts again to re-enter.";
	next;
	switch(select("Enter.:Information.:Reward List:Devil Square Shop:" + ((getgmlevel() >= .@gm_level) ? "GM Menu." : "") + ":Cancel.")) {
	case 1:
		if (.start != 1) { // registration period
			mes "[Devil Square Guardian]";
			mes "Devil Square is now closed.";
			mes "It will open at these times:";
			mes "00:00 , 02:00 , 04:00 , 06:00"; // display your times here
			mes "08:00 , 10:00 , 12:00 , 14:00";
			mes "16:00 , 18:00 , 20:00 , 22:00";
			close;
		}
		if (BaseLevel < 70 || Zeny < 100000) { // requirements
			mes "[Devil Square Guardian]";
			mes "You do not meet the requirements to enter Devil Square.";
			mes " ";
			mes "Requirements:";
			mes " - Base Level 70";
			mes " - Registration fee 100,000z";
			close;
		}
		Zeny -= 100000;
		announce strcharinfo(0)+" has entered Devil Square.", bc_npc;
		percentheal 100,100;
		warp "ordeal_1-1",183,182;
		close;
	case 2:
		mes "[Devil Square Guardian]";
		mes "Inside, you will encounter 9 waves of monsters. The monsters will grow stronger with each wave.";
		next;
		mes "[Devil Square Guardian]";
		mes "If you defeat all the enemies, you will be given the opportunity to open 10 Treasure Chests with items, equipment, and rare cards inside!";
		close;
	case 3:
	mes "[Reward Info]";
	mes "You will get random Devil Square Point";
	mes "when leaving ^FF0000Devil Square^000000.";
	close;
	case 4:
	callshop "dsshop",1;
	end;
	case 5:
		mes "[Devil Square Guardian]";
		switch (.start) { // event states
			case 0: mes "Devil Square has not yet started."; break;
			case 1: mes "Devil Square is currently accepting participants."; break;
			case 2: mes "Devil Square Event is currently running on Round "+ .round +" with "+ .mob +" monsters remaining."; break;
		}
		next;
		switch(select("Start event.:Stop event.:Cancel.")) {
		case 1:
			mes "[Devil Square Guardian]";
			if (.start)
				mes "Devil Square has already started.";
			else {
				mes "Starting Devil Square...";
				donpcevent strnpcinfo(0)+"::OnStartEvent";
			}
			close;
		case 2:
			mes "[Devil Square Guardian]";
			if (!.start)
				mes "Devil Square has not yet started.";
			else {
				mes "Stopping Devil Square...";
				donpcevent strnpcinfo(0)+"::OnStopEvent";
			}
			close;
		case 3:
			break;
		}
		break;
	case 6:
		break;
	}
	mes "[Devil Square Guardian]";
	mes "Farewell.";
	close;

OnStartEvent:
OnClock0000:
OnClock0200:
OnClock0400:
OnClock0600:
OnClock0800:
OnClock1000:
OnClock1200:
OnClock1400:
OnClock1600:
OnClock1800:
OnClock2000:
OnClock2200:
	if (.start) { // prevents duplicate execution
	killmonsterall "ordeal_1-1";
	.start = .round = .mob = 0;
	}
	.start = 1;
	disablenpc "ord11-12"; //from npc/warps/pvp.txt
	disablenpc "Devil Square Guardian#2";
	mapannounce "ordeal_1-1","Devil Square is now closed.", bc_map;
	getmapxy .@map$, .@x, .@y, BL_NPC;
	mapwarp "ordeal_1-1", .@map$, .@x, .@y;
	killmonsterall "ordeal_1-1";
	announce "Devil Square is OPEN. The event will begin in 5 minutes.", bc_all;
	sleep 60000;
	announce "Devil Square will begin in 4 minutes.", bc_all;
	sleep 60000;
	announce "Devil Square will begin in 3 minutes.", bc_all;
	sleep 60000;
	announce "Devil Square will begin in 2 minutes.", bc_all;
	sleep 60000;
	announce "Devil Square will begin in 1 minute.", bc_all;
	sleep 60000;
	announce "Devil Square has STARTED!", bc_all;
	if ( !getmapusers("ordeal_1-1") )
		goto OnStopEvent;
	.start = 2;
	donpcevent strnpcinfo(0)+"::OnStartRounds";

//==================================================
// DEVIL SQUARE ROUND SYSTEM (1–9 + Treasure)
//==================================================

OnStartRounds:
	if (.round > 0) end; // prevent duplicate round start
	.round = 1;
	goto OnSpawnRound;


//==================================================
// Spawn per round
//==================================================
OnSpawnRound:

	if (.round > 9)
		goto OnTreasurePhase;

	mapannounce "ordeal_1-1", "Devil Square : Round "+ .round +" has begun!", bc_map;

	.mob = 50; // 49 normal + 1 MVP

	.@map$ = "ordeal_1-1";

	//--------------------------------
	// Spawn 49 RANDOM NORMAL MONSTERS
	//--------------------------------
	for (.@i = 0; .@i < 49; .@i++) {

		.@rand_index = rand(getarraysize(.NormalMob));
		.@mobid = .NormalMob[.@rand_index];

		areamonster .@map$,183,182,246,244,
			"--ja--",
			.@mobid,
			1,
			strnpcinfo(0)+"::OnMobDead";
	}

	//--------------------------------
	// Spawn 1 RANDOM MVP
	//--------------------------------
	.@rand_index = rand(getarraysize(.MVPMob));
	.@mvp_id = .MVPMob[.@rand_index];

	areamonster .@map$,183,182,246,244,
		"--ja--",
		.@mvp_id,
		1,
		strnpcinfo(0)+"::OnMobDead";

	end;

//==================================================
// MONSTER DEATH HANDLER
//==================================================
OnMobDead:
	.mob--;

	if (.mob == 20 || .mob == 5)
		mapannounce "ordeal_1-1", "Devil Square : "+ .mob +" monsters remaining in round "+ .round +".", bc_map;

	if (.mob <= 0) {
		.round++;
		goto OnSpawnRound;
	}
	end;


//==================================================
// TREASURE PHASE (ROUND 10)
//==================================================
OnTreasurePhase:

	mapannounce "ordeal_1-1", "Devil Square : FINAL reward phase has begun!", bc_map;

	.mob = 10;

	monster "ordeal_1-1",231,250,"Treasure Chest",1324,1,strnpcinfo(0)+"::OnTreasureDead";
	monster "ordeal_1-1",234,247,"Treasure Chest",1324,1,strnpcinfo(0)+"::OnTreasureDead";
	monster "ordeal_1-1",237,244,"Treasure Chest",1324,1,strnpcinfo(0)+"::OnTreasureDead";
	monster "ordeal_1-1",240,241,"Treasure Chest",1324,1,strnpcinfo(0)+"::OnTreasureDead";
	monster "ordeal_1-1",243,238,"Treasure Chest",1324,1,strnpcinfo(0)+"::OnTreasureDead";
	monster "ordeal_1-1",246,235,"Treasure Chest",1324,1,strnpcinfo(0)+"::OnTreasureDead";
	monster "ordeal_1-1",249,232,"Treasure Chest",1324,1,strnpcinfo(0)+"::OnTreasureDead";
	monster "ordeal_1-1",252,229,"Treasure Chest",1324,1,strnpcinfo(0)+"::OnTreasureDead";
	monster "ordeal_1-1",249,241,"Treasure Chest",1324,1,strnpcinfo(0)+"::OnTreasureDead";
	monster "ordeal_1-1",240,249,"Treasure Chest",1324,1,strnpcinfo(0)+"::OnTreasureDead";

	end;


//==================================================
// TREASURE DEAD
//==================================================
OnTreasureDead:
	.mob--;
	announce "Devil Square : "+ strcharinfo(0) +" opened a Treasure Chest!", bc_all;

	if (.mob <= 0)
		goto OnStopEvent;
	end;

OnStopEvent:
	killmonsterall "ordeal_1-1";
	enablenpc "ord11-12"; //from npc/warps/pvp.txt
	enablenpc "Devil Square Guardian#2";
	.start = .round = .mob = 0;
	end;

OnInit:
	disablenpc "Devil Square Guardian#2";
	//--------------------------------
	// NORMAL MONSTER POOL
	//--------------------------------
	setarray .NormalMob[0],
		1028,1189,1165,1016,1149,1164,1253,1193,1213,1257,
		1191,1192,1206,1208,1109,1131,1302,1259,1155,1156,
		1368,1378,1384,1385,1687,1692,1715,1718,1614,1615,
		1121,1145,1151,1169,1209,1098,1154,1065,1069,1108,
		1264,1148,1274,1279,1162,1697,1695,1693,1696,1694,
		1699,1652,1653,1654,1655,1656,1657,1682,1376,1680,
		1271,1313,1317,1391,1207,1163,1276,1622,1617,1219; // <-- YOU FILL THIS

	//--------------------------------
	// MVP MONSTER POOL
	//--------------------------------
	setarray .MVPMob[0],
		1150,1630,1115,1252,1086,1492,1251,1147,1059,1159,1583,1658; // <-- YOU FILL THIS

	end;
}

ordeal_1-1,246,245,7	script	Devil Square Guardian#2	406,{
	mes "[Exit]";
	mes "Farewell.";
	close2;
	warp "SavePoint",0,0;
	set .@rand,rand(1,7);
	#dspoint += .@rand;
	dispbottom "You get "+.@rand+" Devil Square Point";
	dispbottom "Your current Devil Square Point : "+#dspoint+" pt(s).";
//	set .@mora,rand(1,20);
//	getitem 6380,.@mora;
	// Start periodic empty-map check
	addtimer 3000, strnpcinfo(3) + "::OnCheckMap";
	end;

OnCheckMap:
	if (getmapusers("ordeal_1-1") > 0) {
		// Still players inside, check again later
		addtimer 3000, strnpcinfo(3) + "::OnCheckMap";
		end;
	}

	// Map is empty cleanup
	disablenpc strnpcinfo(3);
	end;
}
