	query_sql("SELECT `class` FROM `" + .sql_t$ + "`",.@classes);

		if(inarray(.@c_list,.@classes[.@i]) == -1 && inarray(.FirstClasses,.@classes[.@i]) != -1)
			.@c_list[getarraysize(.@c_list)] = .@classes[.@i];
	}

	// Get all the `char_id`, `max_round` from `other_dimension_instance` and `account_id` from `char` table
	query_sql("SELECT DISTINCT o.`char_id`, o.`max_round`, c.`account_id` FROM `" + .sql_t$ + "` o LEFT JOIN `char` c ON o.`char_id` = c.`char_id`", .@cid1, .@max1, .@acc_id);

	// Arrays to track the highest `max_round` per `account_id`
	set .@account_max_round[getarraysize(.@acc_id)], -1;

	// Arrays to simulate lists of `char_id`s
	setarray .@char_id_list, -1; // List of char_ids (flat array)
	set .@start_index[getarraysize(.@acc_id)], 0; // Starting index of the list for each account
	set .@char_count[getarraysize(.@acc_id)], 0; // Number of chars in the list for each account
	.@current_index = 0; // Tracks the position in the flat array

	// Iterate through the characters and keep track of the highest `max_round` per account
	for(.@i = 0; .@i < getarraysize(.@acc_id); .@i++) {
		.@aid = .@acc_id[.@i]; // Current account ID

		// If this character's `max_round` is higher than the stored value, update it
		if(.@max1[.@i] > .@account_max_round[.@aid]) {
			.@account_max_round[.@aid] = .@max1[.@i];
			.@start_index[.@aid] = .@current_index; // Set new start index for this account
			.@char_count[.@aid] = 1; // Reset count to 1
			.@char_id_list[.@current_index++] = .@cid1[.@i]; // Add the current char_id
		}
		// If this character's `max_round` is equal to the highest, add to the list
		else if(.@max1[.@i] == .@account_max_round[.@aid]) {
			.@char_id_list[.@current_index++] = .@cid1[.@i]; // Add the char_id to the flat list
			.@char_count[.@aid]++; // Increase the count for this account
		}
	}

// Send rewards to the character with the highest max_round per account
for(.@i = 0; .@i < getarraysize(.@acc_id); .@i++) {
	.@aid = .@acc_id[.@i]; // Current account ID

	// Skip if the account has already been processed
	if (.@account_max_round[.@aid] == -1) continue;

	if (.@account_max_round[.@aid] > 0) { // Ensure there's a valid max_round
		// Get the list of char_ids for this account
		.@list_start = .@start_index[.@aid]; // Starting index
		.@list_size = .@char_count[.@aid]; // Number of characters with the highest round

		// Randomly select one character if there are multiple with the same max_round
		.@char_id = .@char_id_list[.@list_start + rand(.@list_size)];
		.@round = .@account_max_round[.@aid];

		// Check if the character is one of the first players (optional)
		if(inarray(.@FirstPlayrs, .@char_id) != -1) {
			.@count = (.rewardFT * .@round);
		} else {
			.@count = (.rewardCN * .@round);
		}

		// Send mail
		mail .@char_id, .sender$, .title$, .body$, 0, .rewardID, .@count, .@null, .@null, .@null, .@null, .@null;

		// Mark the account as processed
		.@account_max_round[.@aid] = -1;
	}
}
	query_sql("TRUNCATE `" + .sql_t$ + "`"); //clear the sql.
end;

	.MaxRewardPerClass = -1;//how many player will resive the reward per class the default is -1 all the Participants will get the rewards.

	if a player cleard up to stage 6 , he will get at the end of the week the count*stage 1*6 6 from the item id 503 and 6 from 502.
	
	if the player was the first in his class and there was more than .MiniParticipants from that class in the rank
