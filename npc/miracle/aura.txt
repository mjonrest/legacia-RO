prontera,175,177,3	script	Princess Mary	640,{
	set .@aura1, get_aura_part(1);
	set .@aura2, get_aura_part(2);
	set .@aura3, get_aura_part(3);
	set .@name$, "[Princess Mary]";

	mes .@name$;
	mes "I can help you customize cosmetics.";
	mes "I can convert your nick color or Auras.";

	setarray .@auras_max[1],
		152,117,35;

	setarray .@AuraTicket[1],
		30058,	// Aura Part 1
		30059,	// Aura Part 2
		30060;	// Aura Part 3
	next;
	switch(select("Color Nickname:Aura Changer")){
	case 1:
	.@costitem2 = 30061;
	.@costname2$ = getitemname(.@costitem2);

	mes .@name$;
	mes "I can change your nickname color.";
	mes "Required Item:";
	mes "^FF00001 x " + .@costname2$ + "^000000";
	next;

	.@original = get_color_nick();
	.@current = .@original;

	while (1) {

		set_color_nick(.@current);

	//	message strcharinfo(0),
	//		"Nickname Color Group : " + .@current + " / 20";

		.@menu$ =
			"Next (^0055FF"+((.@current<20)?.@current+1:0)+"^000000):"+
			"Previous (^0055FF"+((.@current>0)?.@current-1:20)+"^000000):"+
			"Jump to...:"+
			"Revert (^0055FF"+.@original+"^000000):"+
			"Confirm";

		switch(prompt(.@menu$)) {

		case 1:
			.@current = (.@current < 20) ? .@current + 1 : 0;
			break;

		case 2:
			.@current = (.@current > 0) ? .@current - 1 : 20;
			break;

		case 3:
		//	message strcharinfo(0),"Choose a color between 0 and 20.";
			input .@current,0,20;
			break;

		case 4:
			.@current = .@original;
			set_color_nick(.@original);
			break;

		case 5:
			next;
			mes .@name$;
			mes "Proceed with this nickname color?";
			mes "";
			mes "Required:";
			mes "^FF00001 x " + .@costname2$ + "^000000";
			next;
			switch(select("Yes:Maybe Later")) {

			case 1:
				if (countitem(.@costitem2) < 1) {
					mes .@name$;
					mes "You don't have the required item.";
					set_color_nick(.@original);
					close;
				}

				delitem .@costitem2,1;

				mes .@name$;
				mes "Your nickname color has been changed!";
				close;

			case 2:
				set_color_nick(.@original);
				mes .@name$;
				mes "Maybe next time.";
				close;
			}
			break;

		case 255:
			set_color_nick(.@original);
			mes .@name$;
			mes "Cancelled.";
			close;
		}
	}
	case 2:
	next;
	set .@s, select("Aura 1 Part:Aura 2 Part:Aura 3 Part");

	.@costitem = .@AuraTicket[.@s];
	.@costname$ = getitemname(.@costitem);

	set .@revert, get_aura_part(.@s);
	set .@current, .@revert;

	while (1) {

		set_aura_part(.@s, .@current);

	//	message strcharinfo(0),
	//		"Aura " + .@s + " Part = " + .@current;

		set .@menu$,
			"Next (^0055FF"+((.@current<.@auras_max[.@s])?.@current+1:0)+"^000000):"+
			"Previous (^0055FF"+((.@current>0)?.@current-1:.@auras_max[.@s])+"^000000):"+
			"Jump to...:"+
			"Revert to Original (^0055FF"+.@revert+"^000000):"+
			"Confirm";

		switch(prompt(.@menu$)) {

		case 1:
			.@current = ((.@current < .@auras_max[.@s]) ? .@current + 1 : 0);
			break;

		case 2:
			.@current = ((.@current > 0) ? .@current - 1 : .@auras_max[.@s]);
			break;

		case 3:
		//	message strcharinfo(0),
		//		"Choose a type between 0 and "+.@auras_max[.@s]+".";
			input .@current,0,.@auras_max[.@s];
			break;

		case 4:
			.@current = .@revert;
			set_aura_part(.@s,.@revert);
			break;

		case 255:
			mes .@name$;
			mes "Cancelled.";
			set_aura(.@aura1,.@aura2,.@aura3);
			close;

		case 5:

			next;
			mes .@name$;
			mes "This Aura looks great!";
			mes "";
			mes "Required Item:";
			mes "^FF00001 x "+.@costname$+"^000000";
			mes "";
			mes "Proceed?";
			next;
			switch(select("Yes:Maybe Later")) {

			case 1:

				if (countitem(.@costitem) < 1) {
					mes .@name$;
					mes "You don't have the required item.";
					set_aura(.@aura1,.@aura2,.@aura3);
					close;
				}

				delitem .@costitem,1;

				mes .@name$;
				mes "Your Aura has been updated successfully!";
				close;

			case 2:
				mes .@name$;
				mes "Maybe next time.";
				set_aura(.@aura1,.@aura2,.@aura3);
				close;
			}
		}
	}
	}
OnInit:
setunittitle( getnpcid(0), "<Cosmetic Girl>" );
end;
}
