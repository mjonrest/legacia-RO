function	script	F_ChristmasBox	{

	// =========================
	// Reward Arrays
	// =========================
	setarray .@patent[0],
	590012,530006,540011,500017,510019,510020,640011,640012,550012,550013,
	560008,560009,500013,500014,700018,700019,570009,580009,700013,620004,
	590011,610009,610008,21058,600009,510022,650004,830002,800002,820001,
	540010,840001,810001,550015,550015,510071,550081;

	setarray .@faith[0],
	400178,400179,400180,400181,400189,400190,400191,400192,400198,400199,
	400200,400201,400216,400217,400218,400219,400226,400227,400228,400229,
	400230,400231,400233,400234,400235,400236,400237,400238,400239,400240,
	400241,400242,400243,400244;

	setarray .@gray[0],
	450177,450178,480091,480090,470087,470088,490106,490108,490109,490107;

	setarray .@costume[0],
	20222,20545,31132,19955,19976,31910,19697,480405,19981,20592,19682,19685,19702;

	// =========================
	// Random Category (0~3)
	// =========================
	.@group = rand(4);

	if (.@group == 0) {
		.@item = .@patent[rand(getarraysize(.@patent))];
		.@group_name$ = "Patent Weapon";
	}
	else if (.@group == 1) {
		.@item = .@faith[rand(getarraysize(.@faith))];
		.@group_name$ = "Faith Helm";
	}
	else if (.@group == 2) {
		.@item = .@gray[rand(getarraysize(.@gray))];
		.@group_name$ = "Gray Equip";
	}
	else {
		.@item = .@costume[rand(getarraysize(.@costume))];
		.@group_name$ = "Costumes";
	}

	// =========================
	// Give Item (Account Bound)
	// =========================
	getitem .@item, 1;

	// =========================
	// Announcement
	// =========================
	announce "[Christmas Event] " + strcharinfo(0) +
	         " obtained " + itemlink(.@item) +
	         " from the " + .@group_name$ + " Category!", bc_all;

	end;
}
