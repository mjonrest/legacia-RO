// TIME FORMAT
function	script	TimeRemaining	{
	.@remain = getarg(0) - gettimetick(2);
	if (.@remain < 0) return "Expired";
	.@d = .@remain / 86400;
	.@h = (.@remain % 86400) / 3600;
	.@m = (.@remain % 3600) / 60;
	return .@d+"d "+.@h+"h "+.@m+"m";
}
