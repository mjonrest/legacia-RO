// ============================================================
// NO-DUAL LOGIN CHECK ON WORLD BOSS MAP
// ============================================================
-	script	wb_no_dual	-1,{
OnInit:
	// 1 = use Gepard UID, 0 = use IP
	.Gepard = 1;
	end;

OnPCLoadMapEvent:
	// Only act if this map is the current world boss map AND boss is active
	if ( strcharinfo(3) != $world_boss_map$ ) end;
	if ( $world_boss_active != 1 ) end;

	if ( .Gepard ) {
		query_sql( "SELECT last_unique_id FROM login WHERE account_id = "+getcharid(3), .@uid$ );
		query_sql( "SELECT account_id FROM login WHERE last_unique_id = '"+escape_sql(.@uid$)+"'", .@accs );
	} else {
		query_sql( "SELECT last_ip FROM login WHERE account_id = "+getcharid(3), .@ip$ );
		query_sql( "SELECT last_ip FROM login WHERE last_ip = '"+escape_sql(.@ip$)+"'", .@accs );
	}

	.@count = 0;
	.@target = 0;

	for ( .@i = 0; .@i < getarraysize(.@accs); .@i++ ) {
		if ( attachrid( .@accs[.@i] ) ) {
			// Skip GMs (optional)
			if ( getgmlevel() > 0 ) {
				detachrid;
				continue;
			}
			if ( strcharinfo(3) == $world_boss_map$ ) {
				.@count++;
				.@target = getcharid(3);
			}
			detachrid;
		}
	}

	if ( .@count > 1 && .@target ) {
		if ( attachrid( .@target ) ) {
			dispbottom "You already have another character on the World Boss map. Dual logging is not allowed.";
			warp "prontera", 156, 191;
		}
		detachrid;
	}
	end;
}

// ============================================================
// HELPER FUNCTION: comma formatting
// ============================================================
function	script	F_Comma	{
	.@n$ = ""+getarg(0);
	.@len = getstrlen(.@n$);
	.@res$ = "";
	.@neg = 0;
	if ( charat(.@n$,0) == "-" ) {
		.@neg = 1;
		.@n$ = substr(.@n$,1,.@len-1);
		.@len = getstrlen(.@n$);
	}
	for ( .@i = .@len-1; .@i >= 0; .@i-- ) {
		.@res$ = charat(.@n$,.@i) + .@res$;
		.@count++;
		if ( .@count == 3 && .@i > 0 ) {
			.@res$ = "," + .@res$;
			.@count = 0;
		}
	}
	if ( .@neg ) .@res$ = "-" + .@res$;
	return .@res$;
}

// ============================================================
// SAFE RANDOM FUNCTION (prevents range 0 error)
// ============================================================
function	script	F_RandSafe	{
	.@max = getarg(0);
	if (.@max <= 0) return 0;
	return rand(.@max);
}

	end;

	// Only run if entering WB map AND boss is active
	if ( strcharinfo(3) != $world_boss_map$ ) end;
	if ( $world_boss_active != 1 ) end;
	if ( @wb_afk_loop ) end;

	@wb_afk_loop = 1;

	// Optional grace period (recommended)
	sleep2 5000;

	while ( strcharinfo(3) == $world_boss_map$ && $world_boss_active == 1 ) {

		sleep2 10000; // 10 sec check

		// Idle warning 1
		if ( checkidle() > 60 && checkidle() <= 70 )
			dispbottom "Don't AFK or you will be kicked from World Boss map.";

		// Warning 2
		else if ( checkidle() > 90 && checkidle() <= 100 )
			dispbottom "You have 30 seconds to move!";

		// Kick
		else if ( checkidle() > 120 ) {
			dispbottom "You were kicked for being AFK.";
			warp "SavePoint", 0, 0;
			break;
		}
	}

	@wb_afk_loop = 0;
	end;
}

OnMinute00:
OnMinute10:
OnMinute20:
OnMinute30:
OnMinute40:
OnMinute50:
	if ( $world_boss_active == 3) {
		announce "World Boss is clearing...",0,0xff0000;
		end;
	}

	if ( $world_boss_active == 2) {
		announce "World Boss will awake soon",0,0xff0000;
		end;
	}
	if ( $world_boss_active == 1) {
		announce "World Boss is currently active",0,0xff0000;
		end;
	}
	if ( $world_boss_active == 0) {
		query_sql "SELECT `kill_count` FROM `world_boss` WHERE `set` = 'wb' ", .@kill_count;
		announce "World Boss Souls : "+callfunc("F_Comma",.@kill_count)+" collected.",0,0xff0000;
		end;
	}
