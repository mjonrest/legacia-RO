-	script	DonateCommand	-1,{

	bindatcmd "donate", strnpcinfo(3)+"::OnDonate";
	end;

	case 1:
	callshop "Donate1",1;
	end;
	case 2:
	callshop "Donate_Cash",1;
	end;
	}
}

-	pointshop	Donate_Cash	-1,#CASHPOINTS,30018:4,30007:10
