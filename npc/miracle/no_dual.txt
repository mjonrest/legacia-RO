-	script	et_map	-1,{

OnInit:
	// 1 = using Gepard UID, 0 = using IP check
	.Gepard = 1;

	// Endless Tower base maps
	setarray .ETMaps$[0],
		"1@tower","2@tower","3@tower","4@tower","5@tower","6@tower";

	end;

//====================================================
// MAIN CHECK (runs when player loads a map)
//====================================================
OnPCLoadMapEvent:

	// Get base instance map name (removes #0001)
	.@map$ = instance_mapname(strcharinfo(3));

	// Check if player is inside Endless Tower
	.@isET = 0;
	for (.@i = 0; .@i < getarraysize(.ETMaps$); .@i++) {
		if (.@map$ == .ETMaps$[.@i]) {
			.@isET = 1;
			break;
		}
	}

	// Not Endless Tower ? ignore
	if (!.@isET)
		end;

	// Also ensure it's an instance
	if (instance_id() == 0)
		end;

	//--------------------------------------------------
	// GEPARD MODE (UNIQUE ID)
	//--------------------------------------------------
	if (.Gepard) {

		query_sql("SELECT last_unique_id FROM login WHERE account_id = "+getcharid(3), .@uid$);
		query_sql("SELECT account_id FROM login WHERE last_unique_id = '"+.@uid$+"'", .@acc);

		.@count = 0;
		.@target = 0;

		for (.@i = 0; .@i < getarraysize(.@acc); .@i++) {
			if (attachrid(.@acc[.@i])) {

				// GM or allowed account (edit if needed)
				if (.@acc[.@i] == 2000002) {
					detachrid;
					continue;
				}

				if (instance_mapname(strcharinfo(3)) == .@map$) {
					.@count++;
					.@target = getcharid(3);
				}
				detachrid;
			}
		}

	//--------------------------------------------------
	// IP MODE
	//--------------------------------------------------
	} else {

		query_sql("SELECT last_ip FROM login WHERE account_id = "+getcharid(3), .@ip$);
		query_sql("SELECT account_id FROM login WHERE last_ip = '"+.@ip$+"'", .@acc);

		.@count = 0;
		.@target = 0;

		for (.@i = 0; .@i < getarraysize(.@acc); .@i++) {
			if (attachrid(.@acc[.@i])) {

				// Allowed accounts (edit list if needed)
				if (
					.@acc[.@i] == 2000000 ||
					.@acc[.@i] == 2000156 ||
					.@acc[.@i] == 2000249 ||
					.@acc[.@i] == 2000031 ||
					.@acc[.@i] == 2000050 ||
					.@acc[.@i] == 2000021 ||
					.@acc[.@i] == 2000073 ||
					.@acc[.@i] == 2000029
				) {
					detachrid;
					continue;
				}

				if (instance_mapname(strcharinfo(3)) == .@map$) {
					.@count++;
					.@target = getcharid(3);
				}
				detachrid;
			}
		}
	}

	//--------------------------------------------------
	// FINAL CHECK
	//--------------------------------------------------
	if (.@count > 1 && .@target) {
		if (attachrid(.@target)) {
			dispbottom "Sorry, Double Login is not allowed inside Endless Tower.";
			warp "prontera",156,183;
		}
		detachrid;
	}

	end;
}
