	end;

	$world_boss_active = 2;

	announce "[World Boss]: Dark energy gathers across Midgard...",0,0xff0000;
	sleep 10000;
	announce "[World Boss]: The seal will break in 5 minutes!",0,0xff0000;
	sleep 240000;

			.@max_hp = .@wb[3];
			
			// Calculate regen amount
			.@regen_amount = .@max_hp * .@regen_percent / 100;
			
			// Apply regen (capped at max HP)
			.@new_hp = min(.@current_hp + .@regen_amount, .@max_hp);
			setunitdata .GID, UMOB_HP, .@new_hp;
			
			// Optional: Announce to map when regen happens
			// mapannounce "ordeal_2-1", "[World Boss] regenerates " + .@regen_percent + "% HP (" + .@alive_guardians + " guardians alive)!", bc_map;
			unittalk .GID,"HP Regen " + .@regen_percent + "%",bc_area;
	//	}
	}
	end;

OnSpawnGuardians:
	// Spawn 7 guardians
	monster "ordeal_2-1", 245, 245, "[WB] Guardian", 2408, 1, "World Boss::OnGuardDead";
	monster "ordeal_2-1", 273, 149, "[WB] Guardian", 2408, 1, "World Boss::OnGuardDead";
	monster "ordeal_2-1", 245, 52, "[WB] Guardian", 2408, 1, "World Boss::OnGuardDead";
	monster "ordeal_2-1", 149, 26, "[WB] Guardian", 2408, 1, "World Boss::OnGuardDead";
	monster "ordeal_2-1", 54, 54, "[WB] Guardian", 2408, 1, "World Boss::OnGuardDead";
	monster "ordeal_2-1", 26, 150, "[WB] Guardian", 2408, 1, "World Boss::OnGuardDead";
	monster "ordeal_2-1", 53, 245, "[WB] Guardian", 2408, 1, "World Boss::OnGuardDead";
	
	// Reset guardian respawn timer
	$guardian_respawn_time = 0;
	
	mapannounce "ordeal_2-1", "[World Boss]: 7 Guardians have appeared! Kill them to lower the boss's regeneration!", 0;
	end;

OnGuardDead:
	// When a guardian dies, check if all are dead
	.@alive_guardians = mobcount("ordeal_2-1", "World Boss::OnGuardDead");
	
	if (.@alive_guardians <= 0) {
		// All guardians dead, start respawn timer (5 minutes)
		$guardian_respawn_time = gettimetick(2) + (5 * 60);
		mapannounce "ordeal_2-1", "[World Boss]: All guardians have been defeated! Boss regeneration lowered! Guardians will respawn in 5 minutes.", 0;
	}
	end;

OnCheckGuardianRespawn:
	// Check every 30 seconds if guardians need to respawn
	while ($world_boss_active == 1) {
		sleep 30000; // Check every 30 seconds
		
		// Check if all guardians are dead and 5 minutes have passed
		.@alive_guardians = mobcount("ordeal_2-1", "World Boss::OnGuardDead");
		
		if (.@alive_guardians <= 0 && $guardian_respawn_time > 0 && gettimetick(2) >= $guardian_respawn_time) {
			// Respawn guardians
			donpcevent strnpcinfo(0)+"::OnSpawnGuardians";
			$guardian_respawn_time = 0;
		}
	}
	end;

OnTimer3600000:

	setnpctimer 0;
	announce "[World Boss]: Seems like Nobody can defeat World Boss, The Gate suddenly close.",0;
	mapwarp "ordeal_2-1","prontera",156,191,0;
	killmonsterall "ordeal_2-1",0;
	$world_boss_active = 0;
	$guardian_respawn_time = 0;
	end;

	
OnMobDead:
	stopnpctimer;
	setnpctimer 0;

	.@amount = rand(15,25);

	// Rental bonus check

		.@amount += 5;
		dispbottom "[Charm of Brightness] active bonus applied! (+5 coins)",0xFFFF00;
	}

	getitem 30007, .@amount;
	callfunc "F_LogWorldBossCoin", .@amount, "boss_kill";

	announce "[World Boss] Demonic Chaos Dragon has been slained by "+strcharinfo(0)+"!!",0,0xff0000;
	mapannounce "ordeal_2-1","[WB] Demonic Chaos Dragon : This cant be Possible. I've been beaten by mortals",0,0xff0000;
	sleep 5000;
	mapannounce "ordeal_2-1","You may leave thru World Boss Spirit and claim your rewards.",0,0xff0000;

	$world_boss_active = 3;

	// Kill all guardians when boss dies
	killmonster "ordeal_2-1", "World Boss::OnGuardDead";
	$guardian_respawn_time = 0;

	enablenpc "World Boss Spirit";
	end;

OnInit:
	disablenpc "World Boss Spirit";
	$world_boss_active = 0; // 0=inactive,1=active,2=preparing,3=cleanup
	$guardian_respawn_time = 0;
	end;

}

function	script	F_Comma	{
	.@n$ = ""+getarg(0); // convert to string
	.@len = getstrlen(.@n$);
	.@res$ = "";

	// handle negative numbers
	.@neg = 0;
	if (charat(.@n$,0) == "-") {
		.@neg = 1;
		.@n$ = substr(.@n$,1,.@len-1);
		.@len = getstrlen(.@n$);
	}

	for (.@i = .@len - 1; .@i >= 0; .@i--) {
		.@res$ = charat(.@n$,.@i) + .@res$;
		.@count++;
		if (.@count == 3 && .@i > 0) {
			.@res$ = "," + .@res$;
			.@count = 0;
		}
	}

	if (.@neg) .@res$ = "-" + .@res$;

	return .@res$;
}

function	script	F_LogWorldBossCoin	{
	.@amount = getarg(0);
	.@source$ = getarg(1);

	.@aid = getcharid(3);
	.@cid = getcharid(0);
	.@name$ = strcharinfo(0);
	.@map$ = strcharinfo(3);
	.@time$ = gettimestr("%Y-%m-%d %H:%M:%S", 21);

	query_sql(
		"INSERT INTO log_worldboss_coin (account_id, char_id, char_name, amount, map, log_time, source) VALUES ("+
		.@aid+","+.@cid+",'"+escape_sql(.@name$)+"',"+.@amount+",'"+.@map$+"','"+.@time$+"','"+.@source$+"')"
	);

	return;
}
