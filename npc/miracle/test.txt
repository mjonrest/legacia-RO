prontera,191,154,5	script	Grave Enchanter	4_F_KAFRA1,{
	// Constants
	.@frag_id = 23016;
	setarray .@head_ids[0],18971,18972,18973,18974,18975,18976,18977,18978,18979,18980,18981,18982,18983,18984;
	setarray .@enchant_lv1[0],29061,29071,29081,29091,29101,29111;
	setarray .@enchant_lv2[0],29062,29072,29082,29092,29102,29112;
	setarray .@enchant_lv3[0],29063,29073,29083,29093,29103,29113;
	setarray .@enchant_lv4[0],29064,29074,29084,29094,29104,29114;
	setarray .@enchant_lv5[0],29065,29075,29085,29095,29105,29115;

	// Costs by level
	setarray .@frags[0],0,100,200,250,350;
	setarray .@zeny[0],0,3500000,5000000,8000000,14000000;

	mes "[Grave Enchanter]";
	mes "I will upgrade your equipped headgear's enchant (slot 1) to the next level.";
	mes "Only eligible top headgears with enchant in slot 1 are accepted.";
	next;

	// Check equipped top headgear
	.@eqi = EQI_HEAD_TOP;
	.@headgear = getequipid(.@eqi);
	if (.@headgear == -1) {
		mes "You must wear a headgear.";
		close;
	}

	// Check if it's in the allowed list
	.@valid = 0;
	for (.@i = 0; .@i < getarraysize(.@head_ids); .@i++) {
		if (.@headgear == .@head_ids[.@i]) {
			.@valid = 1;
			break;
		}
	}
	if (!.@valid) {
		mes "Your headgear is not eligible.";
		close;
	}

	// Get enchant from card slot 1
	.@enchant = getequipcardid(.@eqi, 1);
	if (.@enchant < 29061 || .@enchant > 29115) {
		mes "No valid enchantment found in slot 1.";
		close;
	}

	// Determine current level and index
	.@level = 0;
	.@index = -1;
	for (.@i = 0; .@i < 6; .@i++) {
		if (.@enchant == .@enchant_lv1[.@i]) { .@level = 1; .@index = .@i; break; }
		if (.@enchant == .@enchant_lv2[.@i]) { .@level = 2; .@index = .@i; break; }
		if (.@enchant == .@enchant_lv3[.@i]) { .@level = 3; .@index = .@i; break; }
		if (.@enchant == .@enchant_lv4[.@i]) { .@level = 4; .@index = .@i; break; }
		if (.@enchant == .@enchant_lv5[.@i]) { .@level = 5; .@index = .@i; break; }
	}
	if (.@level == 0 || .@index == -1 || .@level >= 5) {
		mes "Your enchant cannot be upgraded further.";
		close;
	}

	.@next_level = .@level + 1;
	.@new_enchant = getd(".@enchant_lv"+.@next_level+"["+.@index+"]");

	// Check cost
	if (Zeny < .@zeny[.@level] || countitem(.@frag_id) < .@frags[.@level]) {
		mes "You need "+.@frags[.@level]+" Cursed Fragments and "+.@zeny[.@level]+" Zeny.";
		close;
	}

	// Confirm
	mes "Upgrade "+getitemname(.@enchant)+" to level "+.@next_level+" for:";
	mes .@frags[.@level]+" Cursed Fragments and "+.@zeny[.@level]+" Zeny.";
	if(select("^0055FFProceed^000000:Cancel") == 2) close;

	// Deduct resources
	Zeny -= .@zeny[.@next_level];
	delitem .@frag_id, .@frags[.@next_level];

	// Backup item properties
	.@refine = getequiprefinerycnt(.@eqi);
	.@card0 = getequipcardid(.@eqi,0);
	.@card2 = getequipcardid(.@eqi,2);
	.@card3 = getequipcardid(.@eqi,3);

	// Remove old item
	delequip .@eqi;

	// Give upgraded item
	getitem2 .@headgear, 1, 1, .@refine, 0, .@card0, .@new_enchant, .@card2, .@card3;

	mes "Enchant upgraded successfully!";
	mes "Slot 1 is now: "+getitemname(.@new_enchant);
	close;
}
