poring_w02,81,88,3	script	MVP Bet Girl	612,{
	mes "[MVP Bet Girl]";
	if (!getvariableofnpc(.start, "MVP Battle")) {
		mes "The event is not active right now.";
		close;
	}
	else if (getvariableofnpc(.start, "MVP Battle") != 3) {
		mes "Sorry, betting is not available right now.";
		close;
	}
	else if (Zeny < getvariableofnpc(.bet_cost, "MVP Battle")) {
		mes "You don't have enough Zeny to bet.";
		close;
	}
	else if (@mvp_bet) {
		mes "You already placed your bet.";
		close;
	}
	else {
		mes "Your current Event point : "+#EVENTVARIABLE+" pts.";
		if ( #EVENTVARIABLE < getvariableofnpc(.bet_cost, "MVP Battle") ){
		mes "Sorry but you dont have enough Event point to bet";
		close;
		}
		mes "Which MVP will you bet on this match???";
		next;
		.@s = select("MVP A:MVP B");
		mes "[MVP Bet Girl]";
		mes "Are you sure you want to bet " + F_InsertComma(getvariableofnpc(.bet_cost, "MVP Battle")) + "z on " + (.@s == 1 ? "MVP A" : "MVP B") + "?";
		next;
		if (select("Yes:No") == 2) close;
		if (getvariableofnpc(.start, "MVP Battle") != 3) {
			mes "[MVP Bet Girl]";
			mes "Oops! Betting has already closed.";
			close;
		}
		#EVENTVARIABLE -= getvariableofnpc(.bet_cost, "MVP Battle");
		@mvp_bet = .@s;
		mes "[MVP Bet Girl]";
		mes "Good luck! You bet " + F_InsertComma(getvariableofnpc(.bet_cost, "MVP Battle")) + "z on " + (.@s == 1 ? "MVP A" : "MVP B") + ".";
		mapannounce "poring_w02","[ Bet Girl ] : "+strcharinfo(0)+" just place a bet", bc_blue|bc_map;
		close;
	}
	end;
}
