//=============================================================
// ONLSAUGHT MASTERY
//=============================================================

function	script	Onslaught_M	{

	// STR
	bonus bStr,1 + getarg(0,0) / 32;

	// ATK Rate
	bonus2 bAddClass,Class_All,
		1 + getarg(0,0) / 7
		+ ((1 + getarg(0,0) / 7) * readparam(bStr) / 100);

	// Lv 1
	if (getarg(0,0) >= 1)
		bonus bCritical,1 + getarg(0,0) / 20;

	// Lv 2
	if (getarg(0,0) >= 2)
	//	bonus bBaseAtk,1 + getarg(0,0) / 4;
		bonus bBaseAtk,getarg(0,0) * 5;

	// Lv 4
	if (getarg(0,0) >= 4)
		bonus bAtk2,1 + getarg(0,0) / 4;

	// Lv 8
	if (getarg(0,0) >= 8)
		bonus bPerfectHitAddRate,1 + getarg(0,0) / 21;

	// Lv 16
	if (getarg(0,0) >= 16)
		bonus bShortAtkRate,getarg(0,0) / 4 + 1;

	// Lv 32
	if (getarg(0,0) >= 32)
		bonus bCritAtkRate,getarg(0,0) / 4 + 1;

	// Lv 64
	if (getarg(0,0) >= 64)
		bonus bAspdRate,getarg(0,0) / 22 + 1;

	// Lv 128
	if (getarg(0,0) >= 128)
		bonus bSplashAddRange,1;

	return;
}

//=============================================================
// WEAPON TYPE -> MASTERY
//=============================================================
//
// Returns:
//
//   "Onslaught"
//   "Swiftness"
//   "Ranged"
//   "Mystical"
//   ""
//
//=============================================================

function	script	F_WM_type	{

	//=====================================================
	// ITEM ID EXCLUSIONS
	//
	// getarg(1) = item ID (optional).
	// These daggers use Mystical instead of Swiftness.
	//=====================================================

	switch(getarg(1,0)) {
		case 1231:	// Bazerald
		case 13010:	// Asura
		case 13011:	// Asura
		case 13023:	// Ashura
			return "Mystical";
	}

	switch(getarg(0,0)) {

		//=====================================================
		// ONSLAUGHT
		//=====================================================

		case 2:		// 1H Sword
		case 3:		// 2H Sword
		case 4:		// 1H Spear
		case 5:		// 2H Spear
		case 6:		// 1H Axe
		case 7:		// 2H Axe
		case 8:		// 1H Mace
		case 9:		// 2H Mace
		case 12:	// Knuckle
			return "Onslaught";


		//=====================================================
		// SWIFTNESS
		//=====================================================

		case 1:		// Dagger
		case 15:	// Book
		case 16:	// Katar
		case 22:	// Fuuma Shuriken
			return "Swiftness";


		//=====================================================
		// RANGED
		//=====================================================

		case 11:	// Bow
		case 13:	// Instrument
		case 14:	// Whip
		case 17:	// Revolver
		case 18:	// Rifle
		case 19:	// Gatling Gun
		case 20:	// Shotgun
		case 21:	// Grenade Launcher
			return "Ranged";


		//=====================================================
		// MYSTICAL
		//=====================================================

		case 10:	// Rod
		case 23:	// 2H Staff
		case 24:	// Scythe
			return "Mystical";
	}

	// Shield / armor / invalid / unsupported item
	return "";
}


//=============================================================
// WEAPON MASTERY - APPLY CURRENT EQUIPMENT
//=============================================================
//
// ONLY RIGHT-HAND WEAPON COUNTS.
//
// Left-hand weapon is completely ignored.
//
//=============================================================

function	script	WM_Apply	{

	.@right_id = getequipid(EQI_HAND_R);

	if (.@right_id <= 0)
		return;


	//=========================================================
	// GET RIGHT-HAND WEAPON TYPE
	//=========================================================

	.@right_type = getiteminfo(.@right_id,11);

	.@right_mastery$ = callfunc(
		"F_WM_type",
		.@right_type,
		.@right_id
	);


	// Unsupported weapon type
	if (.@right_mastery$ == "")
		return;


	//=========================================================
	// GET MASTERY LEVEL
	//=========================================================

	.@right_level = getd(
		.@right_mastery$ + "Master"
	);


	//=========================================================
	// APPLY MASTERY
	//=========================================================

	if (.@right_mastery$ == "Onslaught")

		callfunc("Onslaught_M",.@right_level);

	else if (.@right_mastery$ == "Swiftness")

		callfunc("Swiftness_M",.@right_level);

	else if (.@right_mastery$ == "Ranged")

		callfunc("Ranged_M",.@right_level);

	else if (.@right_mastery$ == "Mystical")

		callfunc("Mystical_M",.@right_level);


	return;
}


//=============================================================
// WEAPON MASTERY - REFRESH BONUS
//=============================================================

function	script	WM_Refresh	{

	// Apply current right-hand weapon mastery.
	bonus_script "{ callfunc \"WM_Apply\"; }",31536000,8;

	return;
}


//=============================================================
// WEAPON MASTERY - UNEQUIP
//=============================================================

function	script	WM_Unequip	{

	// Remove previous weapon mastery bonus.
	bonus_script_clear();

	return;
}

//=============================================================
// @wlvl / @wlvlup
//=============================================================

OnWeaponLevel:

	.@M$ = .@atcmd_parameters$[0];
	.@X = atoi(.@atcmd_parameters$[1]);


	//=========================================================
	// VALIDATION
	//=========================================================

	if ((.@M$ != "Onslaught"
		&& .@M$ != "Swiftness"
		&& .@M$ != "Ranged"
		&& .@M$ != "Mystical"
		&& .@M$ != "All")
		|| !.@X) {

		dispbottom "@wlvl <type> <lvl>";
		dispbottom "Onslaught, Swiftness, Ranged, Mystical, All";

		end;
	}


	//=========================================================
	// SET LEVEL
	//=========================================================

	if (.MaxMastery >= .@X) {

		//=====================================================
		// ALL MASTERIES
		//=====================================================

		if (.@M$ == "All") {

			setarray .@Mastery$[0],
				"Onslaught",
				"Swiftness",
				"Ranged",
				"Mystical";

			.@i = 0;

			while (.@i < 4) {

				setd
					.@Mastery$[.@i] + "Master",
					.@X;

				setd
					.@Mastery$[.@i] + "Exp",
					0;

				dispbottom
					.@Mastery$[.@i]
					+ " Mastery has advanced to Lv "
					+ .@X
					+ "!";

				.@i++;
			}
		}

		//=====================================================
		// SINGLE MASTERY
		//=====================================================

		else {

			setd
				.@M$ + "Master",
				.@X;

			setd
				.@M$ + "Exp",
				0;

			dispbottom
				.@M$
				+ " Mastery has advanced to Lv "
				+ .@X
				+ "!";
		}


		// Refresh active weapon mastery.
		callfunc("WM_Refresh");
	}

	else {

		dispbottom
			"Sorry. Your "
			+ .@M$
			+ " Mastery has reached its limit!";
	}

	end;

OnNPCKillEvent:

	if (!.Weapon)
		end;


	.@exp = getmonsterinfo(
		killedrid,
		MOB_JOBEXP
	);


	//=========================================================
	// RIGHT-HAND WEAPON ONLY
	//=========================================================

	.@right_id = getequipid(EQI_HAND_R);

	if (.@right_id <= 0)
		end;


	.@type = getiteminfo(
		.@right_id,
		11
	);

	.@M$ = callfunc(
		"F_WM_type",
		.@type,
		.@right_id
	);


	if (.@M$ == "")
		end;


	//=========================================================
	// GET WEAPON LEVEL
	//=========================================================

	.@wLv = getiteminfo(
		.@right_id,
		13
	);


	//=========================================================
	// GIVE MASTERY EXP TO KILLER
	//=========================================================

	set .@bonus,(.@exp) * (.@wLv * .wLvBonus);
	if ( getstatus(SC_JEXPBOOST,0)){
	set .@bonus,(.@exp + (.@exp/2)) * (.@wLv * .wLvBonus);
	}

	callsub L_AddExp, .@M$, .@bonus;


	//=========================================================
	// PARTY SHARE
	//
	// Party members on the same map as the killer get
	// .PartyShare % of the killer's mastery EXP, added to the
	// mastery of their own right-hand weapon.
	//
	// Only one character per device (Gepard unique ID, or IP
	// when Gepard is not available) gets a share, and never a
	// character on the killer's own device.
	//=========================================================

	.@party_id = getcharid(1);
	.@share = .@bonus * .PartyShare / 100;

	if (.@party_id <= 0 || .@share <= 0)
		end;

	.@killer_cid = getcharid(0);
	.@map$ = strcharinfo(3);

	.@used_dev$[0] = callsub(L_DeviceId);

	getpartymember .@party_id, 1, .@member_cid;
	getpartymember .@party_id, 2, .@member_aid;
	.@count = $@partymembercount;

	for (.@i = 0; .@i < .@count; .@i++) {

		if (.@member_cid[.@i] == .@killer_cid)
			continue;

		if (!isloggedin(.@member_aid[.@i], .@member_cid[.@i]))
			continue;

		if (!attachrid(.@member_aid[.@i]))
			continue;

		if (strcharinfo(3) != .@map$)
			continue;

		.@dev$ = callsub(L_DeviceId);

		if (inarray(.@used_dev$, .@dev$) != -1)
			continue;

		.@used_dev$[getarraysize(.@used_dev$)] = .@dev$;

		.@m_id = getequipid(EQI_HAND_R);

		if (.@m_id <= 0)
			continue;

		.@m_M$ = callfunc(
			"F_WM_type",
			getiteminfo(.@m_id,11),
			.@m_id
		);

		if (.@m_M$ == "")
			continue;

		callsub L_AddExp, .@m_M$, .@share;
	}

	end;


//=============================================================
// ADD MASTERY EXP TO ATTACHED PLAYER
//
// getarg(0) = mastery name
// getarg(1) = EXP to add
//=============================================================

L_AddExp:

	.@sub_M$ = getarg(0);

	.@master = getd(
		.@sub_M$ + "Master"
	);

	if (.MaxMastery > .@master) {

		setd
			.@sub_M$ + "Exp",
		getd(.@sub_M$ + "Exp") + getarg(1);
	//	dispbottom "exp :"+getarg(1)+" pts.";


		//=====================================================
		// LEVEL UP
		//=====================================================

		if (getd(.@sub_M$ + "Exp") > .NeededExp[.@master]) {

			setd
				.@sub_M$ + "Master",
				.@master + 1;

			announce
				.@sub_M$

// ============================================================
// HELPER FUNCTION: comma formatting
// ============================================================
function	script	F_Comma	{
	.@n$ = ""+getarg(0);
	.@len = getstrlen(.@n$);
	.@res$ = "";
	.@neg = 0;
	if ( charat(.@n$,0) == "-" ) {
		.@neg = 1;
		.@n$ = substr(.@n$,1,.@len-1);
		.@len = getstrlen(.@n$);
	}
	for ( .@i = .@len-1; .@i >= 0; .@i-- ) {
		.@res$ = charat(.@n$,.@i) + .@res$;
		.@count++;
		if ( .@count == 3 && .@i > 0 ) {
			.@res$ = "," + .@res$;
			.@count = 0;
		}
	}
	if ( .@neg ) .@res$ = "-" + .@res$;
	return .@res$;
}
