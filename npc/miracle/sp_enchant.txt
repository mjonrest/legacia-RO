    // --------------------------------------------------------

	copyarray .@OptVal[0], @inventorylist_opt_value[.@index], 5;
	copyarray .@OptPar[0], @inventorylist_opt_param[.@index], 5;

    .@card4   = @inventorylist_card4[.@idx];

        close;

    deletearray .@pool[0], getarraysize(.@pool);
	next;
