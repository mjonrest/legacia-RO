		close;

	.@option$[.@option_count] = "Cancel";

		close;

		mes "You need at least 10 " + .@stone_name$ + "!";

	// Build evolution database
	callfunc "InitEvoData";
	end;

function	script	CleanDigitsOnly	{
	.@raw$ = getarg(0);
	.@clean$ = "";
	.@len = getstrlen(.@raw$);
	for (.@i = 0; .@i < .@len; .@i++) {
		.@ch = charat(.@raw$, .@i);
		// Keep only digits
		if (.@ch >= "0" && .@ch <= "9") {
			.@clean$ = .@clean$ + .@ch;
		}
	}
	return .@clean$;
}

function	script	SafeSubstr	{
	.@s$ = getarg(0);
	.@start = getarg(1);
	.@length = getarg(2);

	// Validate input
	if (.@length <= 0) return "";
	if (getstrlen(.@s$) <= .@start) return "";

	// Slice safely
	set .@i, 0;
	set .@result$, "";
	while (.@i < .@length && (.@start + .@i) < getstrlen(.@s$)) {
		set .@result$, .@result$ + charat(.@s$, .@start + .@i);
		set .@i, .@i + 1;
	}
	return .@result$;
}
