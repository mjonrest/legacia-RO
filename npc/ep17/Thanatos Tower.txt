	mes "Are you ready?";

	switch(.@i) {
	case 0:
		if(BaseLevel < 180) {
			mes "[Party]";
			mes "Sorry, you must be at least Lv.180 to enter.";
			mes "You must be at least Lv.180 to enter.";
			close;
		}
		
		if (instance_create(.@md_name$) < 0) {
			mes "[Party]";
			mes "[Error]";
			mes "Party name: "+.@p_name$;
			mes "Party leader: "+strcharinfo(0);

			close;

		mapannounce strnpcinfo(4),"Party "+getpartyname(.@party_id)+" has started the instance "+.@md_name$+"!",bc_map,"0x33ea91";
		mes "[Info]";
		mes ""+.@md_name$+" has been created.";
		mes "Please ^0066CCenter "+.@md_name$+"^000000.";
		
		.@iid = instance_id(IM_PARTY);
		set getinstancevar('party_id, .@iid), getcharid(1);
		//Entry time
		//set Instance_Annal,gettimetick(2);
		close;
	case 1:
		if(BaseLevel < 180) {
			mes "[Info]";
			mes "Sorry, you must be at least Lv.180 to enter."; 
			mes "You must be at least Lv.180 to enter."; 
			close;
		}
	if ( getinstancevar('Win,instance_id(IM_PARTY)) ) {
	mes "The instance has already started, you cannot enter.";
	close;
	}
		switch(checkquest(200075,PLAYTIME)){
			case 0:
			case 1:
				mes "you are on cooldown fella";
				end;
		}

		switch(instance_enter(.@md_name$)) {
		case IE_OTHER:
			mes "An error occurred.";
			close;
		case IE_NOINSTANCE:
			mes .@md_name$+" instance does not exist.";
			mes "Please try again later.";
			close;
		case IE_NOMEMBER:
			mes "You are not a member of the "+.@md_name$+" instance.";
			close;
		case IE_OK:
			setquest 200075;
			end;
		}
	}
	close;
	
OnInit:
	waitingroom "Thanatos Tower",0;
	end;
}
