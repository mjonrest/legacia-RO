//------------------------------ Tomb of Confession Manager
dali02,78,116,4	script	Princess Tiara	640,{

OnStart:
	set .@party_id,getcharid(1);
	set .@p_name$,getpartyname(.@party_id);

	if (!.@party_id) {
		mes "[Tomb of Confession Manager]";
		mes "Where are your party members?";
		close;
	}
	
	.@md_name$ = "Tomb of Remorse";
	
	if (getcharid(0) == getpartyleader(.@party_id,2))
		set @menu$, "Create "+.@md_name$+":Enter "+.@md_name$;
	else
		set @menu$, ":Enter "+.@md_name$;
		
	mes "[Tomb of Confession Manager]";
	mes "What would you like to do?";
	next;
	.@i = select(@menu$)-1;
	switch(.@i) {
	case 0:
		if(BaseLevel < 220) {
			mes "[Repentant Princess Tiara]";
			mes "It's dangerous inside..."; 
			mes "Come back after reaching Lv.220..."; 
			close;
		}
		
		if (instance_create(.@md_name$) < 0) {
			mes "[Repentant Princess Tiara]";
			mes "Party Name: "+.@p_name$;
			mes "Party Leader: "+strcharinfo(0);
			mes "^0000ff"+.@md_name$+" ^000000- Creation failed!";
			close;
		}

		mapannounce strnpcinfo(4),"Party "+getpartyname(.@party_id)+" is preparing to challenge the 【"+.@md_name$+"】 instance!",bc_map,"0x33ea91";	
		mes "[Repentant Princess Tiara]";
		mes ""+.@md_name$+" has been created~";
		mes "Please select ^0066CCEnter "+.@md_name$+"^000000.";
		
		.@iid = instance_id(IM_PARTY);
		set getinstancevar('party_id, .@iid), getcharid(1);
		// Record entry time
		set Instance_Annal,gettimetick(2);
		close;
	case 1:
		if(BaseLevel < 220) {
			mes "[Repentant Princess Tiara]";
			mes "It's dangerous inside..."; 
			mes "Come back after reaching Lv.220..."; 
			close;
		}
		if (getinstancevar('Win,instance_id(IM_PARTY))) {
			mes "The instance has already started. You cannot enter again.";
			close;
		}
		switch(checkquest(16464,PLAYTIME)){
			case 0:
			case 1:
				mes "you are on cooldown fella";
				end;
		}
		switch(instance_enter(.@md_name$)) {
		case IE_OTHER:
			mes "Unknown error.";
			close;
		case IE_NOINSTANCE:
			mes .@md_name$+" instance does not exist.";
			mes "The leader has not created the Memory Maze.";
			close;
		case IE_NOMEMBER:
			mes "Only party members who applied for the "+.@md_name$+" instance can enter.";
			close;
		case IE_OK:
			setquest 16464;
			// Instance Quests
			if(checkquest(16465,HUNTING)<0){setquest 16465;}
			if(checkquest(16467,HUNTING)<0){setquest 16467;}
			if(checkquest(16469,HUNTING)<0){setquest 16469;}
			if(checkquest(16471,HUNTING)<0){setquest 16471;}

			if(checkquest(16465,HUNTING)<0){ if(checkquest(16466,PLAYTIME)==2){ erasequest 16466; setquest 16465;} }
			if(checkquest(16467,HUNTING)<0){ if(checkquest(16468,PLAYTIME)==2){ erasequest 16468; setquest 16467;} }
			if(checkquest(16469,HUNTING)<0){ if(checkquest(16470,PLAYTIME)==2){ erasequest 16470; setquest 16469;} }
			if(checkquest(16471,HUNTING)<0){ if(checkquest(16472,PLAYTIME)==2){ erasequest 16472; setquest 16471;} }
			end;
		}
	}
	close;
	
OnInit:
	waitingroom "Tomb of Confession",0;
	end;
}

		close;

		mes "[Repentant Marquis]";
		mes "Mission to Collect Fragments of the Past.";

		if(.@quest02 == 2) erasequest 16472;
		if(.@quest02 == 0 || .@quest02 == 1){
