// Thanatos Tower Enchanter
// Location: Near Thanatos Tower entrance or designated area

disable_items;
	mes "[Thanatos Enchanter]";
	mes "I can enchant your ^006400Crown of Good and Evil^000000 headgears.";
	mes "Choose an option:";
	next;
	set .@headgear, getequipid(EQI_HEAD_TOP);
	
	// Validate headgear
	if (!.@headgear) {
		mes "Please equip a ^FF0000Crown of Good and Evil^000000 headgear first.";
		close;
	}
	
	// Allowed headgear IDs from Sheet2
	setarray $@allowed_headgears[0],
		400377,400381,400385,400386,400387,400388,400390,400392,400393,
		400394,400389,400395,400378,400374,400379,400375,400383,400376,
		400380,400391,400382,400384;
	
	set .@valid, 0;
	for (set .@i, 0; .@i < getarraysize($@allowed_headgears); set .@i, .@i + 1) {
		if (.@headgear == $@allowed_headgears[.@i]) {
			set .@valid, 1;
			break;
		}
	}
	
	if (!.@valid) {
		mes "Invalid headgear. Only ^006400Crown of Good and Evil^000000 series can be enchanted.";
		close;
	}

	set .@card0, getequipcardid(EQI_HEAD_TOP, 0);
	set .@card1, getequipcardid(EQI_HEAD_TOP, 1);
	set .@card2, getequipcardid(EQI_HEAD_TOP, 2); // Slot 3 enchant
	set .@card3, getequipcardid(EQI_HEAD_TOP, 3); // Slot 4 enchant
	set .@refine, getequiprefinerycnt(EQI_HEAD_TOP);
	set .@grade, getenchantgrade(EQI_HEAD_TOP);
	.@id = getequiprandomoption(EQI_HEAD_TOP, 0, ROA_ID);
	.@val = getequiprandomoption(EQI_HEAD_TOP, 0, ROA_VALUE);
	.@param = getequiprandomoption(EQI_HEAD_TOP, 0, ROA_PARAM);

	
	switch(select("Slot 3 Enchant:Slot 4 Enchant:Reset Enchantments")) {
	case 1: // Slot 3 Enchant
		if (.@card2 != 0) {
			mes "Slot 3 is already enchanted!";
			close;
		}
		
		mes "Slot 3 enchant costs:";
		mes "^009900250x Piece of Sin^000000";
		mes "^0099003x Fragment of Good Will^000000";
		mes "^0099001x Enchant Ticket^000000";
		mes "^FF00003,000,000 Zeny^000000";
		next;
		
		if (select("Enchant Slot 3:Cancel") == 2) close;
		
		if (countitem(1000257) < 250 || countitem(1000263) < 3 || 
			countitem(1000874) < 1 || Zeny < 3000000) {
			mes "Insufficient materials!";
			close;
		}
		
		// Show enchant options
		switch(select("Good Spell:Evil Spell")) {
		case 1:
			set .@enchant, 312014; // Good Spell
			break;
		case 2:
			set .@enchant, 312021; // Evil Spell
			break;
		}
		
		// Remove items
		delitem 1000257, 250; // Piece of Sin
		delitem 1000263, 3;   // Fragment of Good Will
		delitem 1000874, 1;   // Enchant Ticket
		set Zeny, Zeny - 3000000;
		
		// Recreate headgear with enchant
		delitem .@headgear, 1;
//		getitem2 .@headgear, 1, 1, .@refine, 0, .@card0, .@card1, .@enchant, .@card3;
		getitem4 .@headgear, 1, 1, .@refine, 0, .@card0, .@card1, .@enchant, .@card3, .@grade, .@id, .@val, .@param;
		equip EQI_HEAD_TOP;
		mes "Slot 3 enchant successful!";
		close;
		
	case 2: // Slot 4 Enchant
		// Enchant level definitions
		setarray .@enchant_name$[0], 
			"Tenacity", "Adamantine", "Magic Essence", 
			"Master Archer", "Acute", "Affection";
			
		setarray .@enchant_lv1[0],
			29706, 29101, 29071, 29091, 29081, 29111;
			
		setarray .@enchant_lv2[0],
			29707, 29102, 29072, 29092, 29082, 29112;
			
		setarray .@enchant_lv3[0],
			29708, 29103, 29073, 29093, 29083, 29113;
		
		// Check existing enchant
		set .@current_enchant, .@card3;
		set .@current_level, 0;
		set .@enchant_idx, -1;
		
		if (.@current_enchant != 0) {
			for (set .@i, 0; .@i < 6; set .@i, .@i + 1) {
				if (.@current_enchant == .@enchant_lv1[.@i]) {
					set .@current_level, 1;
					set .@enchant_idx, .@i;
					break;
				}
				if (.@current_enchant == .@enchant_lv2[.@i]) {
					set .@current_level, 2;
					set .@enchant_idx, .@i;
					break;
				}
				if (.@current_enchant == .@enchant_lv3[.@i]) {
					set .@current_level, 3;
					set .@enchant_idx, .@i;
					break;
				}
			}
		}
		
		// Upgrade existing enchant
		if (.@current_level > 0 && .@current_level < 3) {
			mes "Current Slot 4 enchant:";
			mes .@enchant_name$[.@enchant_idx] + " Lv" + .@current_level;
			mes "Upgrade to Lv" + (.@current_level + 1) + "?";
			next;
			
			// Show upgrade cost
			if (.@current_level == 1) {
				mes "^0000FFLv1 -> Lv2 Cost:^000000";
				mes "50x Piece of Sin";
				mes "1x Fragment of Good Will";
				mes "1x Enchant Ticket";
			} else if (.@current_level == 2) {
				mes "^0000FFLv2 -> Lv3 Cost:^000000";
				mes "1x Powerful Dimensional Essence";
				mes "70x Piece of Sin";
				mes "70x Fragment of Good Will";
				mes "1x Enchant Ticket";
			}
			next;
			
			if (select("Upgrade:Cancel") == 2) close;
			
			// Verify materials
			if (.@current_level == 1) {
				if (countitem(1000257) < 50 || 
					countitem(1000263) < 1 || 
					countitem(1000874) < 1) {
					mes "Insufficient materials!";
					close;
				}
				
				// Remove materials
				delitem 1000257, 50; // Piece of Sin
				delitem 1000263, 1;  // Fragment of Good Will
				delitem 1000874, 1;  // Enchant Ticket
				
				// Set next level
				set .@new_enchant, .@enchant_lv2[.@enchant_idx];
			} 
			else if (.@current_level == 2) {
				if (countitem(7925) < 1 || 
					countitem(1000257) < 70 || 
					countitem(1000263) < 70 || 
					countitem(1000874) < 1) {
					mes "Insufficient materials!";
					close;
				}
				
				// Remove materials
				delitem 7925, 1;      // Powerful Dimensional Essence
				delitem 1000257, 70; // Piece of Sin
				delitem 1000263, 70; // Fragment of Good Will
				delitem 1000874, 1;  // Enchant Ticket
				
				// Set next level
				set .@new_enchant, .@enchant_lv3[.@enchant_idx];
			}
			
			// Recreate headgear
			delitem .@headgear, 1;
//			getitem2 .@headgear, 1, 1, .@refine, 0, .@card0, .@card1, .@card2, .@new_enchant;
			getitem4 .@headgear, 1, 1, .@refine, 0, .@card0, .@card1, .@card2, .@new_enchant, .@grade, .@id, .@val, .@param;
			equip EQI_HEAD_TOP;
			mes "Upgrade successful!";
			close;
		}
		
		// New enchant
		if (.@current_level == 3) {
			mes "Slot 4 is already at maximum level!";
			close;
		}
		
		// New enchant options
		mes "Choose Slot 4 enchant type:";
		next;
		switch(select("Random Enchant:Specific Enchant")) {
		case 1: // Random
			mes "^0000FFRandom Enchant Cost:^000000";
			mes "1x Powerful Dimensional Essence";
			mes "25x Piece of Sin";
			mes "25x Fragment of Good Will";
			mes "300,000 Zeny";
			next;
			
			if (select("Enchant:Cancel") == 2) close;
			
			if (countitem(7925) < 1 || 
				countitem(1000257) < 25 || 
				countitem(1000263) < 25 || 
				Zeny < 300000) {
				mes "Insufficient materials!";
				close;
			}
			
			// Remove materials
			delitem 7925, 1;      // Powerful Dimensional Essence
			delitem 1000257, 25; // Piece of Sin
			delitem 1000263, 25; // Fragment of Good Will
			set Zeny, Zeny - 300000;
			
			// Random selection
			set .@enchant_idx, rand(6);
			set .@new_enchant, .@enchant_lv1[.@enchant_idx];
			break;
			
		case 2: // Specific
			mes "^0000FFSpecific Enchant Cost:^000000";
			mes "2x Powerful Dimensional Essence";
			mes "100x Piece of Sin";
			mes "100x Fragment of Good Will";
			mes "1x Enchant Ticket";
			mes "3,000,000 Zeny";
			next;
			
			if (select("Continue:Cancel") == 2) close;
			
			// Show enchant options
			set .@menu$, "";
			for (set .@i, 0; .@i < 6; set .@i, .@i + 1) {
				set .@menu$, .@menu$ + .@enchant_name$[.@i] + ":";
			}
			set .@choice, select(.@menu$) - 1;
			
			// Verify materials
			if (countitem(7925) < 2 || 
				countitem(1000257) < 100 || 
				countitem(1000263) < 100 || 
				countitem(1000874) < 1 || 
				Zeny < 3000000) {
				mes "Insufficient materials!";
				close;
			}
			
			// Remove materials
			delitem 7925, 2;       // Powerful Dimensional Essence
			delitem 1000257, 100; // Piece of Sin
			delitem 1000263, 100; // Fragment of Good Will
			delitem 1000874, 1;   // Enchant Ticket
			set Zeny, Zeny - 3000000;
			
			set .@enchant_idx, .@choice;
			set .@new_enchant, .@enchant_lv1[.@choice];
			break;
		}
		
		// Apply new enchant
		delitem .@headgear, 1;
//		getitem2 .@headgear, 1, 1, .@refine, 0, .@card0, .@card1, .@card2, .@new_enchant;
		getitem4 .@headgear, 1, 1, .@refine, 0, .@card0, .@card1, .@card2, .@new_enchant, .@grade, .@id, .@val, .@param;
		equip EQI_HEAD_TOP;
		mes "Enchant successful!";
		mes "Applied: " + .@enchant_name$[.@enchant_idx] + " Lv1";
		close;
		
	case 3: // Reset
		if (.@card2 == 0 && .@card3 == 0) {
			mes "No enchantments to reset!";
			close;
		}
		
		mes "^FF0000Reset will remove ALL enchantments!^000000";
		mes "Cost:";
		mes "50x Piece of Sin";
		mes "100x Fragment of Good Will";
		mes "1,000,000 Zeny";
		next;
		
		if (select("Reset Enchantments:Cancel") == 2) close;
		
		if (countitem(1000257) < 50 || 
			countitem(1000263) < 100 || 
			Zeny < 1000000) {
			mes "Insufficient materials!";
			close;
		}
		
		// Remove materials
		delitem 1000257, 50; // Piece of Sin
		delitem 1000263, 100; // Fragment of Good Will
		set Zeny, Zeny - 1000000;
		
		// Recreate clean headgear
		delitem .@headgear, 1;
//		getitem2 .@headgear, 1, 1, .@refine, 0, .@card0, .@card1, 0, 0;
		getitem4 .@headgear, 1, 1, .@refine, 0, .@card0, .@card1, 0, 0, .@grade, .@id, .@val, .@param;
		equip EQI_HEAD_TOP;
		mes "Enchantments reset successfully!";
		close;
	}
}

	// Allowed weapon IDs from Sheet1
	setarray $@allowed_weapons[0],

		820021,650034,510092,640037,610045,840021,630027,570037,500066,

	set .@card1, getequipcardid(EQI_HAND_R, 1);
	set .@card2, getequipcardid(EQI_HAND_R, 2); // Slot 3 enchant
	set .@card3, getequipcardid(EQI_HAND_R, 3); // Slot 4 enchant
	set .@refine, getequiprefinerycnt(EQI_HAND_R);
	set .@grade, getenchantgrade(EQI_HAND_R);
	.@id = getequiprandomoption(EQI_HAND_R, 0, ROA_ID);
	.@val = getequiprandomoption(EQI_HAND_R, 0, ROA_VALUE);
	.@param = getequiprandomoption(EQI_HAND_R, 0, ROA_PARAM);

	
	switch(select("Slot 3 Enchant:Slot 4 Enchant:Reset Enchantments")) {
	case 1: // Slot 3 Enchant
		if (.@card2 != 0) {
			// Upgrade existing enchant
			goto L_UpgradeSlot3;
		} else {
			// New enchant
			goto L_NewSlot3;
		}
		end;
		
	case 2: // Slot 4 Enchant
		if (.@card3 != 0) {
			// Upgrade existing enchant
			goto L_UpgradeSlot4;
		} else {
			// New enchant
			goto L_NewSlot4;
		}
		end;
		
	case 3: // Reset

			close;
		}

		mes "Cost:";
		mes "20x Fragment of Good Will";
		mes "1,000,000 Zeny";
		next;
		
		if (select("Reset Enchantments:Cancel") == 2) close;
		
		if (countitem(1000263) < 20 || Zeny < 1000000) {
			mes "Insufficient materials!";
			close;
		}
		
		// Remove materials
		delitem 1000263, 20; // Fragment of Good Will
		set Zeny, Zeny - 1000000;
		
		// Recreate clean weapon
		delitem .@weapon, 1;
//		getitem2 .@weapon, 1, 1, .@refine, 0, .@card0, .@card1, 0, 0;
		getitem4 .@weapon, 1, 1, .@refine, 0, .@card0, .@card1, 0, 0, .@grade, .@id, .@val, .@param;
		mes "Enchantments reset successfully!";
		close;
	}
	
	end;
	
//--------------- Slot 3 Functions ---------------//
L_NewSlot3:
	mes "Slot 3 enchant costs:";
	mes "^0099001x Powerful Dimensional Essence^000000";
	mes "^009900250x Fragment of Good Will^000000";
	mes "^FF00003,000,000 Zeny^000000";
	next;
	
	if (select("Enchant Slot 3:Cancel") == 2) close;
	
	if (countitem(7925) < 1 || countitem(1000263) < 250 || Zeny < 3000000) {
		mes "Insufficient materials!";
		close;
	}
	
	// Show enchant options
	mes "Choose Slot 3 enchant type:";
	next;
	switch(select("Expert Archer:Expert Magician:Expert Fighter")) {
	case 1:
		set .@enchant, 4832; // Expert Archer Lv1
		set .@enchant_name$, "Expert Archer";
		break;
	case 2:
		set .@enchant, 311389; // Expert Magician Lv1
		set .@enchant_name$, "Expert Magician";
		break;
	case 3:
		set .@enchant, 311384; // Expert Fighter Lv1
		set .@enchant_name$, "Expert Fighter";
		break;
	}
	
	// Remove materials
	delitem 7925, 1;       // Powerful Dimensional Essence
	delitem 1000263, 250; // Fragment of Good Will
	set Zeny, Zeny - 3000000;
	
	// Recreate weapon with enchant
	delitem .@weapon, 1;
//	getitem2 .@weapon, 1, 1, .@refine, 0, .@card0, .@card1, .@enchant, .@card3;
	getitem4 .@weapon, 1, 1, .@refine, 0, .@card0, .@card1, .@enchant, .@card3, .@grade, .@id, .@val, .@param;
	mes "Slot 3 enchant successful!";
	mes "Applied: " + .@enchant_name$ + " Lv1";
	close;
	
L_UpgradeSlot3:
	// Enchant definitions
	setarray .@archer,  4832,4833,4834,4835,4836;
	setarray .@magician,311389,311390,311391,311392,311393;
	setarray .@fighter, 311384,311385,311386,311387,311388;
	
	// Find current enchant type and level
	set .@current_enchant, .@card2;
	set .@current_level, 0;
	set .@enchant_type, 0;
	
	for (set .@i, 0; .@i < 5; set .@i, .@i + 1) {
		if (.@current_enchant == .@archer[.@i]) {
			set .@current_level, .@i + 1;
			set .@enchant_type, 1;
			set .@enchant_name$, "Expert Archer";
			break;
		}
		if (.@current_enchant == .@magician[.@i]) {
			set .@current_level, .@i + 1;
			set .@enchant_type, 2;
			set .@enchant_name$, "Expert Magician";
			break;
		}
		if (.@current_enchant == .@fighter[.@i]) {
			set .@current_level, .@i + 1;
			set .@enchant_type, 3;
			set .@enchant_name$, "Expert Fighter";
			break;
		}
	}
	
	if (!.@current_level) {
		mes "Invalid enchantment detected!";
		close;
	}
	
	if (.@current_level >= 5) {
		mes .@enchant_name$ + " is already at maximum level!";
		close;
	}
	
	// Calculate next level
	set .@next_level, .@current_level + 1;
	
	// Determine upgrade cost based on level
	switch(.@next_level) {
	case 2:
		mes "Upgrade to Lv2 Cost:";
		mes "50x Fragment of Good Will";
		break;
	case 3:
		mes "Upgrade to Lv3 Cost:";
		mes "55x Fragment of Good Will";
		break;
	case 4:
		mes "Upgrade to Lv4 Cost:";
		mes "1x Powerful Dimensional Essence";
		mes "60x Fragment of Good Will";
		break;
	case 5:
		mes "Upgrade to Lv5 Cost:";
		mes "1x Powerful Dimensional Essence";
		mes "65x Fragment of Good Will";
		// Extra cost for Magician Lv5
		if (.@enchant_type == 2) {
			mes "1x Enchant Ticket";
		}
		break;
	}
	next;
	
	if (select("Upgrade:Cancel") == 2) close;
	
	// Verify materials
	switch(.@next_level) {
	case 2:
		if (countitem(1000263) < 50) {
			mes "Insufficient materials!";
			close;
		}
		break;
	case 3:
		if (countitem(1000263) < 55) {
			mes "Insufficient materials!";
			close;
		}
		break;
	case 4:
		if (countitem(7925) < 1 || countitem(1000263) < 60) {
			mes "Insufficient materials!";
			close;
		}
		break;
	case 5:
		if (countitem(7925) < 1 || countitem(1000263) < 65) {
			mes "Insufficient materials!";
			close;
		}
		// Extra cost for Magician
		if (.@enchant_type == 2 && countitem(1000874) < 1) {
			mes "You need 1x Enchant Ticket!";
			close;
		}
		break;
	}
	
	// Remove materials
	switch(.@next_level) {
	case 2:
		delitem 1000263, 50;
		break;
	case 3:
		delitem 1000263, 55;
		break;
	case 4:
		delitem 7925, 1;
		delitem 1000263, 60;
		break;
	case 5:
		delitem 7925, 1;
		delitem 1000263, 65;
		if (.@enchant_type == 2) delitem 1000874, 1; // Enchant Ticket for Magician
		break;
	}
	
	// Get next enchant ID
	switch(.@enchant_type) {
	case 1: set .@new_enchant, .@archer[.@next_level-1]; break;
	case 2: set .@new_enchant, .@magician[.@next_level-1]; break;
	case 3: set .@new_enchant, .@fighter[.@next_level-1]; break;
	}
	
	// Recreate weapon
	delitem .@weapon, 1;
//	getitem2 .@weapon, 1, 1, .@refine, 0, .@card0, .@card1, .@new_enchant, .@card3;
	getitem4 .@weapon, 1, 1, .@refine, 0, .@card0, .@card1, .@new_enchant, .@card3, .@grade, .@id, .@val, .@param;
	mes "Upgrade successful!";
	mes "Now: " + .@enchant_name$ + " Lv" + .@next_level;
	close;
	
//--------------- Slot 4 Functions ---------------//
L_NewSlot4:
	mes "Slot 4 enchant costs:";
	mes "^009900150x Fragment of Good Will^000000";
	mes "^FF00001,500,000 Zeny^000000";
	next;
	
	if (select("Enchant Slot 4:Cancel") == 2) close;
	
	if (countitem(1000263) < 150 || Zeny < 1500000) {
		mes "Insufficient materials!";
		close;
	}
	
	// Show enchant options
	mes "Choose Slot 4 enchant type:";
	next;
	switch(select("Caster:Hit Plus:DelayAfterAttack:Sharp")) {
	case 1:
		set .@enchant, 311400; // Caster Lv1
		set .@enchant_name$, "Caster";
		break;
	case 2:
		set .@enchant, 311395; // Hit Plus Lv1
		set .@enchant_name$, "Hit Plus";
		break;
	case 3:
		set .@enchant, 4869; // DelayAfterAttack Lv1
		set .@enchant_name$, "DelayAfterAttack";
		break;
	case 4:
		set .@enchant, 4818; // Sharp Lv1
		set .@enchant_name$, "Sharp";
		break;
	}
	
	// Remove materials
	delitem 1000263, 150; // Fragment of Good Will
	set Zeny, Zeny - 1500000;
	
	// Recreate weapon with enchant
	delitem .@weapon, 1;
//	getitem2 .@weapon, 1, 1, .@refine, 0, .@card0, .@card1, .@card2, .@enchant;
	getitem4 .@weapon, 1, 1, .@refine, 0, .@card0, .@card1, .@card2, .@enchant, .@grade, .@id, .@val, .@param;
	mes "Slot 4 enchant successful!";
	mes "Applied: " + .@enchant_name$ + " Lv1";
	close;
	
L_UpgradeSlot4:
	// Enchant definitions
	setarray .@caster,    311400,311401,311402,311403,311404;
	setarray .@hitplus,   311395,311396,311397,311398,311399;
	setarray .@delay,     4869,4872,4873,4881,311394;
	setarray .@sharp,     4818,4817,4816,4843,4844;
