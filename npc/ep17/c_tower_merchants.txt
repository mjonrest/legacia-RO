        mes "[Azzam The Lucky]";
        mes "You need more Zeny!";
        mes "Required: ^FF0000" + .@cost_zeny + " Zeny^000000";
        close;
    }
    if (countitem(.@cost_item) < .@cost_amount) {
        mes "[Azzam The Lucky]";
        mes "You need " + .@cost_amount + "x " + getitemname(.@cost_item) + "!";
        close;
    }
    
    // Consume resources
    Zeny -= .@cost_zeny;
    delitem .@cost_item, .@cost_amount;
    
    // Update the appropriate card
    if (.@slot_type == 2) set .@card1, .@next_id;
    else if (.@slot_type == 3) set .@card2, .@next_id;
    else if (.@slot_type == 4) set .@card3, .@next_id;
    
    // Replace armor
//    unequip EQI_ARMOR;
//    delitem .@armor_id, 1;
    delequip EQI_ARMOR;
//    getitem2 .@armor_id, 1, 1, .@refine, 0, .@card0, .@card1, .@card2, .@card3;
    getitem4 .@armor_id, 1, 1, .@refine, 0, .@card0, .@card1, .@card2, .@card3, .@grade, .@id, .@val, .@param;

    
    mes "[Azzam The Lucky]";
    mes "Cosmic upgrade complete!";
    mes "New enchant: ^55FF55" + getitemname(.@next_id) + "^000000";
    close;
}

    set .@armor_id, getarg(0);

        .@armor_id == 450171 || // Concentration
        .@armor_id == 450172 || // Creative

// Get upgrade level (1=base, 2=mid, 3=high)
function	script	F_GetUpgradeLevel	{
    set .@id, getarg(0);
    set .@next1, callfunc("F_GetNextUpgrade", .@id);
    if (.@next1 == 0) return 3; // Max level
    
    set .@next2, callfunc("F_GetNextUpgrade", .@next1);

    return 1; // Base level
}

			313039, 313040, 1000443,5, 1000445,5, 1000447,4, 1001599,4, 500000,80,
			313040, 313041, 1000443,10, 1000445,10, 1000447,8, 1001599,8, 1000000,65,
			313041, 313042, 1000443,20, 1000445,20, 1000447,16, 1001599,16, 2000000,45,
			313042, 313043, 1000443,35, 1000445,35, 1000447,28, 1001599,28, 3500000,25;
		
		// Find upgrade info
		set .@found, 0;
		for(set .@i, 0; .@i < getarraysize(.@upgrade_path); .@i += 12) {
			if(.@upgrade_path[.@i] == .@card_id) {
				set .@next_id, .@upgrade_path[.@i + 1];
				// Manually extract material costs
				set .@mats[0], .@upgrade_path[.@i+2];
				set .@mats[1], .@upgrade_path[.@i+3];
				set .@mats[2], .@upgrade_path[.@i+4];
				set .@mats[3], .@upgrade_path[.@i+5];
				set .@mats[4], .@upgrade_path[.@i+6];
				set .@mats[5], .@upgrade_path[.@i+7];
				set .@mats[6], .@upgrade_path[.@i+8];
				set .@mats[7], .@upgrade_path[.@i+9];
				set .@zeny, .@upgrade_path[.@i+10];
				set .@rate, .@upgrade_path[.@i+11];
				set .@found, 1;
				break;
			}
		}
		
		if(!.@found) {
			mes "^FF0000No upgrade available for this enchant!^000000";
			close;
		}
		
		mes "Upgrade: ^55FFFF" + getitemname(.@card_id) + "^000000 ? ^55FFFF" + getitemname(.@next_id) + "^000000";
		mes "Success Chance: " + .@rate + "%";
		mes "Cost:";
		for(set .@k, 0; .@k < 8; .@k += 2)
			mes .@mats[.@k + 1] + "x ^0055FF" + getitemname(.@mats[.@k]) + "^000000";
		mes .@zeny + " Zeny";
		next;
		
		if(select("Attempt Upgrade", "Cancel") == 2) close;
		
		// Check materials
		for(set .@k, 0; .@k < 8; .@k += 2) {
			if(countitem(.@mats[.@k]) < .@mats[.@k + 1]) {
				mes "^FF0000Not enough materials!^000000";
				close;
			}
		}
		if(Zeny < .@zeny) {
			mes "^FF0000Not enough zeny!^000000";
			close;
		}
		
		// Pay costs
		for(set .@k, 0; .@k < 8; .@k += 2)
			delitem .@mats[.@k], .@mats[.@k + 1];
		set Zeny, Zeny - .@zeny;
		
		// Attempt upgrade
		if(rand(100) < .@rate) {
			// Success - update slot 2
			set .@cards[1], .@next_id;
			
			// Recreate item with upgraded enchant
			delitem .@item_id, 1;
//			getitem2 .@item_id, 1, 1, .@refine, 0, 
//				.@cards[0], .@cards[1], .@cards[2], .@cards[3];
			getitem4 .@item_id, 1, 1, .@refine, 0, 
				.@cards[0], .@cards[1], .@cards[2], .@cards[3], .@grade, .@id, .@val, .@param;

			
			mes "^00FF00Upgrade successful!^000000";
		} else {
			// Failure (no downgrade)
			mes "^FF0000Upgrade failed!^000000";
			mes "The enchantment remains unchanged.";
		}
		close;
	case 6: // Reset Enchantments
		mes "Resetting an accessory will remove ALL enchantments and return it to its original state.";
		mes "Cost:";
		mes "30x ^0055FFMeteorite Powder^000000";
		mes "500,000 Zeny";
		mes "Success Rate: 100%";
		next;
		
		// Check materials
		if (countitem(1000372) < 30) {
			mes "^FF0000You need 30 Meteorite Powder!^000000";
			close;
		}
		if (Zeny < 500000) {
			mes "^FF0000You need 500,000 Zeny!^000000";
			close;
		}
		
		mes "Select accessory to reset:";
		next;
		set .@target, callfunc("F_SS_SelectAccessory");
		if (.@target < 0) close;
		
		// Get equipment details
		set .@item_id, getequipid(.@target);
		set .@refine, getequiprefinerycnt(.@target);
		set .@real_card, getequipcardid(.@target, 0);
		set .@grade, getenchantgrade(.@target);
		.@id = getequiprandomoption(.@target, 0, ROA_ID);
		.@val = getequiprandomoption(.@target, 0, ROA_VALUE);
		.@param = getequiprandomoption(.@target, 0, ROA_PARAM);

		
		// Pay the cost
		delitem 1000372, 30;
		set Zeny, Zeny - 500000;
		
		// Remove the old accessory
		delitem .@item_id, 1;
		
		// Create a new clean accessory (same refine, no cards)
//		getitem2 .@item_id, 1, 1, .@refine, 0, 0,0,0,0;
		getitem4 .@item_id, 1, 1, .@refine, 0, .@real_card, 0, 0, 0, .@grade, .@id, .@val, .@param;

		
		mes "The accessory has been reset and placed in your inventory.";
		mes "All enchantments have been removed!";
		close;
	}
	end;
}
// Custom accessory selection function
function	script	F_SS_SelectAccessory	{
	// Accessory slots: EQI_ACC_L (0), EQI_ACC_R (1)
	setarray .@acc_slots, 0, 1;
	.@count = 0;
	
	// Build menu of equipped accessories
	for (.@i = 0; .@i < 2; .@i++) {
		.@slot = .@acc_slots[.@i];
		.@item_id = getequipid(.@slot);
		
		// Skip empty slots
		if (.@item_id == 0) continue;
		
		// Check if it's a valid Stellar Seal accessory
		if (!callfunc("F_SS_IsValidAccessory", .@item_id)) continue;
		
		.@menu$ = .@menu$ + ((.@count == 0) ? "" : ":") + "Accessory " + (.@i+1) + ": " + getitemname(.@item_id);
		.@index[.@count] = .@slot;
		.@count++;
	}
	
	if (.@count == 0) {
		mes "You don't have any ^FF55FFStellar Seal accessories^000000 equipped.";
		return -1;
	}
	
	// Auto-select if only one accessory
	if (.@count == 1) {
		return .@index[0];
	}
	
	// Show selection menu
	.@s = select(.@menu$);
	return .@index[.@s-1];
}

// Helper function to check if item is valid accessory
function	script	F_SS_IsValidAccessory	{
	.@item_id = getarg(0);
	switch(.@item_id) {
		case 490132:
		case 490133:
		case 490134:
		case 490135:
		case 490136:
		case 490137:
			return 1;
		default:
			return 0;
	}
}
